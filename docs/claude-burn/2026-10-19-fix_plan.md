## 1. Dictionary workflow: normalize words and stop one judge from casting two votes (Shard 0 #2)

**Verdict: real, but narrower than reported.**
- `:92` only trims, so a generator that returns `Cafe` or `café` ends up in the candidate file. The backend loader then drops it, so the run is wasted.
- `:116` keys votes on the raw `v.word`, and `:120` matches exactly.
- The report missed a worse problem in the same lines. Votes are counted per verdict, not per judge. A single judge that lists a word twice reaches `>= 2` and passes the "both judges must agree" gate alone. Normalizing makes duplicates more likely (`Cafe` and `cafe` collapse), so I fixed both together.

**Test.** The file is a script, not a module. It has a top-level `return` and uses the globals `args`, `log`, `agent`, `pipeline` and `parallel`. The test therefore loads the source into an `AsyncFunction` with stubs. Save it as `.claude/workflows/dictionary-improvement.test.mjs` and run `node --test`.

```js
import test from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';

const AsyncFunction = Object.getPrototypeOf(async () => {}).constructor;
const src = readFileSync(new URL('./dictionary-improvement.js', import.meta.url), 'utf8')
  .replace(/^export const meta/m, 'const meta');

async function run({ lang = 'es', generated, judge }) {
  const persisted = [];
  const agent = async (prompt, opts) => {
    if (opts.label.startsWith('generate:')) return { words: generated };
    if (opts.label.startsWith('judge-')) {
      const words = JSON.parse(/Words: (\[.*\])\.\n/.exec(prompt)[1]);
      return { verdicts: judge(words, opts.label) };
    }
    if (opts.label.startsWith('persist:')) { persisted.push(prompt); return { added: 1, total: 1 }; }
    return 'done';
  };
  const pipeline = (items, ...stages) => Promise.all(items.map(async (item) => {
    let prev;
    for (const [i, s] of stages.entries()) prev = i === 0 ? await s(item) : await s(prev, item);
    return prev;
  }));
  const parallel = (fns) => Promise.all(fns.map((f) => f()));
  const fn = new AsyncFunction('args', 'log', 'phase', 'agent', 'pipeline', 'parallel', src);
  const out = await fn({ lang }, () => {}, () => {}, agent, pipeline, parallel);
  return { out, persisted };
}
const yes = (words) => words.map((word) => ({ word, valid: true, confidence: 0.9 }));

test('normalizes case/accents, keeps ñ, drops malformed, dedupes', async () => {
  const { out, persisted } = await run({
    generated: ['Café', 'cafe', 'Niño', 'hello world', 'perro ', 'PERRO', 'x1'],
    judge: yes,
  });
  assert.equal(out.results[0].kept, 3);
  const kept = JSON.parse(/exact match\): (\[.*\])\./.exec(persisted[0])[1]);
  assert.deepEqual(kept, ['cafe', 'niño', 'perro']);
});

test('judge echoing different casing still matches', async () => {
  const { out } = await run({ generated: ['perro', 'gato'], judge: (w) => yes(w.map((x) => x.toUpperCase())) });
  assert.equal(out.results[0].kept, 2);
});

test('one judge listing a word twice cannot pass the dual-judge gate', async () => {
  const { out, persisted } = await run({
    generated: ['perro'],
    judge: (w, label) => label.startsWith('judge-strict')
      ? [...yes(w), ...yes(w)]
      : w.map((word) => ({ word, valid: false, confidence: 0.9 })),
  });
  assert.equal(out.results[0].kept, 0);
  assert.equal(persisted.length, 0);
});
```
