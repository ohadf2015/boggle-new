/**
 * Remotion Cinematics Render Script
 *
 * Batch renders Remotion compositions to MP4 files.
 * Usage: npx tsx scripts/render-cinematics.ts [compositionId]
 *
 * If compositionId provided, renders only that composition.
 * If omitted, renders all configured cinematics.
 */

import { bundle } from '@remotion/bundler';
import { getCompositions, renderMedia, selectComposition } from '@remotion/renderer';
import * as path from 'path';
import * as fs from 'fs/promises';

interface RenderConfig {
  compositionId: string;
  outputName: string;
  props?: Record<string, unknown>;
}

// Cinematics to render when running full batch
const CINEMATICS_TO_RENDER: RenderConfig[] = [
  {
    compositionId: 'BossEntranceCinematic',
    outputName: 'boss-entrance.mp4',
    props: { bossId: 'world1-boss' },
  },
  {
    compositionId: 'VictoryCinematic',
    outputName: 'victory.mp4',
    props: { level: 1, stars: 3 },
  },
  {
    compositionId: 'DefeatCinematic',
    outputName: 'defeat.mp4',
  },
  // Add more compositions as needed
];

async function ensureOutputDir(): Promise<void> {
  const outputDir = path.join(process.cwd(), 'public', 'videos');
  try {
    await fs.mkdir(outputDir, { recursive: true });
  } catch {
    // Directory exists
  }
}

async function renderCinematics(targetCompositionId?: string): Promise<void> {
  console.log('📦 Bundling Remotion project...');

  // Find the Remotion entry point
  const entryPoint = path.join(process.cwd(), 'remotion', 'index.ts');

  // Check if entry point exists
  try {
    await fs.access(entryPoint);
  } catch {
    console.error('❌ Remotion entry point not found at:', entryPoint);
    console.error('   Expected: remotion/index.ts');
    console.error('   Make sure Remotion is set up correctly.');
    process.exit(1);
  }

  // Bundle the Remotion project
  const bundled = await bundle({
    entryPoint,
    webpackOverride: (config) => config,
  });

  console.log('✅ Bundle complete\n');

  // Get all available compositions
  const compositions = await getCompositions(bundled);
  console.log(`Found ${compositions.length} compositions:`);
  compositions.forEach((c) => console.log(`  - ${c.id}`));
  console.log('');

  // Ensure output directory exists
  await ensureOutputDir();

  // Filter to target composition if specified
  const configsToRender = targetCompositionId
    ? CINEMATICS_TO_RENDER.filter((c) => c.compositionId === targetCompositionId)
    : CINEMATICS_TO_RENDER;

  if (targetCompositionId && configsToRender.length === 0) {
    // If specific composition requested but not in config, try to render it anyway
    const composition = compositions.find((c) => c.id === targetCompositionId);
    if (composition) {
      configsToRender.push({
        compositionId: targetCompositionId,
        outputName: `${targetCompositionId.toLowerCase()}.mp4`,
      });
    } else {
      console.error(`❌ Composition not found: ${targetCompositionId}`);
      console.log('Available compositions:', compositions.map((c) => c.id).join(', '));
      process.exit(1);
    }
  }

  // Render each cinematic
  let successCount = 0;
  let failCount = 0;

  for (const config of configsToRender) {
    console.log(`🎬 Rendering: ${config.compositionId}...`);

    const composition = compositions.find((c) => c.id === config.compositionId);
    if (!composition) {
      console.error(`   ❌ Composition not found: ${config.compositionId}`);
      failCount++;
      continue;
    }

    const outputLocation = path.join(
      process.cwd(),
      'public',
      'videos',
      config.outputName
    );

    try {
      await renderMedia({
        codec: 'h264',
        composition,
        serveUrl: bundled,
        outputLocation,
        inputProps: config.props || {},
        // Quality settings
        crf: 23, // Quality (18-28 recommended, lower = higher quality)
        pixelFormat: 'yuv420p', // Compatibility with all browsers
        audioBitrate: '128k',
        videoBitrate: '2M',
        // Progress logging
        onProgress: ({ progress }) => {
          process.stdout.write(`\r   Progress: ${Math.round(progress * 100)}%`);
        },
      });

      console.log(`\n   ✅ Saved: ${outputLocation}`);

      // Get file size
      const stats = await fs.stat(outputLocation);
      const sizeMB = (stats.size / (1024 * 1024)).toFixed(2);
      console.log(`   📦 Size: ${sizeMB} MB\n`);

      successCount++;
    } catch (error) {
      console.error(`\n   ❌ Render failed:`, error);
      failCount++;
    }
  }

  // Summary
  console.log('═'.repeat(50));
  console.log(`🎉 Rendering complete!`);
  console.log(`   ✅ Success: ${successCount}`);
  if (failCount > 0) {
    console.log(`   ❌ Failed: ${failCount}`);
  }
  console.log(`   📁 Output: public/videos/`);
}

// Parse CLI args
const targetComposition = process.argv[2];

// Run
renderCinematics(targetComposition).catch((error) => {
  console.error('Fatal error:', error);
  process.exit(1);
});
