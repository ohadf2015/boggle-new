**Create useLexiStuckDetection hook (game-aware wrapper around useInactivityDetection):**

```typescript
// hooks/useLexiStuckDetection.ts
import { useCallback } from 'react';
import { useInactivityDetection } from './useInactivityDetection';

interface UseLexiStuckDetectionOptions {
  /** Called when player appears stuck (30s inactivity) */
  onStuck: () => void;
  /** Game is actively playing (not paused, not in modal) */
  isPlaying: boolean;
  /** Game is paused */
  isPaused: boolean;
  /** Modal is open (tutorial, hint, etc.) */
  isModalOpen?: boolean;
  /** Timeout in ms (default: 30000) */
  timeout?: number;
  /** Adjust timeout for boss levels (default: 45000) */
  isBossLevel?: boolean;
}

export function useLexiStuckDetection({
  onStuck,
  isPlaying,
  isPaused,
  isModalOpen = false,
  timeout = 30000,
  isBossLevel = false,
}: UseLexiStuckDetectionOptions) {
  // Disable detection when not actively playing
  const enabled = isPlaying && !isPaused && !isModalOpen;

  // Use longer timeout for boss levels (more thinking time)
  const effectiveTimeout = isBossLevel ? 45000 : timeout;

  const { reset } = useInactivityDetection({
    timeout: effectiveTimeout,
    onInactive: onStuck,
    enabled,
  });

  // Expose reset for game actions to call
  const resetOnGameAction = useCallback(() => {
    reset();
  }, [reset]);

  return {
    /** Call when player performs game action (word submit, tile click) */
    resetOnGameAction,
  };
}
```

**Create test file:**
```typescript
// hooks/__tests__/useLexiStuckDetection.test.ts
import { renderHook, act } from '@testing-library/react';
import { useLexiStuckDetection } from '../useLexiStuckDetection';

describe('useLexiStuckDetection', () => {
  beforeEach(() => {
    jest.useFakeTimers();
  });

  afterEach(() => {
    jest.useRealTimers();
  });

  it('triggers onStuck after 30s inactivity when playing', () => {
    const onStuck = jest.fn();
    renderHook(() => useLexiStuckDetection({
      onStuck,
      isPlaying: true,
      isPaused: false,
    }));

    act(() => {
      jest.advanceTimersByTime(30000);
    });

    expect(onStuck).toHaveBeenCalledTimes(1);
  });

  it('does not trigger when paused', () => {
    const onStuck = jest.fn();
    renderHook(() => useLexiStuckDetection({
      onStuck,
      isPlaying: true,
      isPaused: true,
    }));

    act(() => {
      jest.advanceTimersByTime(60000);
    });

    expect(onStuck).not.toHaveBeenCalled();
  });

  it('does not trigger when modal open', () => {
    const onStuck = jest.fn();
    renderHook(() => useLexiStuckDetection({
      onStuck,
      isPlaying: true,
      isPaused: false,
      isModalOpen: true,
    }));

    act(() => {
      jest.advanceTimersByTime(60000);
    });

    expect(onStuck).not.toHaveBeenCalled();
  });

  it('uses 45s timeout for boss levels', () => {
    const onStuck = jest.fn();
    renderHook(() => useLexiStuckDetection({
      onStuck,
      isPlaying: true,
      isPaused: false,
      isBossLevel: true,
    }));

    act(() => {
      jest.advanceTimersByTime(30000);
    });
    expect(onStuck).not.toHaveBeenCalled();

    act(() => {
      jest.advanceTimersByTime(15000);
    });
    expect(onStuck).toHaveBeenCalledTimes(1);
  });

  it('resetOnGameAction restarts timer', () => {
    const onStuck = jest.fn();
    const { result } = renderHook(() => useLexiStuckDetection({
      onStuck,
      isPlaying: true,
      isPaused: false,
    }));

    act(() => {
      jest.advanceTimersByTime(25000);
      result.current.resetOnGameAction();
    });

    act(() => {
      jest.advanceTimersByTime(25000);
    });

    expect(onStuck).not.toHaveBeenCalled();
  });
});
```
