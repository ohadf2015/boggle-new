```typescript
/**
 * VictoryCinematic Component
 *
 * Remotion composition for level victory celebration.
 * 6-second sequence with:
 * 1. Victory text burst (0-2s)
 * 2. Star reveal animation (2-4s)
 * 3. Stats display (3-6s)
 *
 * Uses Remotion primitives for frame-based animation.
 */

import React, { useMemo } from 'react';
import {
  AbsoluteFill,
  Sequence,
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
} from 'remotion';
import { useLanguage } from '@/contexts/LanguageContext';

// ==============================================
// CONSTANTS
// ==============================================

/** Frame rate for victory cinematic */
export const VICTORY_FPS = 30;

/** Total duration in frames (6 seconds at 30fps) */
export const VICTORY_DURATION_FRAMES = 180;

/** Phase timing in frames */
const PHASE_FRAMES = {
  TITLE_START: 0,
  TITLE_END: 60,
  STARS_START: 60,
  STARS_END: 120,
  STATS_START: 90,
  STATS_END: 180,
};

// ==============================================
// TYPES
// ==============================================

export interface VictoryCinematicProps {
  /** Level number */
  level: number;
  /** Stars earned (1-3) */
  stars: number;
  /** Final score */
  score: number;
  /** Time in seconds */
  time: number;
}

// ==============================================
// SUB-COMPONENTS
// ==============================================

/** Animated star component */
function StarIcon({ index, filled }: { index: number; filled: boolean }) {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  const scale = spring({
    frame: frame - (PHASE_FRAMES.STARS_START + index * 10),
    fps,
    config: {
      damping: 12,
      mass: 0.5,
      stiffness: 200,
    },
  });

  return (
    <div
      data-testid={`star-${index}`}
      className="text-6xl md:text-8xl"
      style={{
        transform: `scale(${scale})`,
        color: filled ? '#FFE135' : '#4a4a4a',
        textShadow: filled ? '4px 4px 0px black' : 'none',
      }}
    >
      ★
    </div>
  );
}

/** Stat row with animated reveal */
function StatRow({ label, value, delay }: { label: string; value: string | number; delay: number }) {
  const frame = useCurrentFrame();

  const opacity = interpolate(
    frame,
    [PHASE_FRAMES.STATS_START + delay, PHASE_FRAMES.STATS_START + delay + 10],
    [0, 1],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  const translateY = interpolate(
    frame,
    [PHASE_FRAMES.STATS_START + delay, PHASE_FRAMES.STATS_START + delay + 10],
    [20, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  return (
    <div
      className="flex justify-between gap-8 font-neo-body text-2xl text-white"
      style={{
        opacity,
        transform: `translateY(${translateY}px)`,
      }}
    >
      <span>{label}</span>
      <span className="text-neo-yellow font-bold">{value}</span>
    </div>
  );
}

// ==============================================
// MAIN COMPONENT
// ==============================================

/**
 * Victory cinematic Remotion composition
 *
 * @example
 * ```tsx
 * <CinematicPlayer
 *   composition={VictoryCinematic}
 *   compositionProps={{ level: 5, stars: 3, score: 15000, time: 45 }}
 *   durationSeconds={6}
 *   onComplete={handleComplete}
 * />
 * ```
 */
export const VictoryCinematic: React.FC<VictoryCinematicProps> = ({
  level,
  stars,
  score,
  time,
}) => {
  const { t } = useLanguage();
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  // Title animation
  const titleScale = spring({
    frame,
    fps,
    config: {
      damping: 10,
      mass: 0.8,
      stiffness: 150,
    },
  });

  const titleOpacity = interpolate(
    frame,
    [0, 15],
    [0, 1],
    { extrapolateRight: 'clamp' }
  );

  // Format time as MM:SS
  const formattedTime = useMemo(() => {
    const minutes = Math.floor(time / 60);
    const seconds = time % 60;
    return `${minutes}:${seconds.toString().padStart(2, '0')}`;
  }, [time]);

  return (
    <AbsoluteFill
      className="bg-neo-navy flex flex-col items-center justify-center"
      style={{ backgroundColor: '#1a1a2e' }}
    >
      {/* Sequence 1: Victory Title (0-2s) */}
      <Sequence from={PHASE_FRAMES.TITLE_START} durationInFrames={PHASE_FRAMES.STATS_END}>
        <div
          className="text-center"
          style={{
            opacity: titleOpacity,
            transform: `scale(${titleScale})`,
          }}
        >
          <h1
            className="font-neo-display text-6xl md:text-8xl lg:text-9xl text-neo-yellow font-black"
            style={{ textShadow: '6px 6px 0px black' }}
          >
            {t('adventure.victory.title')}
          </h1>
          <p className="font-neo-body text-2xl text-white mt-4">
            {t('adventure.victory.levelComplete')} {level}
          </p>
        </div>
      </Sequence>

      {/* Sequence 2: Stars (2-4s) */}
      <Sequence from={PHASE_FRAMES.STARS_START} durationInFrames={120}>
        <div className="flex gap-4 mt-8">
          {[0, 1, 2].map((index) => (
            <StarIcon key={index} index={index} filled={index < stars} />
          ))}
        </div>
      </Sequence>

      {/* Sequence 3: Stats (3-6s) */}
      <Sequence from={PHASE_FRAMES.STATS_START} durationInFrames={90}>
        <div className="mt-8 w-64 space-y-4">
          <StatRow label={t('adventure.victory.score')} value={score.toLocaleString()} delay={0} />
          <StatRow label={t('adventure.victory.time')} value={formattedTime} delay={10} />
        </div>
      </Sequence>
    </AbsoluteFill>
  );
};

VictoryCinematic.displayName = 'VictoryCinematic';

export default VictoryCinematic;

/**
 * DefeatCinematic Component
 *
 * Remotion composition for game over/defeat sequence.
 * 5-second encouraging sequence with:
 * 1. Defeat text (0-1.5s)
 * 2. Encouraging message (1-3s)
 * 3. Progress summary (2-5s)
 *
 * Tone: Encouraging, not punishing. Show what player achieved.
 */

import React from 'react';
import {
  AbsoluteFill,
  Sequence,
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
} from 'remotion';
import { useLanguage } from '@/contexts/LanguageContext';

// ==============================================
// CONSTANTS
// ==============================================

/** Frame rate for defeat cinematic */
export const DEFEAT_FPS = 30;

/** Total duration in frames (5 seconds at 30fps) */
export const DEFEAT_DURATION_FRAMES = 150;

/** Phase timing in frames */
const PHASE_FRAMES = {
  TITLE_START: 0,
  TITLE_END: 45,
  ENCOURAGE_START: 30,
  ENCOURAGE_END: 90,
  PROGRESS_START: 60,
  PROGRESS_END: 150,
};

// ==============================================
// TYPES
// ==============================================

export interface DefeatCinematicProps {
  /** Level number */
  level: number;
  /** Score achieved */
  score: number;
  /** Number of words found */
  wordsFound: number;
  /** Best word found (longest or highest score) */
  bestWord: string;
}

// ==============================================
// MAIN COMPONENT
// ==============================================

/**
 * Defeat cinematic Remotion composition
 *
 * @example
 * ```tsx
 * <CinematicPlayer
 *   composition={DefeatCinematic}
 *   compositionProps={{ level: 5, score: 8000, wordsFound: 12, bestWord: 'AMAZING' }}
 *   durationSeconds={5}
 *   onComplete={handleComplete}
 * />
 * ```
 */
export const DefeatCinematic: React.FC<DefeatCinematicProps> = ({
  level,
  score,
  wordsFound,
  bestWord,
}) => {
  const { t } = useLanguage();
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  // Title animation (gentle shake, not aggressive)
  const titleScale = spring({
    frame,
    fps,
    config: {
      damping: 15,
      mass: 1,
      stiffness: 100,
    },
  });

  const titleOpacity = interpolate(
    frame,
    [0, 15],
    [0, 1],
    { extrapolateRight: 'clamp' }
  );

  // Encouragement animation
  const encourageOpacity = interpolate(
    frame,
    [PHASE_FRAMES.ENCOURAGE_START, PHASE_FRAMES.ENCOURAGE_START + 15],
    [0, 1],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Progress animation
  const progressOpacity = interpolate(
    frame,
    [PHASE_FRAMES.PROGRESS_START, PHASE_FRAMES.PROGRESS_START + 15],
    [0, 1],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  return (
    <AbsoluteFill
      className="bg-neo-navy flex flex-col items-center justify-center"
      style={{ backgroundColor: '#1a1a2e' }}
    >
      {/* Sequence 1: Defeat Title (0-1.5s) */}
      <Sequence from={PHASE_FRAMES.TITLE_START} durationInFrames={150}>
        <div
          className="text-center"
          style={{
            opacity: titleOpacity,
            transform: `scale(${titleScale})`,
          }}
        >
          <h1
            className="font-neo-display text-5xl md:text-7xl text-neo-orange font-black"
            style={{ textShadow: '4px 4px 0px black' }}
          >
            {t('adventure.defeat.title')}
          </h1>
        </div>
      </Sequence>

      {/* Sequence 2: Encouraging Message (1-3s) */}
      <Sequence from={PHASE_FRAMES.ENCOURAGE_START} durationInFrames={120}>
        <div
          className="text-center mt-6"
          style={{ opacity: encourageOpacity }}
        >
          <p className="font-neo-body text-2xl text-white">
            {t('adventure.defeat.encouragement')}
          </p>
          <p className="font-neo-body text-lg text-gray-400 mt-2">
            {t('adventure.defeat.progress')}
          </p>
        </div>
      </Sequence>

      {/* Sequence 3: Progress Summary (2-5s) */}
      <Sequence from={PHASE_FRAMES.PROGRESS_START} durationInFrames={90}>
        <div
          className="mt-8 text-center space-y-4"
          style={{ opacity: progressOpacity }}
        >
          {/* Words found */}
          <div className="flex items-center justify-center gap-4">
            <span className="font-neo-body text-xl text-gray-300">
              {t('adventure.defeat.wordsFound', { defaultValue: 'Words Found' })}
            </span>
            <span className="font-neo-display text-3xl text-neo-cyan font-bold">
              {wordsFound}
            </span>
          </div>

          {/* Best word */}
          {bestWord && (
            <div className="flex items-center justify-center gap-4">
              <span className="font-neo-body text-xl text-gray-300">
                {t('adventure.defeat.bestWord', { defaultValue: 'Best Word' })}
              </span>
              <span className="font-neo-display text-2xl text-neo-lime font-bold uppercase">
                {bestWord}
              </span>
            </div>
          )}

          {/* Score */}
          <div className="flex items-center justify-center gap-4">
            <span className="font-neo-body text-xl text-gray-300">
              {t('adventure.defeat.score', { defaultValue: 'Score' })}
            </span>
            <span className="font-neo-display text-3xl text-neo-yellow font-bold">
              {score.toLocaleString()}
            </span>
          </div>
        </div>
      </Sequence>
    </AbsoluteFill>
  );
};

DefeatCinematic.displayName = 'DefeatCinematic';

export default DefeatCinematic;
