/**
 * Combo Flash Animation
 *
 * Screen flash effect for 10/15/20+ combo milestones.
 * Uses neo-yellow for brand consistency.
 * Respects prefers-reduced-motion.
 */

@keyframes combo-flash {
  0%, 100% {
    background-color: transparent;
  }
  50% {
    background-color: rgba(255, 225, 53, 0.3); /* neo-yellow at 30% */
  }
}

.combo-flash {
  animation: combo-flash 300ms ease-out;
  pointer-events: none;
}

/* Static feedback for reduced motion users */
.flash-feedback {
  background-color: rgba(255, 225, 53, 0.15);
  transition: background-color 100ms ease-out;
}

/* Accessibility: Disable flash animation for reduced motion */
@media (prefers-reduced-motion: reduce) {
  .combo-flash {
    animation: none;
    background-color: rgba(255, 225, 53, 0.15);
  }
}
```

```typescript
/**
 * ComboMilestoneOverlay Component
 *
 * Full-screen overlay for combo milestone celebrations.
 * Displays giant animated text (INCREDIBLE!/UNSTOPPABLE!/LEGENDARY!).
 * Uses Framer Motion for entrance/exit animations.
 */

'use client';

import { memo } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { useLanguage } from '@/contexts/LanguageContext';
import type { ComboMilestoneConfig } from '@/hooks/useComboMilestone';

// ==============================================
// TYPES
// ==============================================

export interface ComboMilestoneOverlayProps {
  /** Current active milestone (or null) */
  milestone: ComboMilestoneConfig | null;
}

// ==============================================
// ANIMATION VARIANTS
// ==============================================

const overlayVariants = {
  initial: { opacity: 0 },
  animate: { opacity: 1 },
  exit: { opacity: 0 },
};

const textVariants = {
  initial: {
    scale: 0.5,
    opacity: 0,
    rotate: -10,
  },
  animate: {
    scale: 1,
    opacity: 1,
    rotate: 0,
    transition: {
      type: 'spring',
      stiffness: 300,
      damping: 15,
    },
  },
  exit: {
    scale: 1.5,
    opacity: 0,
    transition: {
      duration: 0.3,
    },
  },
};

// ==============================================
// COMPONENT
// ==============================================

/**
 * Full-screen combo milestone overlay
 *
 * @example
 * ```tsx
 * const { currentMilestone } = useComboMilestone();
 * <ComboMilestoneOverlay milestone={currentMilestone} />
 * ```
 */
export const ComboMilestoneOverlay = memo(function ComboMilestoneOverlay({
  milestone,
}: ComboMilestoneOverlayProps) {
  const { t } = useLanguage();

  return (
    <AnimatePresence>
      {milestone && (
        <motion.div
          key={milestone.threshold}
          className="fixed inset-0 z-[9000] pointer-events-none flex items-center justify-center"
          variants={overlayVariants}
          initial="initial"
          animate="animate"
          exit="exit"
          data-testid="combo-milestone-overlay"
        >
          {/* Giant milestone text */}
          <motion.div
            className="
              font-neo-display text-neo-yellow
              text-6xl sm:text-7xl md:text-8xl lg:text-9xl
              font-black uppercase tracking-tight
              drop-shadow-[4px_4px_0px_rgba(0,0,0,1)]
              select-none
            "
            variants={textVariants}
            initial="initial"
            animate="animate"
            exit="exit"
          >
            {t(milestone.labelKey)}
          </motion.div>
        </motion.div>
      )}
    </AnimatePresence>
  );
});

ComboMilestoneOverlay.displayName = 'ComboMilestoneOverlay';

export default ComboMilestoneOverlay;
```
