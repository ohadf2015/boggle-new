export interface LayeredCelebrationBudget {
  combo: number;
  max: number;
  tier: 'low' | 'medium' | 'high';
  levelUp: number;
  word: number;
}

/**
 * Fire layered full-screen celebration for major milestones
 *
 * Layers:
 * - Background: Slow, large particles (20% of budget)
 * - Mid-ground: Medium particles, main burst (60% of budget)
 * - Foreground: Fast, small sparkles (20% of budget)
 */
export function fireLayeredCelebration(
  duration: number,
  budget: LayeredCelebrationBudget
): void {
  const totalParticles = Math.floor(budget.combo * 0.8);
  const bgParticles = Math.floor(totalParticles * 0.2);
  const midParticles = Math.floor(totalParticles * 0.6);
  const fgParticles = Math.floor(totalParticles * 0.2);

  // Layer 1: Background (slow, large, drifting)
  fireConfetti({
    particleCount: bgParticles,
    spread: 360,
    origin: { x: 0.5, y: 0.3 },
    colors: ['#FFE135', '#BFFF00'], // Yellow/lime
    scalar: 2.0,
    gravity: 0.5,
    ticks: 200,
    zIndex: Z_INDEX.BACKGROUND_PARTICLES,
  });

  // Layer 2: Mid-ground (100ms delay)
  setTimeout(() => {
    fireConfetti({
      particleCount: midParticles,
      spread: 120,
      origin: { x: 0.5, y: 0.5 },
      colors: ['#FF1493', '#00FFFF', '#FF3366'],
      scalar: 1.4,
      gravity: 1.0,
      ticks: 150,
      zIndex: Z_INDEX.MIDGROUND_PARTICLES,
    });
  }, 100);

  // Layer 3: Foreground (200ms delay)
  setTimeout(() => {
    fireConfetti({
      particleCount: fgParticles,
      spread: 90,
      origin: { x: 0.5, y: 0.6 },
      colors: ['#FFFFFF', '#FFE135'],
      scalar: 0.8,
      gravity: 1.5,
      ticks: 100,
      zIndex: Z_INDEX.FOREGROUND_PARTICLES,
      startVelocity: 60,
    });
  }, 200);
}

/**
 * useLayeredCelebration Hook
 *
 * Provides budget-aware layered particle celebrations.
 * Respects reduced motion preferences.
 */

import { useCallback } from 'react';
import { useParticleBudget } from './useParticleBudget';
import { usePrefersReducedMotion } from './usePrefersReducedMotion';
import { fireLayeredCelebration } from '@/utils/confettiUtils';

export interface UseLayeredCelebrationReturn {
  /** Trigger a layered celebration */
  fireCelebration: (duration: number) => void;
  /** Whether user prefers reduced motion */
  prefersReducedMotion: boolean;
  /** Current particle budget tier */
  tier: 'low' | 'medium' | 'high';
}

/**
 * Hook for triggering layered particle celebrations
 *
 * @example
 * ```tsx
 * const { fireCelebration, prefersReducedMotion } = useLayeredCelebration();
 *
 * // Fire celebration on milestone
 * if (!prefersReducedMotion) {
 *   fireCelebration(2000); // 2 second celebration
 * }
 * ```
 */
export function useLayeredCelebration(): UseLayeredCelebrationReturn {
  const budget = useParticleBudget();
  const prefersReducedMotion = usePrefersReducedMotion();

  const fireCelebration = useCallback((duration: number) => {
    // Skip if reduced motion preferred
    if (prefersReducedMotion) {
      return;
    }

    // Skip if no particle budget
    if (budget.combo === 0) {
      return;
    }

    fireLayeredCelebration(duration, budget);
  }, [budget, prefersReducedMotion]);

  return {
    fireCelebration,
    prefersReducedMotion,
    tier: budget.tier,
  };
}
