Component structure:
```typescript
export default function JoinClassroomForm() {
  const { t, language } = useLanguage();
  const router = useRouter();
  const { joinClassroom } = useJoinClassroom();
  const [joinCode, setJoinCode] = useState('');
  const [isJoining, setIsJoining] = useState(false);
  const isRTL = language === 'he';

  const handlePaste = async () => {
    try {
      const text = await navigator.clipboard.readText();
      setJoinCode(text.trim().toUpperCase().slice(0, 6));
    } catch {
      // Clipboard API not available
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!joinCode.trim() || joinCode.length !== 6) return;

    setIsJoining(true);
    const result = await joinClassroom(joinCode);
    setIsJoining(false);

    if (result.success) {
      toast.success(t('student.join.success'));
      router.push(`/${language}/student`);
    } else {
      toast.error(result.error || t('student.join.invalidCode'));
    }
  };

  return (
    <Card className="border-3 border-neo-black shadow-hard max-w-md mx-auto">
      <CardContent className="p-6 sm:p-8">
        <form onSubmit={handleSubmit} className="space-y-6">
          {/* Title */}
          <div className="text-center mb-6">
            <h2 className="text-2xl font-neo-display text-neo-white mb-2">
              {t('student.join.title')}
            </h2>
            <p className="text-neo-white/70 font-neo-body">
              {t('student.join.subtitle')}
            </p>
          </div>

          {/* Join Code Input */}
          <div className="space-y-2">
            <Label htmlFor="join-code" className="text-sm font-bold uppercase text-slate-400">
              {t('student.join.codeLabel')}
            </Label>
            <div className="relative">
              <Input
                id="join-code"
                value={joinCode}
                onChange={(e) => setJoinCode(e.target.value.toUpperCase().slice(0, 6))}
                placeholder="ABC123"
                maxLength={6}
                className="h-14 text-xl text-center font-mono font-bold tracking-widest uppercase pr-14 bg-neo-navy/50 border-neo-black/50 text-neo-white placeholder:text-slate-500"
              />
              <Button
                type="button"
                size="icon"
                variant="outline"
                onClick={handlePaste}
                className={cn(
                  'absolute top-1/2 -translate-y-1/2 h-10 w-10 bg-neo-cream text-neo-black hover:bg-neo-lime',
                  isRTL ? 'left-2' : 'right-2'
                )}
                aria-label={t('common.paste')}
              >
                <ClipboardPaste className="w-4 h-4" />
              </Button>
            </div>
            <p className="text-xs text-slate-400">
              {t('student.join.codeHint')}
            </p>
          </div>

          {/* Submit Button */}
          <Button
            type="submit"
            disabled={isJoining || joinCode.length !== 6}
            className="w-full h-14 text-lg font-black uppercase bg-neo-cyan hover:bg-neo-cyan/90 text-neo-black shadow-hard hover:shadow-hard-lg active:shadow-hard-pressed disabled:opacity-50"
          >
            <LogIn className={cn('w-5 h-5', isRTL ? 'ml-2' : 'mr-2')} />
            {isJoining ? t('common.loading') : t('student.join.button')}
          </Button>

          {/* Back Link */}
          <Button
            type="button"
            variant="ghost"
            onClick={() => router.push(`/${language}/student`)}
            className="w-full text-neo-white/70 hover:text-neo-white"
          >
            <ArrowLeft className={cn('w-4 h-4', isRTL ? 'ml-2 rotate-180' : 'mr-2')} />
            {t('common.back')}
          </Button>
        </form>
      </CardContent>
    </Card>
  );
}

/**
 * Student Join Classroom Page
 *
 * Allows students to join a teacher's classroom using a 6-character code
 */

'use client';

import { useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { useAuth } from '@/contexts/AuthContext';
import { useLanguage } from '@/contexts/LanguageContext';
import Header from '@/components/Header';
import { NeoLoader } from '@/components/ui/NeoLoader';
import JoinClassroomForm from '@/components/student/JoinClassroomForm';
import { cn } from '@/lib/utils';

export default function StudentJoinPage() {
  const { user, isAuthenticated, loading: authLoading } = useAuth();
  const { t, language } = useLanguage();
  const router = useRouter();
  const isRTL = language === 'he';

  // Redirect if not authenticated
  useEffect(() => {
    if (!authLoading && !isAuthenticated) {
      router.push(`/${language}`);
    }
  }, [authLoading, isAuthenticated, router, language]);

  // Show loading while checking auth
  if (authLoading) {
    return (
      <div className="flex-1 flex items-center justify-center bg-neo-navy min-h-screen">
        <NeoLoader variant="mascot-letters" size="lg" text={t('common.loading')} />
      </div>
    );
  }

  // Not authenticated - will redirect
  if (!isAuthenticated) {
    return null;
  }

  return (
    <div className={cn('flex-1 flex flex-col bg-neo-navy min-h-screen', isRTL && 'rtl')}>
      <Header />

      <main className="flex-1 flex items-center justify-center px-4 py-8">
        <JoinClassroomForm />
      </main>
    </div>
  );
}
```

Key points:
- Requires authentication (redirects to home if not signed in)
- Shows NeoLoader during auth check to prevent flash
- Centered JoinClassroomForm component
- RTL support for Hebrew
