/**
 * Get list of students in a classroom with their profile info
 */
export async function getClassroomStudents(
  classroomId: string
): Promise<{ data: ClassroomStudent[]; error: { message: string } | null }> {
  if (!supabase) return { data: [], error: { message: 'Supabase not configured' } };

  try {
    // Get memberships with student profile info
    const { data: memberships, error: memberError } = await supabase
      .from('classroom_memberships')
      .select(`
        id,
        student_id,
        joined_at,
        profiles:student_id (
          username,
          email,
          avatar_url
        )
      `)
      .eq('classroom_id', classroomId)
      .order('joined_at', { ascending: false });

    if (memberError) {
      logger.error('Error fetching classroom students:', memberError);
      return { data: [], error: { message: memberError.message } };
    }

    // Transform to expected shape
    const students: ClassroomStudent[] = (memberships || []).map((m) => ({
      id: m.id,
      student_id: m.student_id,
      joined_at: m.joined_at,
      student_profile: m.profiles as ClassroomStudent['student_profile'],
    }));

    return { data: students, error: null };
  } catch (err) {
    const error = err instanceof Error ? err.message : 'Unknown error';
    logger.error('Exception in getClassroomStudents:', error);
    return { data: [], error: { message: error } };
  }
}
```

Note: The `profiles:student_id` syntax creates a join where `student_id` references the `profiles` table's primary key.

/**
 * ClassroomStudentList Component
 *
 * Displays list of students who have joined a classroom
 */

'use client';

import { useState, useEffect } from 'react';
import { useLanguage } from '@/contexts/LanguageContext';
import { getClassroomStudents, type ClassroomStudent } from '@/lib/supabase/teacher';
import { NeoLoader } from '@/components/ui/NeoLoader';
import Avatar from '@/components/Avatar';
import { cn } from '@/lib/utils';
import { Users, Calendar } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';
import logger from '@/utils/logger';

interface ClassroomStudentListProps {
  classroomId: string;
  className?: string;
}

export default function ClassroomStudentList({
  classroomId,
  className,
}: ClassroomStudentListProps) {
  const { t, language } = useLanguage();
  const isRTL = language === 'he';

  const [students, setStudents] = useState<ClassroomStudent[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    async function fetchStudents() {
      setIsLoading(true);
      const { data, error } = await getClassroomStudents(classroomId);

      if (error) {
        logger.error('Error loading students:', error);
        setError(error.message);
      } else {
        setStudents(data);
        setError(null);
      }
      setIsLoading(false);
    }

    if (classroomId) {
      fetchStudents();
    }
  }, [classroomId]);

  if (isLoading) {
    return (
      <div className={cn('flex justify-center py-6', className)}>
        <NeoLoader variant="dots" size="sm" />
      </div>
    );
  }

  if (error) {
    return (
      <div className={cn('text-neo-pink text-center py-4', className)}>
        {t('teacher.classrooms.students.error')}
      </div>
    );
  }

  if (students.length === 0) {
    return (
      <div className={cn('text-center py-6', className)}>
        <Users className="w-12 h-12 mx-auto mb-3 text-neo-white/30" />
        <p className="text-neo-white/70 font-neo-body">
          {t('teacher.classrooms.students.empty')}
        </p>
        <p className="text-xs text-neo-white/50 mt-2">
          {t('teacher.classrooms.students.emptyHint')}
        </p>
      </div>
    );
  }

  return (
    <div className={cn('space-y-2', className)}>
      <div className="flex items-center gap-2 mb-3">
        <Users className="w-4 h-4 text-neo-cyan" />
        <span className="text-sm font-bold text-neo-white/70">
          {students.length} {t('teacher.classrooms.students.count')}
        </span>
      </div>

      <div className="divide-y divide-neo-black/30">
        {students.map((student) => (
          <div
            key={student.id}
            className={cn(
              'flex items-center gap-3 py-3',
              isRTL && 'flex-row-reverse'
            )}
          >
            {/* Avatar */}
            <Avatar
              avatarImage={student.student_profile?.avatar_url || 'default'}
              size="md"
              className="border-2 border-neo-black shrink-0"
            />

            {/* Student Info */}
            <div className={cn('flex-1 min-w-0', isRTL && 'text-right')}>
              <p className="font-bold text-neo-white truncate">
                {student.student_profile?.username || t('teacher.classrooms.students.unknown')}
              </p>
              {student.student_profile?.email && (
                <p className="text-xs text-neo-white/50 truncate">
                  {student.student_profile.email}
                </p>
              )}
            </div>

            {/* Join Date */}
            <div className={cn(
              'flex items-center gap-1 text-xs text-neo-white/50 shrink-0',
              isRTL && 'flex-row-reverse'
            )}>
              <Calendar className="w-3 h-3" />
              <span>
                {formatDistanceToNow(new Date(student.joined_at), { addSuffix: true })}
              </span>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
```

Key features:
- Fetches students on mount
- Shows avatar, username, email, join date
- Empty state with hint to share join code
- Loading and error states
- RTL support
