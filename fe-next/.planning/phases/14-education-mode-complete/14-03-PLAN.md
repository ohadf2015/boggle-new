'use client';

import { useCallback } from 'react';
import { assignLesson as assignLessonAPI, type LessonAssignment } from '@/lib/supabase/teacher';
import logger from '@/utils/logger';

interface UseAssignLessonReturn {
  assignLesson: (lessonId: string, classroomId: string, dueDate?: string) => Promise<{
    success: boolean;
    data?: LessonAssignment;
    error?: string;
  }>;
}

/**
 * Hook for assigning lessons to classrooms
 */
export function useAssignLesson(): UseAssignLessonReturn {
  const assignLesson = useCallback(async (
    lessonId: string,
    classroomId: string,
    dueDate?: string
  ): Promise<{ success: boolean; data?: LessonAssignment; error?: string }> => {
    try {
      const { data, error } = await assignLessonAPI(lessonId, classroomId, dueDate);

      if (error) {
        return { success: false, error: error.message };
      }

      return { success: true, data: data || undefined };
    } catch (err) {
      const error = err instanceof Error ? err.message : 'Failed to assign lesson';
      logger.error('Exception in useAssignLesson:', error);
      return { success: false, error };
    }
  }, []);

  return { assignLesson };
}
```

/**
 * LessonAssignmentDialog Component
 *
 * Dialog for assigning a lesson to one of the teacher's classrooms
 */

'use client';

import { useState } from 'react';
import * as Dialog from '@radix-ui/react-dialog';
import { useLanguage } from '@/contexts/LanguageContext';
import { useClassrooms } from '@/hooks/useClassroom';
import { useAssignLesson } from '@/hooks/useLessons';
import { Button } from '@/components/ui/button';
import { Label } from '@/components/ui/label';
import { NeoLoader } from '@/components/ui/NeoLoader';
import { X, Users, Check } from 'lucide-react';
import { cn } from '@/lib/utils';
import toast from 'react-hot-toast';

interface LessonAssignmentDialogProps {
  lessonId: string;
  lessonName: string;
  isOpen: boolean;
  onClose: () => void;
  onAssigned?: () => void;
}

export default function LessonAssignmentDialog({
  lessonId,
  lessonName,
  isOpen,
  onClose,
  onAssigned,
}: LessonAssignmentDialogProps) {
  const { t, language } = useLanguage();
  const isRTL = language === 'he';
  const { classrooms, isLoading: classroomsLoading } = useClassrooms();
  const { assignLesson } = useAssignLesson();

  const [selectedClassroom, setSelectedClassroom] = useState<string>('');
  const [isAssigning, setIsAssigning] = useState(false);

  const handleAssign = async () => {
    if (!selectedClassroom) {
      toast.error(t('teacher.lessons.assign.selectClassroom'));
      return;
    }

    setIsAssigning(true);
    const result = await assignLesson(lessonId, selectedClassroom);
    setIsAssigning(false);

    if (result.success) {
      toast.success(t('teacher.lessons.assign.success'));
      onAssigned?.();
      onClose();
      setSelectedClassroom('');
    } else {
      // Check for "already assigned" error
      if (result.error?.includes('already assigned')) {
        toast.error(t('teacher.lessons.assign.alreadyAssigned'));
      } else {
        toast.error(result.error || t('teacher.lessons.assign.error'));
      }
    }
  };

  return (
    <Dialog.Root open={isOpen} onOpenChange={(open) => !open && onClose()}>
      <Dialog.Portal>
        <Dialog.Overlay className="fixed inset-0 bg-neo-black/80 z-50" />
        <Dialog.Content
          className={cn(
            'fixed top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 z-50',
            'w-full max-w-md p-6 bg-neo-navy border-3 border-neo-black shadow-hard-lg',
            'focus:outline-none',
            isRTL && 'rtl'
          )}
        >
          {/* Header */}
          <div className="flex items-center justify-between mb-6">
            <Dialog.Title className="text-xl font-neo-display text-neo-white">
              {t('teacher.lessons.assign.title')}
            </Dialog.Title>
            <Dialog.Close asChild>
              <Button
                variant="ghost"
                size="icon"
                className="text-neo-white/70 hover:text-neo-white"
              >
                <X className="w-5 h-5" />
              </Button>
            </Dialog.Close>
          </div>

          {/* Lesson Info */}
          <div className="mb-6 p-3 bg-neo-cyan/10 border-2 border-neo-cyan/30 rounded-neo">
            <p className="text-sm text-neo-white/70 mb-1">
              {t('teacher.lessons.assign.lessonLabel')}
            </p>
            <p className="font-neo-body font-bold text-neo-white">{lessonName}</p>
          </div>

          {/* Classroom Selection */}
          <div className="space-y-3 mb-6">
            <Label className="text-sm font-bold uppercase text-slate-400">
              {t('teacher.lessons.assign.classroomLabel')}
            </Label>

            {classroomsLoading ? (
              <div className="flex justify-center py-4">
                <NeoLoader variant="dots" size="sm" />
              </div>
            ) : classrooms.length === 0 ? (
              <p className="text-neo-white/70 text-center py-4">
                {t('teacher.lessons.assign.noClassrooms')}
              </p>
            ) : (
              <div className="space-y-2 max-h-48 overflow-y-auto">
                {classrooms.map((classroom) => (
                  <button
                    key={classroom.id}
                    type="button"
                    onClick={() => setSelectedClassroom(classroom.id)}
                    className={cn(
                      'w-full flex items-center gap-3 p-3 rounded-neo border-2 transition-all',
                      selectedClassroom === classroom.id
                        ? 'border-neo-cyan bg-neo-cyan/20 text-neo-white'
                        : 'border-neo-black/50 bg-neo-navy/50 text-neo-white/70 hover:border-neo-cyan/50'
                    )}
                  >
                    <Users className="w-5 h-5 shrink-0" />
                    <div className={cn('flex-1 text-left', isRTL && 'text-right')}>
                      <p className="font-bold">{classroom.name}</p>
                      <p className="text-xs text-neo-white/50">
                        {classroom.member_count} {t('teacher.classrooms.students')}
                      </p>
                    </div>
                    {selectedClassroom === classroom.id && (
                      <Check className="w-5 h-5 text-neo-cyan" />
                    )}
                  </button>
                ))}
              </div>
            )}
          </div>

          {/* Actions */}
          <div className="flex gap-3">
            <Button
              variant="outline"
              onClick={onClose}
              className="flex-1 text-neo-white border-neo-black"
            >
              {t('common.cancel')}
            </Button>
            <Button
              onClick={handleAssign}
              disabled={!selectedClassroom || isAssigning || classrooms.length === 0}
              className="flex-1 bg-neo-cyan text-neo-black font-bold shadow-hard hover:shadow-hard-lg active:shadow-hard-pressed disabled:opacity-50"
            >
              {isAssigning ? t('common.loading') : t('teacher.lessons.assign.button')}
            </Button>
          </div>
        </Dialog.Content>
      </Dialog.Portal>
    </Dialog.Root>
  );
}
```

Key features:
- Radix Dialog for accessibility
- Classroom selection with visual feedback
- Loading states for classrooms and assignment
- Error handling with specific messages for "already assigned"
- RTL support
