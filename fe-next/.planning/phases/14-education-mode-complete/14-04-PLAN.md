```typescript
interface UseStudentProgressState {
  lessons: StudentLesson[];
  isLoading: boolean;
  error: string | null;
}

export function useStudentProgress(): UseStudentProgressState & { refresh: () => Promise<void> } {
  const { isAuthenticated, user } = useAuth();
  const isMounted = useMounted();

  const [state, setState] = useState<UseStudentProgressState>({
    lessons: [],
    isLoading: true,
    error: null,
  });

  const fetchData = useCallback(async () => {
    if (!isAuthenticated || !user) {
      setState({ lessons: [], isLoading: false, error: null });
      return;
    }

    try {
      // Fetch both assigned lessons and progress in parallel
      const [assignedResult, progressResult] = await Promise.all([
        getStudentAssignedLessons(user.id),
        getStudentProgress(user.id),
      ]);

      if (!isMounted.current) return;

      // Handle errors
      if (assignedResult.error) {
        setState({ lessons: [], isLoading: false, error: assignedResult.error.message });
        return;
      }

      // Build progress map by lesson_id
      const progressMap = new Map<string, StudentLessonProgress>();
      (progressResult.data || []).forEach((p) => {
        progressMap.set(p.lesson_id, p);
      });

      // Combine assigned lessons with their progress
      const combinedLessons: StudentLesson[] = assignedResult.data.map((item) => {
        const progress = progressMap.get(item.id);
        const isCompleted = progress?.completed_at !== null;
        const isStarted = !!progress;

        return {
          lesson: item,
          assignment: item.assignment,
          progress,
          status: isCompleted ? 'completed' : isStarted ? 'started' : 'assigned',
        };
      });

      // Sort: assigned (new) first, then started, then completed
      combinedLessons.sort((a, b) => {
        const order = { assigned: 0, started: 1, completed: 2 };
        return order[a.status] - order[b.status];
      });

      setState({
        lessons: combinedLessons,
        isLoading: false,
        error: null,
      });
    } catch (err) {
      logger.error('Error in useStudentProgress:', err);
      if (isMounted.current) {
        setState({
          lessons: [],
          isLoading: false,
          error: 'Failed to load lessons',
        });
      }
    }
  }, [isAuthenticated, user, isMounted]);

  const refresh = useCallback(async () => {
    setState((prev) => ({ ...prev, isLoading: true }));
    await fetchData();
  }, [fetchData]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  return { ...state, refresh };
}
```
