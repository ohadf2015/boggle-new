import { AbsoluteFill, useCurrentFrame, interpolate, spring, useVideoConfig, Sequence } from 'remotion';

interface WorldIntroProps {
  worldName: string;
  worldNumber: number;
  backgroundColor: string;
  accentColor: string;
}

export const WorldIntro: React.FC<WorldIntroProps> = ({
  worldName,
  worldNumber,
  backgroundColor,
  accentColor,
}) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  // Title animation: scale up with bounce
  const titleScale = spring({
    frame,
    fps,
    config: { damping: 12, stiffness: 100 },
  });

  // World number fade in
  const numberOpacity = interpolate(frame, [0, 15], [0, 1], {
    extrapolateRight: 'clamp',
  });

  // Background pulse effect
  const pulse = Math.sin(frame / 10) * 0.02 + 1;

  // Exit animation (starts at frame 120)
  const exitOpacity = interpolate(frame, [120, 150], [1, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  return (
    <AbsoluteFill
      style={{
        backgroundColor,
        justifyContent: 'center',
        alignItems: 'center',
        opacity: exitOpacity,
      }}
    >
      {/* Pulsing background decoration */}
      <div
        style={{
          position: 'absolute',
          width: 400 * pulse,
          height: 400 * pulse,
          borderRadius: '50%',
          backgroundColor: accentColor,
          opacity: 0.2,
        }}
      />

      {/* World number */}
      <Sequence from={0} durationInFrames={150}>
        <div
          style={{
            position: 'absolute',
            top: 150,
            fontSize: 200,
            fontFamily: 'Fredoka, sans-serif',
            color: accentColor,
            opacity: numberOpacity,
            textShadow: '8px 8px 0px black',
          }}
        >
          {worldNumber}
        </div>
      </Sequence>

      {/* World name with Neo-Brutalist styling */}
      <Sequence from={15} durationInFrames={135}>
        <h1
          style={{
            fontSize: 72,
            fontFamily: 'Fredoka, sans-serif',
            color: '#FFFFFF',
            transform: `scale(${titleScale})`,
            textShadow: '4px 4px 0px black',
            padding: '20px 40px',
            backgroundColor: accentColor,
            border: '4px solid black',
            marginTop: 200,
          }}
        >
          {worldName}
        </h1>
      </Sequence>
    </AbsoluteFill>
  );
};

// Default props for testing
WorldIntro.defaultProps = {
  worldName: 'Alphabet Meadows',
  worldNumber: 1,
  backgroundColor: '#1a1a2e',
  accentColor: '#FFE135',
};
```

**scripts/render-video.ts:**

#!/usr/bin/env tsx
/**
 * Video rendering script for Remotion compositions
 * Renders videos with options for different quality/size tradeoffs
 *
 * Usage:
 *   npx tsx scripts/render-video.ts WorldIntro --output=public/videos/world-1-intro.mp4
 *   npx tsx scripts/render-video.ts WorldIntro-2 --quality=medium
 */

import { bundle } from '@remotion/bundler';
import { renderMedia, selectComposition } from '@remotion/renderer';
import * as path from 'path';
import * as fs from 'fs/promises';

const ENTRY_POINT = path.join(process.cwd(), 'remotion', 'index.ts');
const DEFAULT_OUTPUT_DIR = 'public/videos';

interface RenderOptions {
  quality: 'high' | 'medium' | 'low';
  outputPath?: string;
}

const QUALITY_PRESETS = {
  high: { crf: 18, codec: 'h264' as const },
  medium: { crf: 23, codec: 'h264' as const },
  low: { crf: 28, codec: 'h264' as const },
};

async function renderVideo(
  compositionId: string,
  options: RenderOptions
): Promise<string> {
  const { quality, outputPath } = options;
  const preset = QUALITY_PRESETS[quality];

  console.log(`Bundling Remotion project...`);
  const bundled = await bundle({
    entryPoint: ENTRY_POINT,
    onProgress: (progress) => {
      if (progress % 25 === 0) {
        console.log(`  Bundle progress: ${progress}%`);
      }
    },
  });

  console.log(`Selecting composition: ${compositionId}`);
  const composition = await selectComposition({
    serveUrl: bundled,
    id: compositionId,
  });

  const finalOutputPath =
    outputPath ||
    path.join(DEFAULT_OUTPUT_DIR, `${compositionId.toLowerCase()}.mp4`);

  // Ensure output directory exists
  await fs.mkdir(path.dirname(finalOutputPath), { recursive: true });

  console.log(`Rendering video (quality: ${quality}, crf: ${preset.crf})...`);

  await renderMedia({
    composition,
    serveUrl: bundled,
    codec: preset.codec,
    outputLocation: finalOutputPath,
    crf: preset.crf,
    onProgress: ({ progress }) => {
      const percent = Math.round(progress * 100);
      if (percent % 10 === 0) {
        process.stdout.write(`\r  Render progress: ${percent}%`);
      }
    },
  });

  console.log(`\n✓ Rendered: ${finalOutputPath}`);

  // Get file size
  const stats = await fs.stat(finalOutputPath);
  const sizeMb = (stats.size / 1024 / 1024).toFixed(2);
  console.log(`  Size: ${sizeMb}MB`);

  return finalOutputPath;
}

async function main() {
  const args = process.argv.slice(2);

  if (args.length === 0 || args.includes('--help')) {
    console.log(`
Video Rendering Script

Usage:
  npx tsx scripts/render-video.ts <composition-id> [options]

Arguments:
  composition-id    ID of the Remotion composition to render

Options:
  --output=<path>   Output file path (default: public/videos/<id>.mp4)
  --quality=<level> Quality preset: high, medium, low (default: medium)
                    high: CRF 18 (larger files, best quality)
                    medium: CRF 23 (balanced)
                    low: CRF 28 (smaller files, lower quality)

Examples:
  npx tsx scripts/render-video.ts WorldIntro
  npx tsx scripts/render-video.ts WorldIntro --quality=high --output=out/world1.mp4
  npx tsx scripts/render-video.ts WorldIntro-2 --quality=low
    `);
    process.exit(0);
  }

  const compositionId = args.find((a) => !a.startsWith('--')) || '';
  const quality = (args.find((a) => a.startsWith('--quality='))?.split('=')[1] ||
    'medium') as RenderOptions['quality'];
  const outputPath = args.find((a) => a.startsWith('--output='))?.split('=')[1];

  if (!compositionId) {
    console.error('Error: Composition ID required');
    process.exit(1);
  }

  if (!['high', 'medium', 'low'].includes(quality)) {
    console.error('Error: Quality must be high, medium, or low');
    process.exit(1);
  }

  try {
    await renderVideo(compositionId, { quality, outputPath });
  } catch (error) {
    console.error('Render failed:', error instanceof Error ? error.message : error);
    process.exit(1);
  }
}

main();
```

# Video Delivery Strategy

## Decision: Pre-rendered Videos (MVP) with Lambda Path for Scale

### Chosen Approach

For MVP/Phase 1, we use **pre-rendered videos bundled with the app**:

1. Render videos locally using Remotion CLI
2. Store in `public/videos/` directory
3. Serve directly via Next.js/Vercel CDN
4. Videos cached at edge (fast global delivery)

### Why This Approach

| Factor | Pre-rendered | Remotion Lambda |
|--------|--------------|-----------------|
| Setup complexity | Low | Medium (AWS setup) |
| Cost | $0 | ~$3-5/month for 100 videos |
| Latency | Zero (cached) | 10-60s first render |
| Personalization | No | Yes |
| Bundle size impact | Yes (few MB) | No |

**MVP Reality:** We have ~12 videos (4 languages x 3 worlds). This is small enough to bundle.

### Video Specifications

- **Resolution:** 1920x1080 (scales down on mobile)
- **Duration:** 5 seconds (150 frames @ 30fps)
- **Codec:** H.264 (universal compatibility)
- **Quality:** CRF 23 (balanced size/quality)
- **Expected size:** ~2-5MB per video

### iOS Safari Compatibility

All videos MUST have these attributes:
```html
<video
  src="/videos/world-1-intro.mp4"
  autoPlay
  muted
  playsInline
/>
```

- `muted` - Required for autoplay on iOS
- `playsInline` - Prevents fullscreen takeover
- `autoPlay` - Standard attribute

**Important:** Videos should NOT have audio tracks at all. Even muted videos with audio tracks may be blocked.

### Multi-Language Strategy

**Current:** Separate videos per language (simple)
- `/videos/en/world-1-intro.mp4`
- `/videos/he/world-1-intro.mp4` (RTL support in composition)
- `/videos/sv/world-1-intro.mp4`
- `/videos/ja/world-1-intro.mp4`

**Future (if storage becomes concern):**
- Single video with dynamic text overlay
- Or Remotion Lambda for on-demand rendering

### Future: Remotion Lambda Migration

When to migrate:
- Need personalized videos (user stats, names)
- Video count exceeds 50+
- Bundle size impacts initial load time

Migration path:
1. Set up AWS account with Remotion IAM policy
2. Deploy Lambda function via `npx remotion lambda deploy`
3. Call render API from Next.js API route
4. Store results in Cloudflare R2 (zero egress cost)
5. Serve via CDN with long cache headers

Estimated Lambda cost: $0.02-0.05 per video render

### File Structure

```
public/videos/
├── .gitkeep
├── en/
│   ├── world-1-intro.mp4
│   ├── world-2-intro.mp4
│   └── world-3-intro.mp4
├── he/
│   └── ... (same structure)
├── sv/
│   └── ...
└── ja/
    └── ...
```

### Rendering Commands

```bash
# Render single video
npm run video:render WorldIntro --quality=medium

# Render all world intros
npm run video:render:all

# Custom output path
npm run video:render WorldIntro --output=public/videos/en/world-1-intro.mp4
```
