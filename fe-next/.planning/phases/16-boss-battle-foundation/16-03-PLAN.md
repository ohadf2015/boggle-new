```typescript
/**
 * BossOverlay - Compound component for boss battle UI
 *
 * Encapsulates all boss-related rendering:
 * - BossIntro modal
 * - BossHPBar
 * - BossDialogue taunts
 * - BossVictory/Defeat modals
 *
 * This keeps AdventureGame.tsx focused on game orchestration.
 */

'use client';

import React from 'react';
import type { BossConfig, BossHealthState } from '@/types/boss';
import BossIntro from '../BossIntro';
import BossHPBar from '../BossHPBar';
import BossDialogue from '../BossDialogue';
import BossVictory from '../BossVictory';

interface BossOverlayProps {
  bossConfig: BossConfig | null;
  healthState: BossHealthState | null;
  worldNumber: number;
  currentTaunt: string;
  showTaunt: boolean;
  gameState: {
    stars: number;
    score: number;
    wordsFound: number;
  };
  handlers: {
    onStart: () => void;
    onSkip: () => void;
    onContinue: () => void;
    onRetry: () => void;
  };
}

export function BossOverlay({
  bossConfig,
  healthState,
  worldNumber,
  currentTaunt,
  showTaunt,
  gameState,
  handlers,
}: BossOverlayProps) {
  if (!bossConfig || !healthState) return null;

  const { onStart, onSkip, onContinue, onRetry } = handlers;
  const isVictoryOrDefeat = healthState.phase === 'victory' || healthState.phase === 'defeat';

  return (
    <>
      {/* Boss Intro Modal */}
      {healthState.phase === 'intro' && (
        <BossIntro
          boss={bossConfig}
          worldNumber={worldNumber}
          onStart={onStart}
          onSkip={onSkip}
        />
      )}

      {/* Boss HP Bar - visible during active/enraged phases */}
      {(healthState.phase === 'active' || healthState.phase === 'enraged') && (
        <div className="mb-4">
          <BossHPBar
            boss={bossConfig}
            healthState={healthState}
            worldNumber={worldNumber}
          />
        </div>
      )}

      {/* Boss Dialogue */}
      <BossDialogue
        boss={bossConfig}
        currentTaunt={currentTaunt}
        isVisible={showTaunt}
        position="top"
      />

      {/* Boss Victory/Defeat Modal */}
      {isVictoryOrDefeat && (
        <BossVictory
          boss={bossConfig}
          isVictory={healthState.phase === 'victory'}
          stars={gameState.stars}
          score={gameState.score}
          wordsFound={gameState.wordsFound}
          gameState={gameState}
          onContinue={onContinue}
          onRetry={onRetry}
        />
      )}
    </>
  );
}

export default BossOverlay;
```
