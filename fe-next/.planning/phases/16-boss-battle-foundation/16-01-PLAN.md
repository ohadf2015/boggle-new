/**
 * Boss battle phases
 * - intro: Pre-battle cutscene showing
 * - active: Normal gameplay, player dealing damage
 * - enraged: Boss below 25% HP, mechanics intensify
 * - victory: Player defeated boss (HP reached 0)
 * - defeat: Player lost (timer expired)

export type BossPhase = 'intro' | 'active' | 'enraged' | 'victory' | 'defeat';

/**
 * Boss health state managed by useBossHealth hook
 */
export interface BossHealthState {
  /** Current boss HP (0 to maxHP) */
  currentHP: number;
  /** Maximum HP for this boss */
  maxHP: number;
  /** Current battle phase */
  phase: BossPhase;
  /** Total damage dealt this battle */
  totalDamageDealt: number;
  /** Whether boss battle is active (not in intro/victory/defeat) */
  isActive: boolean;
}

  /** Reset boss health to initial state */

```typescript
/**
 * useBossHealth Hook
 *
 * Manages boss HP tracking, damage calculation, and phase transitions.
 * Integrates with Phase 15 combo system for damage multipliers.
 */

'use client';

import { useState, useCallback, useMemo } from 'react';
import type { BossHealthState, BossPhase, UseBossHealthReturn } from '@/types/boss';

// ==============================================
// CONSTANTS
// ==============================================

/** Base HP for world 1 boss */
const BASE_HP = 500;

/** HP scaling per world (20% increase per world) */
const HP_WORLD_MULTIPLIER = 0.2;

/** Enraged threshold percentage (25%) */
const ENRAGED_THRESHOLD = 0.25;

/** Combo damage multiplier per combo count (10% per combo) */
const COMBO_DAMAGE_MULTIPLIER = 0.1;

// ==============================================
// HELPER FUNCTIONS
// ==============================================

/**
 * Calculate boss max HP based on world number
 * World 1: 500, World 3: 700, World 5: 900, World 10: 1400
 */
function calculateMaxHP(worldId: number): number {
  const worldMultiplier = 1 + (worldId - 1) * HP_WORLD_MULTIPLIER;
  return Math.round(BASE_HP * worldMultiplier);
}

/**
 * Calculate total damage with combo and mechanic multipliers
 */
function calculateDamage(
  baseDamage: number,
  comboCount: number,
  mechanicMultiplier: number
): number {
  const comboMultiplier = 1 + comboCount * COMBO_DAMAGE_MULTIPLIER;
  return Math.round(baseDamage * comboMultiplier * mechanicMultiplier);
}

/**
 * Determine boss phase based on HP percentage
 */
function determinePhase(
  currentHP: number,
  maxHP: number,
  currentPhase: BossPhase
): BossPhase {
  if (currentPhase === 'intro' || currentPhase === 'victory' || currentPhase === 'defeat') {
    return currentPhase;
  }

  if (currentHP <= 0) {
    return 'victory';
  }

  const hpPercentage = currentHP / maxHP;
  if (hpPercentage <= ENRAGED_THRESHOLD) {
    return 'enraged';
  }

  return 'active';
}

// ==============================================
// TYPES
// ==============================================

interface UseBossHealthProps {
  /** World ID to determine boss HP (null for non-boss levels) */
  worldId: number | null;
}

// ==============================================
// HOOK
// ==============================================

export function useBossHealth({
  worldId,
}: UseBossHealthProps): UseBossHealthReturn | null {
  // Return null for non-boss levels
  if (worldId === null) {
    return null;
  }

  const maxHP = useMemo(() => calculateMaxHP(worldId), [worldId]);

  const [healthState, setHealthState] = useState<BossHealthState>(() => ({
    currentHP: maxHP,
    maxHP,
    phase: 'intro',
    totalDamageDealt: 0,
    isActive: false,
  }));

  // Deal damage to boss
  const dealDamage = useCallback(
    (baseDamage: number, comboCount: number, mechanicMultiplier: number): number => {
      // Don't deal damage during non-active phases
      if (healthState.phase !== 'active' && healthState.phase !== 'enraged') {
        return 0;
      }

      const damage = calculateDamage(baseDamage, comboCount, mechanicMultiplier);

      setHealthState((prev) => {
        const newHP = Math.max(0, prev.currentHP - damage);
        const newPhase = determinePhase(newHP, prev.maxHP, prev.phase);

        return {
          ...prev,
          currentHP: newHP,
          phase: newPhase,
          totalDamageDealt: prev.totalDamageDealt + damage,
          isActive: newPhase === 'active' || newPhase === 'enraged',
        };
      });

      return damage;
    },
    [healthState.phase]
  );

  // Start the boss battle
  const startBattle = useCallback(() => {
    setHealthState((prev) => ({
      ...prev,
      phase: 'active',
      isActive: true,
    }));
  }, []);

  // End the battle
  const endBattle = useCallback((isVictory: boolean) => {
    setHealthState((prev) => ({
      ...prev,
      phase: isVictory ? 'victory' : 'defeat',
      isActive: false,
    }));
  }, []);

  // Reset to initial state
  const resetHealth = useCallback(() => {
    setHealthState({
      currentHP: maxHP,
      maxHP,
      phase: 'intro',
      totalDamageDealt: 0,
      isActive: false,
    });
  }, [maxHP]);

  // Computed values
  const hpPercentage = useMemo(
    () => Math.round((healthState.currentHP / healthState.maxHP) * 100),
    [healthState.currentHP, healthState.maxHP]
  );

  const isEnraged = useMemo(
    () => healthState.phase === 'enraged' || healthState.currentHP / healthState.maxHP <= ENRAGED_THRESHOLD,
    [healthState.phase, healthState.currentHP, healthState.maxHP]
  );

  return {
    healthState,
    dealDamage,
    startBattle,
    endBattle,
    resetHealth,
    hpPercentage,
    isEnraged,
  };
}
```
