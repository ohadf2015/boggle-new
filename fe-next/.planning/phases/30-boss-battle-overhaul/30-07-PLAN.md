// hooks/useCinematic.ts

/**
 * Cinematic Playback Hook
 *
 * Manages cinematic state with skip functionality.
 * Skip button unlocks after SKIP_DELAY_MS (2 seconds per BOSS-04).
 */

import { useState, useEffect, useCallback, useRef } from 'react';

/** Minimum time before skip is allowed (ms) */
export const SKIP_DELAY_MS = 2000;

/** Default cinematic duration (frames at 30fps) */
export const DEFAULT_DURATION_FRAMES = 240; // 8 seconds

export interface UseCinematicOptions {
  /** Duration in frames (at 30fps) */
  durationFrames?: number;
  /** Callback when cinematic ends (natural or skipped) */
  onComplete?: () => void;
  /** Callback when skip becomes available */
  onSkipAvailable?: () => void;
  /** Whether to auto-play on mount */
  autoPlay?: boolean;
}

export interface UseCinematicReturn {
  /** Whether cinematic is currently playing */
  isPlaying: boolean;
  /** Whether skip button should be visible */
  canSkip: boolean;
  /** Current frame number */
  currentFrame: number;
  /** Progress as percentage (0-100) */
  progress: number;
  /** Skip the cinematic */
  skip: () => void;
  /** Start playing */
  play: () => void;
  /** Pause playback */
  pause: () => void;
  /** Reset to beginning */
  reset: () => void;
  /** Callback for Player component */
  onFrameUpdate: (frame: number) => void;
}

export function useCinematic(options: UseCinematicOptions = {}): UseCinematicReturn {
  const {
    durationFrames = DEFAULT_DURATION_FRAMES,
    onComplete,
    onSkipAvailable,
    autoPlay = true,
  } = options;

  const [isPlaying, setIsPlaying] = useState(autoPlay);
  const [canSkip, setCanSkip] = useState(false);
  const [currentFrame, setCurrentFrame] = useState(0);

  const skipTimerRef = useRef<NodeJS.Timeout | null>(null);
  const hasCompletedRef = useRef(false);

  // Enable skip after delay
  useEffect(() => {
    if (isPlaying && !canSkip) {
      skipTimerRef.current = setTimeout(() => {
        setCanSkip(true);
        onSkipAvailable?.();
      }, SKIP_DELAY_MS);
    }

    return () => {
      if (skipTimerRef.current) {
        clearTimeout(skipTimerRef.current);
      }
    };
  }, [isPlaying, canSkip, onSkipAvailable]);

  // Handle completion
  const handleComplete = useCallback(() => {
    if (hasCompletedRef.current) return;
    hasCompletedRef.current = true;
    setIsPlaying(false);
    onComplete?.();
  }, [onComplete]);

  // Frame update callback
  const onFrameUpdate = useCallback(
    (frame: number) => {
      setCurrentFrame(frame);

      // Check for natural completion
      if (frame >= durationFrames - 1 && !hasCompletedRef.current) {
        handleComplete();
      }
    },
    [durationFrames, handleComplete]
  );

  // Skip function
  const skip = useCallback(() => {
    if (canSkip) {
      handleComplete();
    }
  }, [canSkip, handleComplete]);

  // Playback controls
  const play = useCallback(() => setIsPlaying(true), []);
  const pause = useCallback(() => setIsPlaying(false), []);

  const reset = useCallback(() => {
    setIsPlaying(false);
    setCanSkip(false);
    setCurrentFrame(0);
    hasCompletedRef.current = false;
  }, []);

  // Calculate progress
  const progress = Math.min(100, (currentFrame / durationFrames) * 100);

  return {
    isPlaying,
    canSkip,
    currentFrame,
    progress,
    skip,
    play,
    pause,
    reset,
    onFrameUpdate,
  };
}

// components/adventure/boss/cinematics/CinematicPlayer.tsx

/**
 * Cinematic Player Component
 *
 * Wraps Remotion Player with skip button and keyboard controls.
 * Skip unlocks after 2 seconds per BOSS-04 requirement.
 */

'use client';

import React, { useEffect, useCallback, ComponentType } from 'react';
import { Player, PlayerRef } from '@remotion/player';
import { motion, AnimatePresence } from 'framer-motion';
import { useCinematic, SKIP_DELAY_MS } from '../../../../hooks/useCinematic';
import { useLanguage } from '../../../../contexts/LanguageContext';

const FPS = 30;

export interface CinematicPlayerProps {
  /** Remotion composition component */
  composition: ComponentType<Record<string, unknown>>;
  /** Props to pass to the composition */
  compositionProps?: Record<string, unknown>;
  /** Duration in seconds */
  durationSeconds: number;
  /** Called when cinematic ends (natural or skipped) */
  onComplete: () => void;
  /** Width of the player */
  width?: number;
  /** Height of the player */
  height?: number;
  /** Whether to show as fullscreen overlay */
  fullscreen?: boolean;
}

export function CinematicPlayer({
  composition,
  compositionProps = {},
  durationSeconds,
  onComplete,
  width = 1280,
  height = 720,
  fullscreen = true,
}: CinematicPlayerProps) {
  const { t } = useLanguage();
  const durationFrames = Math.round(durationSeconds * FPS);

  const {
    isPlaying,
    canSkip,
    progress,
    skip,
    onFrameUpdate,
  } = useCinematic({
    durationFrames,
    onComplete,
    autoPlay: true,
  });

  const playerRef = React.useRef<PlayerRef>(null);

  // Keyboard skip (ESC key)
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape' && canSkip) {
        skip();
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [canSkip, skip]);

  // Sync player with hook state
  useEffect(() => {
    const player = playerRef.current;
    if (!player) return;

    if (isPlaying) {
      player.play();
    } else {
      player.pause();
    }
  }, [isPlaying]);

  const containerClasses = fullscreen
    ? 'fixed inset-0 z-50 bg-black flex items-center justify-center'
    : 'relative';

  return (
    <div className={containerClasses}>
      {/* Remotion Player */}
      <Player
        ref={playerRef}
        component={composition}
        inputProps={compositionProps}
        durationInFrames={durationFrames}
        compositionWidth={width}
        compositionHeight={height}
        fps={FPS}
        style={{
          width: fullscreen ? '100vw' : width,
          height: fullscreen ? '100vh' : height,
          maxWidth: '100%',
          maxHeight: '100%',
        }}
        autoPlay
        loop={false}
        showVolumeControls={false}
        controls={false}
      />

      {/* Skip Button */}
      <AnimatePresence>
        <motion.div
          className="absolute bottom-8 right-8"
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ delay: 0.5 }}
        >
          <button
            onClick={skip}
            disabled={!canSkip}
            className={`
              px-6 py-3 rounded-neo border-neo border-black
              font-neo-display text-lg
              transition-all duration-200
              ${
                canSkip
                  ? 'bg-neo-yellow hover:bg-neo-orange cursor-pointer shadow-hard hover:shadow-hard-lg'
                  : 'bg-gray-600 text-gray-400 cursor-not-allowed opacity-70'
              }
            `}
          >
            {canSkip ? (
              <span>
                {t('adventure.bosses.cinematics.skip')} (ESC)
              </span>
            ) : (
              <span className="flex items-center gap-2">
                <SkipCountdown />
              </span>
            )}
          </button>
        </motion.div>
      </AnimatePresence>

      {/* Progress Bar */}
      <div className="absolute bottom-0 left-0 right-0 h-1 bg-gray-800">
        <motion.div
          className="h-full bg-neo-yellow"
          style={{ width: `${progress}%` }}
          transition={{ duration: 0.1 }}
        />
      </div>
    </div>
  );
}

/** Countdown component for skip button */
function SkipCountdown() {
  const [countdown, setCountdown] = React.useState(Math.ceil(SKIP_DELAY_MS / 1000));

  useEffect(() => {
    const interval = setInterval(() => {
      setCountdown((c) => Math.max(0, c - 1));
    }, 1000);

    return () => clearInterval(interval);
  }, []);

  return <span>Skip in {countdown}...</span>;
}

// components/adventure/boss/cinematics/BossEntranceCinematic.tsx

/**
 * Boss Entrance Cinematic
 *
 * 5-10 second dramatic entrance sequence:
 * 1. Dark fade in (0-1s)
 * 2. Boss silhouette reveal (1-3s)
 * 3. Boss full reveal with particles (3-5s)
 * 4. Boss name title (5-7s)
 * 5. Battle ready transition (7-8s)
 */

import React from 'react';
import {
  AbsoluteFill,
  Sequence,
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
  Img,
} from 'remotion';

const FPS = 30;

export interface BossEntranceCinematicProps {
  /** Boss display name */
  bossName: string;
  /** Boss subtitle (e.g., "Guardian of World 1") */
  bossTitle?: string;
  /** Path to boss image */
  bossImagePath: string;
  /** Primary color for boss (hex) */
  primaryColor?: string;
  /** World number */
  worldNumber?: number;
}

export const BossEntranceCinematic: React.FC<BossEntranceCinematicProps> = ({
  bossName,
  bossTitle,
  bossImagePath,
  primaryColor = '#FFE135',
  worldNumber = 1,
}) => {
  const frame = useCurrentFrame();
  const { fps, width, height } = useVideoConfig();

  // Animation values
  const fadeIn = interpolate(frame, [0, 30], [0, 1], { extrapolateRight: 'clamp' });

  const silhouetteReveal = interpolate(frame, [30, 90], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  const bossReveal = spring({
    frame: frame - 90,
    fps,
    config: { damping: 15, stiffness: 100 },
  });

  const titleReveal = interpolate(frame, [150, 180], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  const zoomPulse = interpolate(
    frame,
    [180, 210, 240],
    [1, 1.05, 1],
    { extrapolateRight: 'clamp' }
  );

  return (
    <AbsoluteFill style={{ backgroundColor: '#0a0a1a' }}>
      {/* Background gradient */}
      <div
        style={{
          position: 'absolute',
          inset: 0,
          background: `radial-gradient(circle at 50% 50%, ${primaryColor}22, transparent 70%)`,
          opacity: fadeIn,
        }}
      />

      {/* Lightning flashes */}
      <Sequence from={15} durationInFrames={5}>
        <AbsoluteFill style={{ backgroundColor: 'white', opacity: 0.3 }} />
      </Sequence>
      <Sequence from={45} durationInFrames={3}>
        <AbsoluteFill style={{ backgroundColor: 'white', opacity: 0.2 }} />
      </Sequence>

      {/* Boss silhouette (dark) */}
      <Sequence from={30} durationInFrames={60}>
        <div
          style={{
            position: 'absolute',
            top: '50%',
            left: '50%',
            transform: `translate(-50%, -50%) scale(${0.5 + silhouetteReveal * 0.5})`,
            opacity: silhouetteReveal,
            filter: 'brightness(0)',
          }}
        >
          <Img
            src={bossImagePath}
            style={{
              maxWidth: 500,
              maxHeight: 500,
              objectFit: 'contain',
            }}
          />
        </div>
      </Sequence>

      {/* Boss full reveal */}
      <Sequence from={90}>
        <div
          style={{
            position: 'absolute',
            top: '50%',
            left: '50%',
            transform: `translate(-50%, -60%) scale(${bossReveal * zoomPulse})`,
            opacity: bossReveal,
          }}
        >
          <Img
            src={bossImagePath}
            style={{
              maxWidth: 500,
              maxHeight: 500,
              objectFit: 'contain',
              filter: `drop-shadow(0 0 30px ${primaryColor}66)`,
            }}
          />
        </div>
      </Sequence>

      {/* Boss name title */}
      <Sequence from={150}>
        <div
          style={{
            position: 'absolute',
            bottom: '15%',
            left: 0,
            right: 0,
            textAlign: 'center',
            opacity: titleReveal,
            transform: `translateY(${(1 - titleReveal) * 30}px)`,
          }}
        >
          <h1
            style={{
              fontFamily: 'Fredoka, sans-serif',
              fontSize: 72,
              fontWeight: 700,
              color: 'white',
              textShadow: `
                4px 4px 0 black,
                -2px -2px 0 black,
                0 0 30px ${primaryColor}
              `,
              letterSpacing: '0.05em',
              margin: 0,
            }}
          >
            {bossName}
          </h1>
          {bossTitle && (
            <p
              style={{
                fontFamily: 'Rubik, sans-serif',
                fontSize: 28,
                color: primaryColor,
                textShadow: '2px 2px 0 black',
                marginTop: 10,
              }}
            >
              {bossTitle}
            </p>
          )}
        </div>
      </Sequence>

      {/* World indicator */}
      <Sequence from={150}>
        <div
          style={{
            position: 'absolute',
            top: 40,
            left: 40,
            opacity: titleReveal,
          }}
        >
          <span
            style={{
              fontFamily: 'Fredoka, sans-serif',
              fontSize: 24,
              color: 'white',
              padding: '8px 16px',
              backgroundColor: '#00000088',
              border: '3px solid white',
            }}
          >
            WORLD {worldNumber}
          </span>
        </div>
      </Sequence>

      {/* Particle effects (decorative dots) */}
      <Sequence from={90}>
        <ParticleLayer
          count={20}
          color={primaryColor}
          frame={frame - 90}
          width={width}
          height={height}
        />
      </Sequence>
    </AbsoluteFill>
  );
};

/** Simple particle layer for decoration */
const ParticleLayer: React.FC<{
  count: number;
  color: string;
  frame: number;
  width: number;
  height: number;
}> = ({ count, color, frame, width, height }) => {
  const particles = React.useMemo(() => {
    return Array.from({ length: count }, (_, i) => ({
      x: Math.random() * width,
      y: Math.random() * height,
      size: 4 + Math.random() * 8,
      speed: 0.5 + Math.random() * 1.5,
      delay: Math.random() * 30,
    }));
  }, [count, width, height]);

  return (
    <>
      {particles.map((p, i) => {
        const adjustedFrame = Math.max(0, frame - p.delay);
        const opacity = interpolate(adjustedFrame, [0, 30], [0, 0.8], {
          extrapolateRight: 'clamp',
        });
        const yOffset = adjustedFrame * p.speed;

        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: p.x,
              top: p.y - yOffset,
              width: p.size,
              height: p.size,
              borderRadius: '50%',
              backgroundColor: color,
              opacity: opacity * (1 - (yOffset / height)),
              boxShadow: `0 0 ${p.size}px ${color}`,
            }}
          />
        );
      })}
    </>
  );
};

export default BossEntranceCinematic;

const ConfettiLayer: React.FC<{

  frame: number;
  width: number;
  height: number;
}> = ({ count, colors, frame, width, height }) => {
  const confetti = React.useMemo(() => {
    return Array.from({ length: count }, (_, i) => ({
      x: width / 2 + (Math.random() - 0.5) * 200,
      y: height / 2,
      vx: (Math.random() - 0.5) * 15,
      vy: -10 - Math.random() * 10,
      rotation: Math.random() * 360,
      rotationSpeed: (Math.random() - 0.5) * 20,
      size: 8 + Math.random() * 12,
      color: colors[Math.floor(Math.random() * colors.length)],
    }));
  }, [count, colors, width, height]);

  const gravity = 0.3;

  return (
    <>
      {confetti.map((p, i) => {
        const x = p.x + p.vx * frame;
        const y = p.y + p.vy * frame + 0.5 * gravity * frame * frame;
        const rotation = p.rotation + p.rotationSpeed * frame;
        const opacity = interpolate(frame, [0, 120], [1, 0], {
          extrapolateRight: 'clamp',
        });

        if (y > height + 50) return null;

        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: p.size,
              height: p.size * 0.6,
              backgroundColor: p.color,
              transform: `rotate(${rotation}deg)`,
              opacity,
              boxShadow: `0 0 ${p.size / 2}px ${p.color}88`,
            }}
          />
        );
      })}
    </>
  );
};
