// types/bossStateMachine.ts

/**
 * Boss State Machine Types
 *
 * 5-phase state machine: intro → phase1 → phase2 → enraged → victory/defeat
 * Phase transitions at HP thresholds: 66% (phase1→phase2), 33% (phase2→enraged)
 */

/** Context (extended state) for boss state machine */
export interface BossStateMachineContext {
  /** Current HP (0 to maxHP) */
  hp: number;
  /** Maximum HP for this boss */
  maxHP: number;
  /** Total damage dealt this battle */
  totalDamageDealt: number;
  /** Boss ID for ability lookup */
  bossId: string;
}

/** Events that trigger state transitions */
export type BossStateMachineEvent =
  | { type: 'START_BATTLE' }
  | { type: 'DEAL_DAMAGE'; amount: number }
  | { type: 'TIMER_EXPIRED' }
  | { type: 'RESET' };

/** Possible states in the boss state machine */
export type BossStateMachineState =
  | 'intro'
  | 'phase1'
  | 'phase2'
  | 'enraged'
  | 'victory'
  | 'defeat';

/** HP thresholds for phase transitions */
export const BOSS_PHASE_THRESHOLDS = {
  /** HP percentage below which boss enters phase2 */
  PHASE2_THRESHOLD: 66,
  /** HP percentage below which boss enters enraged */
  ENRAGED_THRESHOLD: 33,
} as const;

/** Return type for useBossStateMachine hook */
export interface UseBossStateMachineReturn {
  /** Current state value (intro, phase1, phase2, enraged, victory, defeat) */
  state: BossStateMachineState;
  /** Current context (hp, maxHP, etc.) */
  context: BossStateMachineContext;
  /** Send event to state machine */
  send: (event: BossStateMachineEvent) => void;
  /** Start the battle (transition from intro to phase1) */
  startBattle: () => void;
  /** Deal damage to boss */
  dealDamage: (amount: number) => void;
  /** End battle due to timer expiration (defeat) */
  timerExpired: () => void;
  /** Reset state machine to initial state */
  reset: () => void;
  /** HP as percentage (0-100) */
  hpPercentage: number;
  /** Whether battle is in active phase (phase1, phase2, or enraged) */
  isActive: boolean;
  /** Whether boss is enraged */
  isEnraged: boolean;
  /** Whether boss is defeated */
  isVictory: boolean;
  /** Whether player lost */
  isDefeat: boolean;
}

// hooks/useBossStateMachine.ts

/**
 * Boss State Machine Hook
 *
 * XState-powered 5-phase state machine for boss battles:
 * intro → phase1 → phase2 → enraged → victory/defeat
 *
 * Phase transitions at HP thresholds:
 * - 66%: phase1 → phase2
 * - 33%: phase2 → enraged
 * - 0%: enraged → victory
 */

import { useMemo, useCallback } from 'react';
import { createMachine, assign } from 'xstate';
import { useMachine } from '@xstate/react';
import type {
  BossStateMachineContext,
  BossStateMachineEvent,
  BossStateMachineState,
  UseBossStateMachineReturn,
  BOSS_PHASE_THRESHOLDS,
} from '../types/bossStateMachine';

/**
 * Create the XState machine for boss battles
 */
const createBossMachine = (maxHP: number, bossId: string) =>
  createMachine({
    id: 'boss',
    initial: 'intro' as const,
    context: {
      hp: maxHP,
      maxHP,
      totalDamageDealt: 0,
      bossId,
    } as BossStateMachineContext,
    states: {
      intro: {
        on: {
          START_BATTLE: { target: 'phase1' },
          RESET: { target: 'intro', actions: 'resetContext' },
        },
      },
      phase1: {
        on: {
          DEAL_DAMAGE: [
            // Victory if damage brings HP to 0
            {
              target: 'victory',
              guard: ({ context, event }) =>
                context.hp - event.amount <= 0,
              actions: 'dealDamage',
            },
            // Enraged if damage brings HP below 33%
            {
              target: 'enraged',
              guard: ({ context, event }) => {
                const newHP = Math.max(0, context.hp - event.amount);
                const newPercentage = (newHP / context.maxHP) * 100;
                return newPercentage < 33;
              },
              actions: 'dealDamage',
            },
            // Phase2 if damage brings HP below 66%
            {
              target: 'phase2',
              guard: ({ context, event }) => {
                const newHP = Math.max(0, context.hp - event.amount);
                const newPercentage = (newHP / context.maxHP) * 100;
                return newPercentage < 66;
              },
              actions: 'dealDamage',
            },
            // Stay in phase1 if HP still >= 66%
            {
              actions: 'dealDamage',
            },
          ],
          TIMER_EXPIRED: { target: 'defeat' },
          RESET: { target: 'intro', actions: 'resetContext' },
        },
      },
      phase2: {
        on: {
          DEAL_DAMAGE: [
            // Victory if damage brings HP to 0
            {
              target: 'victory',
              guard: ({ context, event }) =>
                context.hp - event.amount <= 0,
              actions: 'dealDamage',
            },
            // Enraged if damage brings HP below 33%
            {
              target: 'enraged',
              guard: ({ context, event }) => {
                const newHP = Math.max(0, context.hp - event.amount);
                const newPercentage = (newHP / context.maxHP) * 100;
                return newPercentage < 33;
              },
              actions: 'dealDamage',
            },
            // Stay in phase2 if HP still >= 33%
            {
              actions: 'dealDamage',
            },
          ],
          TIMER_EXPIRED: { target: 'defeat' },
          RESET: { target: 'intro', actions: 'resetContext' },
        },
      },
      enraged: {
        on: {
          DEAL_DAMAGE: [
            // Victory if damage brings HP to 0
            {
              target: 'victory',
              guard: ({ context, event }) =>
                context.hp - event.amount <= 0,
              actions: 'dealDamage',
            },
            // Stay in enraged
            {
              actions: 'dealDamage',
            },
          ],
          TIMER_EXPIRED: { target: 'defeat' },
          RESET: { target: 'intro', actions: 'resetContext' },
        },
      },
      victory: {
        type: 'final' as const,
        on: {
          RESET: { target: 'intro', actions: 'resetContext' },
        },
      },
      defeat: {
        type: 'final' as const,
        on: {
          RESET: { target: 'intro', actions: 'resetContext' },
        },
      },
    },
  }).provide({
    actions: {
      dealDamage: assign(({ context, event }) => {
        if (event.type !== 'DEAL_DAMAGE') return context;
        const damage = Math.min(event.amount, context.hp);
        return {
          hp: Math.max(0, context.hp - damage),
          totalDamageDealt: context.totalDamageDealt + damage,
        };
      }),
      resetContext: assign(({ context }) => ({
        hp: context.maxHP,
        totalDamageDealt: 0,
      })),
    },
  });

/**
 * Hook for managing boss battles with XState state machine
 *
 * @param maxHP - Maximum HP for this boss battle
 * @param bossId - Boss identifier for ability lookup
 * @returns Boss state machine state and control functions
 */
export function useBossStateMachine(
  maxHP: number,
  bossId: string
): UseBossStateMachineReturn {
  // Create machine instance (memoized to prevent recreation)
  const machine = useMemo(
    () => createBossMachine(maxHP, bossId),
    [maxHP, bossId]
  );

  // Use XState machine
  const [state, send] = useMachine(machine);

  // Current state value
  const currentState = state.value as BossStateMachineState;
  const context = state.context;

  // Derived state
  const hpPercentage = useMemo(() => {
    if (context.maxHP === 0) return 0;
    return Math.round((context.hp / context.maxHP) * 100);
  }, [context.hp, context.maxHP]);

  const isActive = useMemo(
    () =>
      currentState === 'phase1' ||
      currentState === 'phase2' ||
      currentState === 'enraged',
    [currentState]
  );

  const isEnraged = currentState === 'enraged';
  const isVictory = currentState === 'victory';
  const isDefeat = currentState === 'defeat';

  // Control functions
  const startBattle = useCallback(() => {
    send({ type: 'START_BATTLE' });
  }, [send]);

  const dealDamage = useCallback(
    (amount: number) => {
      send({ type: 'DEAL_DAMAGE', amount });
    },
    [send]
  );

  const timerExpired = useCallback(() => {
    send({ type: 'TIMER_EXPIRED' });
  }, [send]);

  const reset = useCallback(() => {
    send({ type: 'RESET' });
  }, [send]);

  return {
    state: currentState,
    context,
    send,
    startBattle,
    dealDamage,
    timerExpired,
    reset,
    hpPercentage,
    isActive,
    isEnraged,
    isVictory,
    isDefeat,
  };
}
