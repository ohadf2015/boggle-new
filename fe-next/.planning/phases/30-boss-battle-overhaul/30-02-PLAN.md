// components/adventure/boss/PhaseIndicator.tsx

/**
 * Phase Indicator Badge
 *
 * Shows current boss battle phase (PHASE 1, PHASE 2, ENRAGED)
 * Neo-brutalist styling with chunky borders and hard shadows
 */

'use client';

import { motion } from 'framer-motion';
import { useLanguage } from '../../../contexts/LanguageContext';
import type { BossStateMachineState } from '../../../types/bossStateMachine';

export interface PhaseIndicatorProps {
  /** Current phase from state machine */
  phase: BossStateMachineState;
  /** Optional className for styling overrides */
  className?: string;
}

/**
 * Get display text and color for each phase
 */
function getPhaseConfig(phase: BossStateMachineState) {
  switch (phase) {
    case 'phase1':
      return { label: 'adventure.bosses.phase1', color: 'bg-lime-400', number: 1 };
    case 'phase2':
      return { label: 'adventure.bosses.phase2', color: 'bg-yellow-400', number: 2 };
    case 'enraged':
      return { label: 'adventure.bosses.enraged', color: 'bg-red-500', number: 3 };
    case 'victory':
      return { label: 'adventure.bosses.victory', color: 'bg-neo-cyan', number: 0 };
    case 'defeat':
      return { label: 'adventure.bosses.defeat', color: 'bg-gray-500', number: 0 };
    default:
      return { label: '', color: 'bg-gray-400', number: 0 };
  }
}

export function PhaseIndicator({ phase, className = '' }: PhaseIndicatorProps) {
  const { t } = useLanguage();
  const config = getPhaseConfig(phase);

  // Don't show indicator in intro state
  if (phase === 'intro') {
    return null;
  }

  return (
    <motion.div
      initial={{ scale: 0.8, opacity: 0 }}
      animate={{ scale: 1, opacity: 1 }}
      className={`
        inline-flex items-center gap-2 px-3 py-1
        ${config.color}
        border-3 border-black rounded-neo
        shadow-hard font-neo-display font-bold text-black
        ${className}
      `}
      data-testid="phase-indicator"
    >
      {config.number > 0 && (
        <span className="text-xs uppercase tracking-wide">
          {t('adventure.bosses.phaseLabel')}
        </span>
      )}
      <span className="text-sm">
        {config.number > 0 ? config.number : t(config.label)}
      </span>
    </motion.div>
  );
}

// components/adventure/boss/SegmentedHPBar.tsx

/**
 * Segmented HP Bar
 *
 * 3-segment HP bar showing boss phase thresholds:
 * - Segment 1 (right): 66-100% HP (lime/green - Phase 1)
 * - Segment 2 (middle): 33-66% HP (yellow - Phase 2)
 * - Segment 3 (left): 0-33% HP (red - Enraged)
 *
 * Neo-brutalist styling with chunky borders, hard shadows, and smooth animations.
 */

'use client';

import { useMemo } from 'react';
import { motion } from 'framer-motion';
import { usePrefersReducedMotion } from '../../../hooks/usePrefersReducedMotion';
import type { BossStateMachineState } from '../../../types/bossStateMachine';

export interface SegmentedHPBarProps {
  /** Current HP (0 to maxHP) */
  currentHP: number;
  /** Maximum HP for this boss */
  maxHP: number;
  /** Current phase for styling */
  phase: BossStateMachineState;
  /** Boss display name (translation key) */
  bossName?: string;
  /** Optional className for styling overrides */
  className?: string;
}

interface HPSegment {
  id: string;
  minPercent: number;
  maxPercent: number;
  color: string;
  emptyColor: string;
}

const SEGMENTS: HPSegment[] = [
  { id: 'enraged', minPercent: 0, maxPercent: 33, color: 'bg-red-500', emptyColor: 'bg-red-900/30' },
  { id: 'phase2', minPercent: 33, maxPercent: 66, color: 'bg-yellow-400', emptyColor: 'bg-yellow-900/30' },
  { id: 'phase1', minPercent: 66, maxPercent: 100, color: 'bg-lime-400', emptyColor: 'bg-lime-900/30' },
];

/**
 * Calculate fill percentage for a segment based on current HP
 */
function calculateSegmentFill(segment: HPSegment, hpPercent: number): number {
  if (hpPercent >= segment.maxPercent) {
    return 100; // Segment fully filled
  }
  if (hpPercent <= segment.minPercent) {
    return 0; // Segment empty
  }
  // Partial fill
  const segmentRange = segment.maxPercent - segment.minPercent;
  const hpInSegment = hpPercent - segment.minPercent;
  return (hpInSegment / segmentRange) * 100;
}

export function SegmentedHPBar({
  currentHP,
  maxHP,
  phase,
  bossName,
  className = '',
}: SegmentedHPBarProps) {
  const prefersReducedMotion = usePrefersReducedMotion();

  // Calculate HP percentage
  const hpPercent = useMemo(() => {
    if (maxHP === 0) return 0;
    return Math.round((currentHP / maxHP) * 100);
  }, [currentHP, maxHP]);

  // Calculate fill for each segment
  const segmentFills = useMemo(() => {
    return SEGMENTS.map((segment) => ({
      ...segment,
      fill: calculateSegmentFill(segment, hpPercent),
    }));
  }, [hpPercent]);

  // Determine active segment border color
  const getSegmentBorderClass = (segmentId: string) => {
    if (phase === 'enraged' && segmentId === 'enraged') return 'ring-2 ring-red-300';
    if (phase === 'phase2' && segmentId === 'phase2') return 'ring-2 ring-yellow-300';
    if (phase === 'phase1' && segmentId === 'phase1') return 'ring-2 ring-lime-300';
    return '';
  };

  return (
    <div
      className={`
        w-full
        ${className}
      `}
      data-testid="segmented-hp-bar"
    >
      {/* HP Text */}
      <div className="flex justify-between items-center mb-1">
        {bossName && (
          <span className="font-neo-display font-bold text-white text-sm">
            {bossName}
          </span>
        )}
        <span
          className="font-neo-body text-white text-xs"
          data-testid="hp-text"
        >
          {currentHP} / {maxHP}
        </span>
      </div>

      {/* Segmented Bar Container */}
      <div
        className="
          flex gap-1
          p-1
          bg-neo-navy/80
          border-3 border-black
          rounded-neo
          shadow-hard
        "
      >
        {/* Render segments in reverse order (right to left: phase1, phase2, enraged) */}
        {[...segmentFills].reverse().map((segment) => (
          <div
            key={segment.id}
            className={`
              relative flex-1 h-6
              ${segment.emptyColor}
              rounded-sm
              overflow-hidden
              ${getSegmentBorderClass(segment.id)}
            `}
            data-testid={`hp-segment-${segment.id}`}
          >
            {/* Filled portion */}
            <motion.div
              className={`
                absolute inset-y-0 left-0
                ${segment.color}
              `}
              initial={false}
              animate={{
                width: `${segment.fill}%`,
              }}
              transition={
                prefersReducedMotion
                  ? { duration: 0 }
                  : { type: 'spring', stiffness: 300, damping: 30 }
              }
              data-testid={`hp-segment-fill-${segment.id}`}
            />

            {/* Segment divider lines (visual only) */}
            <div className="absolute inset-y-0 right-0 w-px bg-black/30" />
          </div>
        ))}
      </div>

      {/* Phase Thresholds Labels (optional, hidden on mobile) */}
      <div className="hidden sm:flex justify-between mt-1 px-1">
        <span className="text-[10px] text-red-400/70">0%</span>
        <span className="text-[10px] text-yellow-400/70">33%</span>
        <span className="text-[10px] text-lime-400/70">66%</span>
        <span className="text-[10px] text-lime-400/70">100%</span>
      </div>
    </div>
  );
}
