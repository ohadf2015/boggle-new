  <files>components/adventure/HintMessage.tsx, components/adventure/__tests__/HintMessage.test.tsx</files>
  <action>
    Create a component to display hint messages using translation keys.

    **HintMessage.tsx:**
    ```typescript
    import { useLanguage } from '@/contexts/LanguageContext';
    import type { HintData } from '@/lib/adaptiveDifficulty';

    interface HintMessageProps {
      hintData: HintData;
      className?: string;
    }

    export function HintMessage({ hintData, className }: HintMessageProps) {
      const { t } = useLanguage();

      // Don't render if no hint
      if (hintData.level === 'none') {
        return null;
      }

      // Get translation key based on hint level
      const translationKey = `difficulty.hint.${hintData.level}`;

      // Build translation params based on hint data
      const params: Record<string, string | number> = {};

      if (hintData.message) {
        // Parse message for params (e.g., "length:5" or "letter:A")
        // Or use hintData fields directly
        if (hintData.highlightTiles && hintData.highlightTiles.length > 0) {
          params.length = hintData.highlightTiles.length;
          // First tile letter would come from grid data
        }
      }

      return (
        <div
          className={`
            bg-neo-navy/90 border-neo border-neo-yellow
            rounded-neo px-4 py-2 text-neo-yellow
            font-neo-body text-sm animate-neo-pop
            ${className ?? ''}
          `}
        >
          {t(translationKey, params)}
        </div>
      );
    }
    ```

    **Tests:**
    - Returns null when hintData.level is 'none'
    - Renders hint message for 'length' level
    - Renders hint message for 'lengthAndStart' level
    - Renders hint message for 'fullReveal' level
    - Applies custom className
    - Uses correct translation keys
  </action>
