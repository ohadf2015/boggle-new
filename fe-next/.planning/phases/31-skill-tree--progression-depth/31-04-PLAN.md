```typescript
/**
 * SkillNode Component
 *
 * Displays a single skill in the tree with locked/unlocked/available states.
 * Neo-brutalist design with tier-colored borders and hard shadows.
 */

'use client';

import React, { memo } from 'react';
import { motion } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import type { SkillNode as SkillNodeType, SkillTreeState } from '@/types/adventure';
import { canUnlockSkill } from '@/utils/skillTreeUtils';

// ==============================================
// TYPES
// ==============================================

export interface SkillNodeProps {
  /** Skill data */
  skill: SkillNodeType;
  /** Current skill tree state */
  state: SkillTreeState;
  /** Callback when skill is clicked */
  onSelect: (skill: SkillNodeType) => void;
  /** Whether this skill is currently selected */
  isSelected?: boolean;
}

// ==============================================
// CONSTANTS
// ==============================================

const PATH_COLORS = {
  power: {
    bg: 'bg-neo-orange/20',
    border: 'border-neo-orange',
    text: 'text-neo-orange',
    glow: 'shadow-[0_0_10px_rgba(255,107,53,0.5)]',
  },
  strategy: {
    bg: 'bg-neo-cyan/20',
    border: 'border-neo-cyan',
    text: 'text-neo-cyan',
    glow: 'shadow-[0_0_10px_rgba(0,255,255,0.5)]',
  },
  utility: {
    bg: 'bg-neo-yellow/20',
    border: 'border-neo-yellow',
    text: 'text-neo-yellow',
    glow: 'shadow-[0_0_10px_rgba(255,225,53,0.5)]',
  },
} as const;

// ==============================================
// COMPONENT
// ==============================================

const SkillNode = memo<SkillNodeProps>(({ skill, state, onSelect, isSelected = false }) => {
  const { t } = useLanguage();

  // Determine skill state
  const isUnlocked = state.unlockedSkills.has(skill.id);
  const canUnlock = canUnlockSkill(skill.id, state);
  const isLocked = !isUnlocked && !canUnlock;

  // Get path colors
  const colors = PATH_COLORS[skill.path];

  // Status for display
  const statusKey = isUnlocked
    ? 'adventure.skills.status.unlocked'
    : canUnlock
    ? 'adventure.skills.status.available'
    : 'adventure.skills.status.locked';

  return (
    <motion.button
      onClick={() => onSelect(skill)}
      disabled={isLocked}
      className={cn(
        'relative w-full p-3',
        'flex flex-col items-center gap-2',
        'border-3 rounded-neo',
        'transition-all duration-200',
        // Base styles
        'bg-neo-navy',
        colors.border,
        // State-based styling
        isUnlocked && [colors.bg, colors.glow],
        canUnlock && 'hover:scale-105 hover:shadow-hard cursor-pointer',
        isLocked && 'opacity-50 cursor-not-allowed grayscale',
        isSelected && 'ring-2 ring-neo-white ring-offset-2 ring-offset-neo-navy',
        // Focus styles
        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-neo-cyan'
      )}
      data-testid={`skill-node-${skill.id}`}
      data-state={isUnlocked ? 'unlocked' : canUnlock ? 'available' : 'locked'}
      whileHover={canUnlock ? { scale: 1.05 } : {}}
      whileTap={canUnlock ? { scale: 0.95 } : {}}
    >
      {/* Icon */}
      <span className="text-3xl" role="img" aria-hidden="true">
        {skill.icon}
      </span>

      {/* Name */}
      <span
        className={cn(
          'text-sm font-bold text-center leading-tight',
          isUnlocked ? colors.text : 'text-neo-white/80'
        )}
      >
        {t(skill.nameKey)}
      </span>

      {/* Cost badge (only show if not unlocked) */}
      {!isUnlocked && (
        <span
          className={cn(
            'absolute -top-2 -right-2',
            'px-2 py-0.5',
            'text-xs font-bold',
            'bg-neo-navy border-2 rounded-full',
            colors.border,
            colors.text
          )}
        >
          {skill.cost}
        </span>
      )}

      {/* Unlocked checkmark */}
      {isUnlocked && (
        <span
          className={cn(
            'absolute -top-2 -right-2',
            'w-6 h-6',
            'flex items-center justify-center',
            'bg-neo-green border-2 border-neo-black rounded-full',
            'text-xs'
          )}
          role="img"
          aria-label={t('adventure.skills.status.unlocked')}
        >
          ✓
        </span>
      )}
    </motion.button>
  );
});

SkillNode.displayName = 'SkillNode';

export { SkillNode };
export default SkillNode;
```

```typescript
/**
 * SkillPath Component
 *
 * Displays a vertical path of skills (Power, Strategy, or Utility).
 * Shows path header with icon and skills organized by tier.
 */

'use client';

import React, { memo } from 'react';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import type { SkillNode as SkillNodeType, SkillPath as SkillPathType, SkillTreeState } from '@/types/adventure';
import { SkillNode } from './SkillNode';
import { getSkillsByPath } from '@/utils/skillTreeUtils';

// ==============================================
// TYPES
// ==============================================

export interface SkillPathProps {
  /** Which path to display */
  path: SkillPathType;
  /** Current skill tree state */
  state: SkillTreeState;
  /** Callback when a skill is selected */
  onSelectSkill: (skill: SkillNodeType) => void;
  /** Currently selected skill ID */
  selectedSkillId?: string;
}

// ==============================================
// CONSTANTS
// ==============================================

const PATH_CONFIG = {
  power: {
    icon: '⚔️',
    headerClass: 'bg-neo-orange/20 border-neo-orange text-neo-orange',
  },
  strategy: {
    icon: '🧠',
    headerClass: 'bg-neo-cyan/20 border-neo-cyan text-neo-cyan',
  },
  utility: {
    icon: '🔧',
    headerClass: 'bg-neo-yellow/20 border-neo-yellow text-neo-yellow',
  },
} as const;

// ==============================================
// COMPONENT
// ==============================================

const SkillPath = memo<SkillPathProps>(({
  path,
  state,
  onSelectSkill,
  selectedSkillId,
}) => {
  const { t } = useLanguage();
  const config = PATH_CONFIG[path];

  // Get skills for this path, sorted by tier
  const skills = getSkillsByPath(path);

  // Group by tier
  const tier1 = skills.filter(s => s.tier === 1);
  const tier2 = skills.filter(s => s.tier === 2);
  const tier3 = skills.filter(s => s.tier === 3);

  return (
    <div
      className="flex flex-col gap-4"
      data-testid={`skill-path-${path}`}
    >
      {/* Path Header */}
      <div
        className={cn(
          'flex items-center justify-center gap-2',
          'px-4 py-2',
          'border-3 rounded-neo',
          config.headerClass
        )}
      >
        <span className="text-xl" role="img" aria-hidden="true">
          {config.icon}
        </span>
        <span className="font-bold text-sm uppercase tracking-wide">
          {t(`adventure.skills.paths.${path}`)}
        </span>
      </div>

      {/* Tier 1 Skills */}
      <div className="flex flex-col gap-2">
        {tier1.map((skill) => (
          <SkillNode
            key={skill.id}
            skill={skill}
            state={state}
            onSelect={onSelectSkill}
            isSelected={selectedSkillId === skill.id}
          />
        ))}
      </div>

      {/* Connector Line */}
      {tier2.length > 0 && (
        <div className="flex justify-center">
          <div className="w-0.5 h-4 bg-neo-white/30" />
        </div>
      )}

      {/* Tier 2 Skills */}
      {tier2.length > 0 && (
        <div className="flex flex-col gap-2">
          {tier2.map((skill) => (
            <SkillNode
              key={skill.id}
              skill={skill}
              state={state}
              onSelect={onSelectSkill}
              isSelected={selectedSkillId === skill.id}
            />
          ))}
        </div>
      )}

      {/* Connector Line */}
      {tier3.length > 0 && (
        <div className="flex justify-center">
          <div className="w-0.5 h-4 bg-neo-white/30" />
        </div>
      )}

      {/* Tier 3 Skills */}
      {tier3.length > 0 && (
        <div className="flex flex-col gap-2">
          {tier3.map((skill) => (
            <SkillNode
              key={skill.id}
              skill={skill}
              state={state}
              onSelect={onSelectSkill}
              isSelected={selectedSkillId === skill.id}
            />
          ))}
        </div>
      )}
    </div>
  );
});

SkillPath.displayName = 'SkillPath';

export { SkillPath };
export default SkillPath;
```

```typescript
// SkillTreeView.tsx
/**
 * SkillTreeView Component
 *
 * Main skill tree visualization with 3 paths side-by-side.
 * Includes skill detail panel and unlock confirmation.
 * Mobile-friendly with horizontal scroll on small screens.
 */

'use client';

import React, { memo, useState, useCallback } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import { useSkillTreeStore } from '@/hooks/useSkillTreeStore';
import { SkillPath } from './SkillPath';
import { canUnlockSkill, getSkillById } from '@/utils/skillTreeUtils';
import type { SkillNode as SkillNodeType, SkillTreeState } from '@/types/adventure';

// ==============================================
// TYPES
// ==============================================

export interface SkillTreeViewProps {
  /** Callback when a skill is unlocked */
  onSkillUnlock?: (skill: SkillNodeType) => void;
  /** Optional class name */
  className?: string;
}

// ==============================================
// COMPONENT
// ==============================================

const SkillTreeView = memo<SkillTreeViewProps>(({ onSkillUnlock, className }) => {
  const { t } = useLanguage();
  const [selectedSkillId, setSelectedSkillId] = useState<string | null>(null);

  // Store integration
  const unlockedSkills = useSkillTreeStore((state) => state.unlockedSkills);
  const availablePoints = useSkillTreeStore((state) => state.availablePoints);
  const totalPointsEarned = useSkillTreeStore((state) => state.totalPointsEarned);
  const unlockSkill = useSkillTreeStore((state) => state.unlockSkill);

  // Build state object for components
  const state: SkillTreeState = {
    unlockedSkills,
    availablePoints,
    totalPointsEarned,
  };

  // Get selected skill details
  const selectedSkill = selectedSkillId ? getSkillById(selectedSkillId) : null;
  const canUnlockSelected = selectedSkill
    ? canUnlockSkill(selectedSkill.id, state)
    : false;

  // Handle skill selection
  const handleSelectSkill = useCallback((skill: SkillNodeType) => {
    setSelectedSkillId((current) => (current === skill.id ? null : skill.id));
  }, []);

  // Handle unlock
  const handleUnlock = useCallback(() => {
    if (!selectedSkill || !canUnlockSelected) return;

    const success = unlockSkill(selectedSkill.id, selectedSkill.cost);
    if (success) {
      onSkillUnlock?.(selectedSkill);
      setSelectedSkillId(null);
    }
  }, [selectedSkill, canUnlockSelected, unlockSkill, onSkillUnlock]);

  return (
    <div
      className={cn('flex flex-col gap-6', className)}
      data-testid="skill-tree-view"
    >
      {/* Header with points display */}
      <div className="flex items-center justify-between">
        <h2 className="text-2xl font-black text-neo-white">
          {t('adventure.skills.title')}
        </h2>
        <div
          className={cn(
            'px-4 py-2',
            'bg-neo-navy border-3 border-neo-yellow rounded-neo',
            'text-neo-yellow font-bold'
          )}
          data-testid="skill-points-display"
        >
          {t('adventure.skills.points_available', { count: availablePoints })}
        </div>
      </div>

      {/* Skill Tree Grid */}
      <div
        className={cn(
          'grid grid-cols-1 md:grid-cols-3 gap-6',
          'overflow-x-auto pb-4',
          // Mobile: horizontal scroll with snap
          'snap-x snap-mandatory md:snap-none',
          'scrollbar-thin scrollbar-thumb-neo-white/20 scrollbar-track-transparent'
        )}
      >
        <div className="snap-start min-w-[280px]">
          <SkillPath
            path="power"
            state={state}
            onSelectSkill={handleSelectSkill}
            selectedSkillId={selectedSkillId ?? undefined}
          />
        </div>
        <div className="snap-start min-w-[280px]">
          <SkillPath
            path="strategy"
            state={state}
            onSelectSkill={handleSelectSkill}
            selectedSkillId={selectedSkillId ?? undefined}
          />
        </div>
        <div className="snap-start min-w-[280px]">
          <SkillPath
            path="utility"
            state={state}
            onSelectSkill={handleSelectSkill}
            selectedSkillId={selectedSkillId ?? undefined}
          />
        </div>
      </div>

      {/* Selected Skill Detail Panel */}
      <AnimatePresence>
        {selectedSkill && (
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: 20 }}
            className={cn(
              'p-4',
              'bg-neo-navy border-3 border-neo-white rounded-neo',
              'shadow-hard'
            )}
            data-testid="skill-detail-panel"
          >
            <div className="flex items-start gap-4">
              {/* Icon */}
              <span className="text-4xl">{selectedSkill.icon}</span>

              {/* Details */}
              <div className="flex-1">
                <h3 className="text-xl font-bold text-neo-white">
                  {t(selectedSkill.nameKey)}
                </h3>
                <p className="text-neo-white/70 mt-1">
                  {t(selectedSkill.descriptionKey)}
                </p>

                {/* Prerequisites */}
                {selectedSkill.prerequisites.length > 0 && (
                  <p className="text-sm text-neo-white/50 mt-2">
                    {t('adventure.skills.requires')}:{' '}
                    {selectedSkill.prerequisites.map((prereqId) => {
                      const prereq = getSkillById(prereqId);
                      return prereq ? t(prereq.nameKey) : prereqId;
                    }).join(', ')}
                  </p>
                )}
              </div>

              {/* Unlock Button */}
              {!unlockedSkills.has(selectedSkill.id) && (
                <button
                  onClick={handleUnlock}
                  disabled={!canUnlockSelected}
                  className={cn(
                    'px-4 py-2',
                    'font-bold text-neo-black',
                    'border-3 border-neo-black rounded-neo',
                    canUnlockSelected
                      ? 'bg-neo-yellow hover:bg-neo-orange shadow-hard hover:shadow-hard-lg active:shadow-hard-pressed'
                      : 'bg-neo-white/30 cursor-not-allowed',
                    'transition-all duration-200',
                    'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-neo-cyan'
                  )}
                  data-testid="unlock-skill-button"
                >
                  {canUnlockSelected
                    ? t('adventure.skills.unlock', { cost: selectedSkill.cost })
                    : t('adventure.skills.locked')}
                </button>
              )}

              {/* Already Unlocked */}
              {unlockedSkills.has(selectedSkill.id) && (
                <span className="px-4 py-2 bg-neo-green/20 text-neo-green font-bold rounded-neo border-2 border-neo-green">
                  {t('adventure.skills.status.unlocked')}
                </span>
              )}
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
});

SkillTreeView.displayName = 'SkillTreeView';

export { SkillTreeView };
export default SkillTreeView;
```

```typescript
import React from 'react';

    locale: 'en',

describe('SkillNode', () => {
  const mockSkill = SKILL_CATALOG[0]; // First skill (tier 1)

    totalPointsEarned: 0,

    expect(screen.getByText(mockSkill.icon)).toBeInTheDocument();

        skill={mockSkill}
        state={emptyState}
        onSelect={jest.fn()}

    const button = screen.getByTestId(`skill-node-${mockSkill.id}`);
    expect(button).toHaveAttribute('data-state', 'locked');

        skill={mockSkill}
        state={state}
        onSelect={jest.fn()}
```
