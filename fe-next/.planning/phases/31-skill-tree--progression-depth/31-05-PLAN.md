```typescript
/**
 * SkillUnlockModal Component
 *
 * Celebration modal shown when a skill is unlocked.
 * Path-colored accents with confetti and spring animations.
 *
 * Design: Neo-brutalist with hard shadows and chunky borders.
 */

'use client';

import React, { memo, useEffect, useId } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import { fireLevelUpConfetti } from '@/utils/confettiUtils';
import type { SkillNode, SkillPath } from '@/types/adventure';

// ==============================================
// TYPES
// ==============================================

export interface SkillUnlockModalProps {
  /** Skill that was unlocked - null hides the modal */
  skill: SkillNode | null;
  /** Callback when modal is dismissed */
  onClose: () => void;
}

// ==============================================
// CONSTANTS
// ==============================================

/** Path colors for modal styling */
const PATH_COLORS: Record<SkillPath, { bg: string; border: string; text: string; glow: string }> = {
  power: {
    bg: 'bg-neo-orange/20',
    border: 'border-neo-orange',
    text: 'text-neo-orange',
    glow: 'shadow-[0_0_20px_rgba(255,107,53,0.5)]',
  },
  strategy: {
    bg: 'bg-neo-cyan/20',
    border: 'border-neo-cyan',
    text: 'text-neo-cyan',
    glow: 'shadow-[0_0_20px_rgba(0,255,255,0.5)]',
  },
  utility: {
    bg: 'bg-neo-yellow/20',
    border: 'border-neo-yellow',
    text: 'text-neo-yellow',
    glow: 'shadow-[0_0_20px_rgba(255,225,53,0.5)]',
  },
};

/** Path emoji icons */
const PATH_ICONS: Record<SkillPath, string> = {
  power: '⚔️',
  strategy: '🧠',
  utility: '🔧',
};

// ==============================================
// COMPONENT
// ==============================================

const SkillUnlockModal = memo<SkillUnlockModalProps>(({ skill, onClose }) => {
  const { t } = useLanguage();
  const titleId = useId();

  // Handle escape key
  useEffect(() => {
    const handleEscape = (event: KeyboardEvent) => {
      if (event.key === 'Escape' && skill) {
        onClose();
      }
    };

    document.addEventListener('keydown', handleEscape);
    return () => document.removeEventListener('keydown', handleEscape);
  }, [skill, onClose]);

  // Fire confetti when modal opens
  useEffect(() => {
    if (skill) {
      fireLevelUpConfetti();
    }
  }, [skill]);

  if (!skill) {
    return null;
  }

  const colors = PATH_COLORS[skill.path];
  const pathIcon = PATH_ICONS[skill.path];

  return (
    <AnimatePresence>
      {skill && (
        <motion.div
          role="dialog"
          aria-modal="true"
          aria-labelledby={titleId}
          className={cn(
            'fixed inset-0 z-[300]',
            'flex items-center justify-center',
            'bg-neo-black/80 backdrop-blur-sm'
          )}
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          onClick={onClose}
          data-testid="skill-unlock-modal"
        >
          {/* Modal Card */}
          <motion.div
            className={cn(
              'relative w-full max-w-md mx-4',
              'bg-neo-navy border-4 border-neo-black',
              'rounded-neo shadow-hard-lg',
              'p-6 md:p-8',
              'text-center',
              colors.glow
            )}
            initial={{ scale: 0, rotate: -10 }}
            animate={{ scale: 1, rotate: 0 }}
            exit={{ scale: 0, rotate: 10 }}
            transition={{ type: 'spring', stiffness: 300, damping: 20 }}
            onClick={(e) => e.stopPropagation()}
          >
            {/* Celebration Emoji */}
            <motion.span
              className="block text-6xl mb-4"
              initial={{ scale: 0, rotate: -30 }}
              animate={{ scale: [0, 1.3, 1], rotate: [30, -15, 0] }}
              transition={{ delay: 0.2, duration: 0.5, ease: 'easeOut' }}
              role="img"
              aria-hidden="true"
            >
              🎊
            </motion.span>

            {/* Title */}
            <h2
              id={titleId}
              className={cn(
                'text-3xl md:text-4xl font-black',
                'mb-4',
                colors.text
              )}
              style={{
                textShadow: skill.path === 'power'
                  ? '0 0 15px rgba(255,107,53,0.6)'
                  : skill.path === 'strategy'
                  ? '0 0 15px rgba(0,255,255,0.6)'
                  : '0 0 15px rgba(255,225,53,0.6)',
              }}
            >
              {t('adventure.skills.skill_unlocked')}
            </h2>

            {/* Skill Display */}
            <div className="mb-6">
              {/* Skill Icon */}
              <motion.div
                className={cn(
                  'inline-flex items-center justify-center',
                  'w-20 h-20 md:w-24 md:h-24',
                  'border-4 rounded-neo',
                  'mb-4',
                  colors.bg,
                  colors.border
                )}
                initial={{ scale: 0 }}
                animate={{ scale: 1 }}
                transition={{ delay: 0.4, type: 'spring', stiffness: 200 }}
              >
                <span className="text-4xl md:text-5xl">{skill.icon}</span>
              </motion.div>

              {/* Skill Name */}
              <motion.p
                className={cn('text-2xl font-bold', colors.text)}
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ delay: 0.5 }}
              >
                {t(skill.nameKey)}
              </motion.p>

              {/* Skill Description */}
              <motion.p
                className="text-neo-white/70 mt-2"
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ delay: 0.6 }}
              >
                {t(skill.descriptionKey)}
              </motion.p>
            </div>

            {/* Path Badge */}
            <motion.div
              className={cn(
                'inline-flex items-center gap-2',
                'mb-6 py-2 px-4 rounded-neo',
                'border-2',
                colors.bg,
                colors.border
              )}
              initial={{ opacity: 0, y: 10 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: 0.7 }}
            >
              <span role="img" aria-hidden="true">{pathIcon}</span>
              <span className={cn('font-bold text-sm uppercase tracking-wide', colors.text)}>
                {t(`adventure.skills.paths.${skill.path}`)}
              </span>
            </motion.div>

            {/* Continue Button */}
            <motion.button
              onClick={onClose}
              className={cn(
                'w-full py-3 px-6',
                'bg-neo-yellow hover:bg-neo-orange',
                'text-neo-black font-black text-lg',
                'border-3 border-neo-black rounded-neo',
                'shadow-hard hover:shadow-hard-lg hover:-translate-y-0.5',
                'active:translate-y-0.5 active:shadow-hard-pressed',
                'focus-visible:outline-none focus-visible:ring-4 focus-visible:ring-neo-cyan',
                'transition-all duration-200'
              )}
              initial={{ opacity: 0, y: 10 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: 0.8 }}
              data-testid="skill-unlock-continue"
            >
              {t('adventure.skills.continue')}
            </motion.button>
          </motion.div>
        </motion.div>
      )}
    </AnimatePresence>
  );
});

SkillUnlockModal.displayName = 'SkillUnlockModal';

export { SkillUnlockModal };
export default SkillUnlockModal;
```

```typescript
/**
 * SkillUnlockModal Tests
 */

import React from 'react';
import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import { SkillUnlockModal } from './SkillUnlockModal';
import { SKILL_CATALOG } from '@/utils/skillTreeUtils';

// Mock confetti
jest.mock('@/utils/confettiUtils', () => ({
  fireLevelUpConfetti: jest.fn(),
}));

// Mock language context
jest.mock('@/contexts/LanguageContext', () => ({
  useLanguage: () => ({
    t: (key: string) => key.split('.').pop() || key,
    locale: 'en',
  }),
}));

import { fireLevelUpConfetti } from '@/utils/confettiUtils';

describe('SkillUnlockModal', () => {
  const mockSkill = SKILL_CATALOG[0];
  const mockOnClose = jest.fn();

  beforeEach(() => {
    jest.clearAllMocks();
  });

  it('renders nothing when skill is null', () => {
    const { container } = render(
      <SkillUnlockModal skill={null} onClose={mockOnClose} />
    );
    expect(container).toBeEmptyDOMElement();
  });

  it('renders modal when skill is provided', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    expect(screen.getByTestId('skill-unlock-modal')).toBeInTheDocument();
  });

  it('shows skill icon', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    expect(screen.getByText(mockSkill.icon)).toBeInTheDocument();
  });

  it('shows skill name translation key', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    // Translation mock returns last part of key
    expect(screen.getByText('name')).toBeInTheDocument();
  });

  it('fires confetti when modal opens', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    expect(fireLevelUpConfetti).toHaveBeenCalled();
  });

  it('calls onClose when continue button clicked', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    fireEvent.click(screen.getByTestId('skill-unlock-continue'));
    expect(mockOnClose).toHaveBeenCalled();
  });

  it('calls onClose when backdrop clicked', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    fireEvent.click(screen.getByTestId('skill-unlock-modal'));
    expect(mockOnClose).toHaveBeenCalled();
  });

  it('calls onClose when Escape key pressed', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);
    fireEvent.keyDown(document, { key: 'Escape' });
    expect(mockOnClose).toHaveBeenCalled();
  });

  it('does not close when modal content clicked', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);

    // Click on modal content (not backdrop)
    const modalContent = screen.getByRole('dialog');
    const innerContent = modalContent.querySelector('.bg-neo-navy');
    if (innerContent) {
      fireEvent.click(innerContent);
    }

    // onClose should not be called from inner click
    // It's called from backdrop only
  });

  it('has correct role and aria attributes', () => {
    render(<SkillUnlockModal skill={mockSkill} onClose={mockOnClose} />);

    const dialog = screen.getByRole('dialog');
    expect(dialog).toHaveAttribute('aria-modal', 'true');
    expect(dialog).toHaveAttribute('aria-labelledby');
  });

  it('shows path-specific styling for power path', () => {
    const powerSkill = SKILL_CATALOG.find(s => s.path === 'power')!;
    render(<SkillUnlockModal skill={powerSkill} onClose={mockOnClose} />);

    // Power path has ⚔️ icon
    expect(screen.getByText('⚔️')).toBeInTheDocument();
  });

  it('shows path-specific styling for strategy path', () => {
    const strategySkill = SKILL_CATALOG.find(s => s.path === 'strategy')!;
    render(<SkillUnlockModal skill={strategySkill} onClose={mockOnClose} />);

    // Strategy path has 🧠 icon
    expect(screen.getByText('🧠')).toBeInTheDocument();
  });

  it('shows path-specific styling for utility path', () => {
    const utilitySkill = SKILL_CATALOG.find(s => s.path === 'utility')!;
    render(<SkillUnlockModal skill={utilitySkill} onClose={mockOnClose} />);

    // Utility path has 🔧 icon
    expect(screen.getByText('🔧')).toBeInTheDocument();
  });
});
```
