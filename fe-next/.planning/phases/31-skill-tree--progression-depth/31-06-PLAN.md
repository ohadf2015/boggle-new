export function getAchievementById(
  id: AdventureAchievementId

  return ADVENTURE_ACHIEVEMENTS[id];

/**
 * Check progress for a single achievement
 */
export function checkAchievementProgress(
  achievementId: AdventureAchievementId,
  stats: AchievementStats
): AchievementCheckResult {
  const achievement = ADVENTURE_ACHIEVEMENTS[achievementId];
  if (!achievement) {
    return { earned: false, count: 0 };
  }

  let count = 0;
  const target = achievement.target || 1;

  switch (achievementId) {
    // Gameplay
    case 'FIRST_WORD':
    case 'WORD_WARRIOR':
    case 'WORD_LEGEND':
      count = stats.wordsFound || 0;
      break;
    case 'COMBO_STARTER':
    case 'COMBO_MASTER':
      count = stats.maxCombo || 0;
      break;
    case 'LONG_WORD_FINDER':
      count = stats.longWordsFound || 0;
      break;

    // Progression
    case 'BOSS_SLAYER':
    case 'BOSS_HUNTER':
      count = stats.bossesDefeated || 0;
      break;
    case 'LEVEL_10':
    case 'LEVEL_25':
      count = stats.playerLevel || 0;
      break;
    case 'WORLD_EXPLORER':
      count = stats.worldsCompleted || 0;
      break;
    case 'PERFECTIONIST':
      count = stats.perfectLevels || 0;
      break;

    // Collection
    case 'POWER_UP_FAN':
      count = stats.powerUpsUsed || 0;
      break;
    case 'SPECIAL_TILE_EXPERT':
      count = stats.specialTilesCleared || 0;
      break;

    // Skill
    case 'SKILL_COLLECTOR':
    case 'SKILL_MASTER':
      count = stats.skillsUnlocked || 0;
      break;
    case 'DEDICATED_PLAYER':
      count = stats.streakDays || 0;
      break;

    default:
      count = 0;
  }

  const earned = count >= target;

  return {
    earned,
    count,
    target,
    tier: earned ? calculateTier(count) : null,
  };
}

/**
 * Check progress for multiple achievements
 */
export function checkMultipleAchievements(
  achievementIds: AdventureAchievementId[],
  stats: AchievementStats
): Record<AdventureAchievementId, AchievementCheckResult> {
  const results: Partial<Record<AdventureAchievementId, AchievementCheckResult>> = {};

  for (const id of achievementIds) {
    results[id] = checkAchievementProgress(id, stats);
  }

  return results as Record<AdventureAchievementId, AchievementCheckResult>;
}

/**
 * Get all achievements by category
 */
export function getAchievementsByCategory(
  category: AchievementCategory
): AdventureAchievement[] {
  return Object.values(ADVENTURE_ACHIEVEMENTS).filter(
    (a) => a.category === category
  );
}

/**
 * Get all achievement IDs
 */
export function getAllAchievementIds(): AdventureAchievementId[] {
  return Object.keys(ADVENTURE_ACHIEVEMENTS) as AdventureAchievementId[];
}

          state: {

            earnedAchievements: new Set(parsed.state.earnedAchievements),

        });
      }
      return str;

      return null;
    }

      const parsed = JSON.parse(value);
      const serialized = {
        ...parsed,
        state: {
          ...parsed.state,
          earnedAchievements: parsed.state.earnedAchievements instanceof Set
            ? [...parsed.state.earnedAchievements]
            : parsed.state.earnedAchievements || [],
        },
      };
      localStorage.setItem(name, JSON.stringify(serialized));
    } catch {

  removeItem: (name: string): void => {
