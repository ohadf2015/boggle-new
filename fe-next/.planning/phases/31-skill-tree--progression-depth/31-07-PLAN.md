/**
 * AchievementCard Component
 *
 * Displays a single achievement with tier progress.
 * Shows locked state for unearned achievements.
 */

'use client';

import React, { memo } from 'react';
import { motion } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import {
  getTierProgress,
  getTierDisplay,
  TIER_COLORS,
  type TierName,
} from '@/utils/achievementTiers';
import type { AdventureAchievement } from '@/utils/adventureAchievementUtils';

// ==============================================
// TYPES
// ==============================================

export interface AchievementCardProps {
  /** Achievement definition */
  achievement: AdventureAchievement;
  /** Number of times earned */
  count: number;
  /** Whether the achievement is earned */
  isEarned: boolean;
  /** Optional click handler */
  onClick?: () => void;
}

// ==============================================
// COMPONENT
// ==============================================

const AchievementCard = memo<AchievementCardProps>(({
  achievement,
  count,
  isEarned,
  onClick,
}) => {
  const { t } = useLanguage();

  // Get tier info
  const tierProgress = getTierProgress(count);
  const tierDisplay = getTierDisplay(tierProgress.currentTier);

  // Get tier colors
  const tierColors = tierProgress.currentTier
    ? TIER_COLORS[tierProgress.currentTier]
    : null;

  return (
    <motion.button
      onClick={onClick}
      className={cn(
        'relative w-full p-4',
        'flex flex-col items-center gap-2',
        'bg-neo-navy border-3 rounded-neo',
        'transition-all duration-200',
        isEarned
          ? 'border-neo-white shadow-hard cursor-pointer hover:shadow-hard-lg hover:-translate-y-0.5'
          : 'border-neo-white/30 opacity-60 cursor-default grayscale',
        'focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-neo-cyan'
      )}
      disabled={!isEarned}
      data-testid={`achievement-card-${achievement.id}`}
      whileHover={isEarned ? { scale: 1.02 } : {}}
      whileTap={isEarned ? { scale: 0.98 } : {}}
    >
      {/* Icon */}
      <div
        className={cn(
          'w-12 h-12 flex items-center justify-center',
          'rounded-neo border-2',
          isEarned && tierColors
            ? 'border-current'
            : 'border-neo-white/30'
        )}
        style={isEarned && tierColors ? { borderColor: tierColors.border } : {}}
      >
        <span className="text-2xl">{achievement.icon}</span>
      </div>

      {/* Name */}
      <span className={cn(
        'text-sm font-bold text-center',
        isEarned ? 'text-neo-white' : 'text-neo-white/50'
      )}>
        {t(achievement.nameKey)}
      </span>

      {/* Tier Badge (if earned) */}
      {isEarned && tierDisplay && (
        <div
          className={cn(
            'absolute -top-2 -right-2',
            'px-2 py-0.5',
            'text-xs font-bold rounded-full',
            'border-2'
          )}
          style={{
            backgroundColor: tierColors?.bg,
            borderColor: tierColors?.border,
            color: tierColors?.text,
          }}
        >
          {tierDisplay.icon}
        </div>
      )}

      {/* Progress Bar (if earned but not max tier) */}
      {isEarned && !tierProgress.isMaxTier && (
        <div className="w-full mt-1">
          <div className="h-1 bg-neo-white/20 rounded-full overflow-hidden">
            <div
              className="h-full rounded-full transition-all duration-300"
              style={{
                width: `${tierProgress.progress}%`,
                backgroundColor: tierColors?.bg || '#FFE135',
              }}
            />
          </div>
          <p className="text-[10px] text-neo-white/50 text-center mt-1">
            {count}/{tierProgress.nextThreshold}
          </p>
        </div>
      )}

      {/* Locked Overlay */}
      {!isEarned && (
        <div className="absolute inset-0 flex items-center justify-center">
          <span className="text-3xl opacity-30">🔒</span>
        </div>
      )}
    </motion.button>
  );
});

AchievementCard.displayName = 'AchievementCard';

export { AchievementCard };
export default AchievementCard;

/**
 * AchievementGrid Component
 *
 * Displays all achievements in a grid with category filters.
 * Shows completion progress for each category.
 */

'use client';

import React, { memo, useState, useMemo } from 'react';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import { useAdventureAchievements } from '@/hooks/useAdventureAchievements';
import {
  ADVENTURE_ACHIEVEMENTS,
  getAchievementsByCategory,
  type AchievementCategory,
  type AdventureAchievementId,
} from '@/utils/adventureAchievementUtils';
import { AchievementCard } from './AchievementCard';

// ==============================================
// TYPES
// ==============================================

export interface AchievementGridProps {
  /** Optional callback when achievement is clicked */
  onSelectAchievement?: (id: AdventureAchievementId) => void;
  /** Optional class name */
  className?: string;
}

// ==============================================
// CONSTANTS
// ==============================================

const CATEGORIES: AchievementCategory[] = ['gameplay', 'progression', 'collection', 'skill'];

const CATEGORY_CONFIG: Record<AchievementCategory, { icon: string; color: string }> = {
  gameplay: { icon: '🎮', color: 'neo-orange' },
  progression: { icon: '📈', color: 'neo-cyan' },
  collection: { icon: '💎', color: 'neo-pink' },
  skill: { icon: '🎓', color: 'neo-yellow' },
};

// ==============================================
// COMPONENT
// ==============================================

const AchievementGrid = memo<AchievementGridProps>(({
  onSelectAchievement,
  className,
}) => {
  const { t } = useLanguage();
  const [selectedCategory, setSelectedCategory] = useState<AchievementCategory | 'all'>('all');

  // Get achievement state
  const { earnedAchievements, achievementCounts } = useAdventureAchievements();

  // Get achievements to display
  const achievements = useMemo(() => {
    if (selectedCategory === 'all') {
      return Object.values(ADVENTURE_ACHIEVEMENTS);
    }
    return getAchievementsByCategory(selectedCategory);
  }, [selectedCategory]);

  // Calculate category stats
  const categoryStats = useMemo(() => {
    const stats: Record<AchievementCategory, { earned: number; total: number }> = {
      gameplay: { earned: 0, total: 0 },
      progression: { earned: 0, total: 0 },
      collection: { earned: 0, total: 0 },
      skill: { earned: 0, total: 0 },
    };

    Object.values(ADVENTURE_ACHIEVEMENTS).forEach((achievement) => {
      stats[achievement.category].total++;
      if (earnedAchievements.has(achievement.id as AdventureAchievementId)) {
        stats[achievement.category].earned++;
      }
    });

    return stats;
  }, [earnedAchievements]);

  // Overall progress
  const totalEarned = earnedAchievements.size;
  const totalAchievements = Object.keys(ADVENTURE_ACHIEVEMENTS).length;

  return (
    <div className={cn('flex flex-col gap-6', className)} data-testid="achievement-grid">
      {/* Header with progress */}
      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
        <h2 className="text-2xl font-black text-neo-white">
          {t('adventure.achievements.title')}
        </h2>
        <div
          className={cn(
            'px-4 py-2',
            'bg-neo-navy border-3 border-neo-yellow rounded-neo',
            'text-neo-yellow font-bold'
          )}
          data-testid="achievement-progress"
        >
          {t('adventure.achievements.progress', {
            earned: totalEarned,
            total: totalAchievements,
          })}
        </div>
      </div>

      {/* Category Filters */}
      <div className="flex flex-wrap gap-2">
        <button
          onClick={() => setSelectedCategory('all')}
          className={cn(
            'px-3 py-1.5',
            'font-bold text-sm',
            'border-2 rounded-neo',
            'transition-all duration-200',
            selectedCategory === 'all'
              ? 'bg-neo-yellow text-neo-black border-neo-black shadow-hard'
              : 'bg-neo-navy text-neo-white/70 border-neo-white/30 hover:border-neo-white/50'
          )}
          data-testid="filter-all"
        >
          {t('adventure.achievements.categories.all')} ({totalEarned}/{totalAchievements})
        </button>

        {CATEGORIES.map((category) => {
          const config = CATEGORY_CONFIG[category];
          const stats = categoryStats[category];

          return (
            <button
              key={category}
              onClick={() => setSelectedCategory(category)}
              className={cn(
                'px-3 py-1.5',
                'font-bold text-sm',
                'border-2 rounded-neo',
                'transition-all duration-200',
                selectedCategory === category
                  ? 'bg-neo-yellow text-neo-black border-neo-black shadow-hard'
                  : 'bg-neo-navy text-neo-white/70 border-neo-white/30 hover:border-neo-white/50'
              )}
              data-testid={`filter-${category}`}
            >
              {config.icon} {t(`adventure.achievements.categories.${category}`)} ({stats.earned}/{stats.total})
            </button>
          );
        })}
      </div>

      {/* Achievement Grid */}
      <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4">
        {achievements.map((achievement) => {
          const id = achievement.id as AdventureAchievementId;
          const isEarned = earnedAchievements.has(id);
          const count = achievementCounts[id] || 0;

          return (
            <AchievementCard
              key={id}
              achievement={achievement}
              count={count}
              isEarned={isEarned}
              onClick={() => onSelectAchievement?.(id)}
            />
          );
        })}
      </div>

      {/* Empty State */}
      {achievements.length === 0 && (
        <div className="text-center py-12">
          <span className="text-4xl mb-4 block">🏆</span>
          <p className="text-neo-white/70">
            {t('adventure.achievements.no_achievements')}
          </p>
        </div>
      )}
    </div>
  );
});

AchievementGrid.displayName = 'AchievementGrid';

export { AchievementGrid };
export default AchievementGrid;

// AchievementUnlockModal.tsx
/**
 * AchievementUnlockModal Component
 *
 * Celebration modal for adventure mode achievements.
 * Shows tier-colored styling and confetti on unlock.
 */

'use client';

import React, { memo, useEffect, useId } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useLanguage } from '@/contexts/LanguageContext';
import { fireLevelUpConfetti } from '@/utils/confettiUtils';
import { getTierProgress, getTierDisplay, TIER_COLORS } from '@/utils/achievementTiers';
import type { AdventureAchievement } from '@/utils/adventureAchievementUtils';

// ==============================================
// TYPES
// ==============================================

export interface AchievementUnlockModalProps {
  /** Achievement that was earned - null hides modal */
  achievement: AdventureAchievement | null;
  /** Times this achievement has been earned (for tier display) */
  count: number;
  /** Whether this is a first-time unlock or tier upgrade */
  isNew: boolean;
  /** Callback when modal is dismissed */
  onClose: () => void;
}

// ==============================================
// COMPONENT
// ==============================================

const AchievementUnlockModal = memo<AchievementUnlockModalProps>(({
  achievement,
  count,
  isNew,
  onClose,
}) => {
  const { t } = useLanguage();
  const titleId = useId();

  // Get tier info
  const tierProgress = getTierProgress(count);
  const tierDisplay = getTierDisplay(tierProgress.currentTier);
  const tierColors = tierProgress.currentTier
    ? TIER_COLORS[tierProgress.currentTier]
    : null;

  // Escape key handler
  useEffect(() => {
    const handleEscape = (e: KeyboardEvent) => {
      if (e.key === 'Escape' && achievement) {
        onClose();
      }
    };
    document.addEventListener('keydown', handleEscape);
    return () => document.removeEventListener('keydown', handleEscape);
  }, [achievement, onClose]);

  // Fire confetti for Gold/Platinum
  useEffect(() => {
    if (achievement && (tierProgress.currentTier === 'GOLD' || tierProgress.currentTier === 'PLATINUM')) {
      fireLevelUpConfetti();
    }
  }, [achievement, tierProgress.currentTier]);

  if (!achievement) return null;

  return (
    <AnimatePresence>
      {achievement && (
        <motion.div
          role="dialog"
          aria-modal="true"
          aria-labelledby={titleId}
          className={cn(
            'fixed inset-0 z-[300]',
            'flex items-center justify-center',
            'bg-neo-black/80 backdrop-blur-sm'
          )}
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          exit={{ opacity: 0 }}
          onClick={onClose}
          data-testid="achievement-unlock-modal"
        >
          <motion.div
            className={cn(
              'relative w-full max-w-md mx-4',
              'bg-neo-navy border-4 border-neo-black',
              'rounded-neo shadow-hard-lg',
              'p-6 md:p-8 text-center'
            )}
            style={tierColors ? { boxShadow: `0 0 20px ${tierColors.glow}` } : {}}
            initial={{ scale: 0, rotate: -10 }}
            animate={{ scale: 1, rotate: 0 }}
            exit={{ scale: 0, rotate: 10 }}
            transition={{ type: 'spring', stiffness: 300, damping: 20 }}
            onClick={(e) => e.stopPropagation()}
          >
            {/* Celebration Emoji */}
            <motion.span
              className="block text-6xl mb-4"
              initial={{ scale: 0 }}
              animate={{ scale: [0, 1.3, 1] }}
              transition={{ delay: 0.2, duration: 0.5 }}
            >
              🏆
            </motion.span>

            {/* Title */}
            <h2
              id={titleId}
              className="text-3xl md:text-4xl font-black mb-4"
              style={tierColors ? { color: tierColors.bg } : { color: '#FFE135' }}
            >
              {isNew
                ? t('adventure.achievements.earned')
                : t('adventure.achievements.tier_up')}
            </h2>

            {/* Achievement Icon */}
            <motion.div
              className={cn(
                'inline-flex items-center justify-center',
                'w-20 h-20 md:w-24 md:h-24',
                'border-4 rounded-neo mb-4'
              )}
              style={tierColors ? {
                borderColor: tierColors.border,
                backgroundColor: `${tierColors.bg}20`,
              } : {}}
              initial={{ scale: 0 }}
              animate={{ scale: 1 }}
              transition={{ delay: 0.4, type: 'spring' }}
            >
              <span className="text-4xl md:text-5xl">{achievement.icon}</span>
            </motion.div>

            {/* Achievement Name */}
            <p className="text-2xl font-bold text-neo-white mb-2">
              {t(achievement.nameKey)}
            </p>

            {/* Tier Badge */}
            {tierDisplay && (
              <motion.div
                className={cn(
                  'inline-flex items-center gap-2',
                  'mb-4 py-2 px-4 rounded-neo border-2'
                )}
                style={{
                  borderColor: tierColors?.border,
                  backgroundColor: `${tierColors?.bg}20`,
                }}
                initial={{ opacity: 0, y: 10 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ delay: 0.6 }}
              >
                <span>{tierDisplay.icon}</span>
                <span
                  className="font-bold text-sm uppercase"
                  style={{ color: tierColors?.bg }}
                >
                  {t(`adventure.achievements.tiers.${tierProgress.currentTier?.toLowerCase()}`)}
                </span>
              </motion.div>
            )}

            {/* Continue Button */}
            <motion.button
              onClick={onClose}
              className={cn(
                'w-full py-3 px-6',
                'bg-neo-yellow hover:bg-neo-orange',
                'text-neo-black font-black text-lg',
                'border-3 border-neo-black rounded-neo',
                'shadow-hard hover:shadow-hard-lg',
                'active:shadow-hard-pressed',
                'transition-all duration-200'
              )}
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              transition={{ delay: 0.8 }}
              data-testid="achievement-continue"
            >
              {t('adventure.achievements.continue')}
            </motion.button>
          </motion.div>
        </motion.div>
      )}
    </AnimatePresence>
  );
});

AchievementUnlockModal.displayName = 'AchievementUnlockModal';

export { AchievementUnlockModal };
export default AchievementUnlockModal;
