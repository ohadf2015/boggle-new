// hooks/useCrazyGamesAds.ts
import { useState, useCallback } from 'react';
import { Howler } from 'howler';
import { getCrazyGamesSDK } from '@/lib/crazygames/sdk';

export function useCrazyGamesAds() {
  const [isAdPlaying, setIsAdPlaying] = useState(false);
  const [hasAdblock, setHasAdblock] = useState(false);

  // Check for adblock on mount
  useEffect(() => {
    const checkAdblock = async () => {
      const sdk = await getCrazyGamesSDK();
      if (sdk) {
        const blocked = await sdk.ad.hasAdblock();
        setHasAdblock(blocked);
      }
    };
    checkAdblock();
  }, []);

  const requestMidgameAd = useCallback(async (): Promise<boolean> => {
    const sdk = await getCrazyGamesSDK();
    if (!sdk || hasAdblock) return false;

    setIsAdPlaying(true);
    sdk.game.gameplayStop();

    return new Promise((resolve) => {
      sdk.ad.requestAd('midgame', {
        adStarted: () => {
          Howler.mute(true);
        },
        adFinished: () => {
          Howler.mute(false);
          sdk.game.gameplayStart();
          setIsAdPlaying(false);
          resolve(true);
        },
        adError: (error, errorData) => {
          Howler.mute(false);
          sdk.game.gameplayStart();
          setIsAdPlaying(false);
          console.log('Midgame ad error:', errorData?.reason);
          resolve(false);
        },
      });
    });
  }, [hasAdblock]);

  const requestRewardedAd = useCallback(async (): Promise<boolean> => {
    const sdk = await getCrazyGamesSDK();
    if (!sdk || hasAdblock) return false;

    setIsAdPlaying(true);
    sdk.game.gameplayStop();

    return new Promise((resolve) => {
      sdk.ad.requestAd('rewarded', {
        adStarted: () => {
          Howler.mute(true);
        },
        adFinished: () => {
          Howler.mute(false);
          sdk.game.gameplayStart();
          setIsAdPlaying(false);
          resolve(true);
        },
        adError: (error, errorData) => {
          Howler.mute(false);
          sdk.game.gameplayStart();
          setIsAdPlaying(false);
          console.log('Rewarded ad error:', errorData?.reason);
          resolve(false);
        },
      });
    });
  }, [hasAdblock]);

  return {
    requestMidgameAd,
    requestRewardedAd,
    isAdPlaying,
    hasAdblock,
  };
}
```

// utils/crazygames/cloudSave.ts
import { getCrazyGamesSDK } from '@/lib/crazygames/sdk';

export interface SaveData {
  adventureProgress: {
    worldId: number;
    levelId: number;
    stars: number;
    completedLevels: string[];
  };
  educationProgress: {
    totalXp: number;
    level: number;
    streak: number;
    achievements: string[];
  };
  preferences: {
    musicVolume: number;
    soundVolume: number;
    language: string;
  };
}

export async function saveToCloud(data: SaveData): Promise<boolean> {
  const sdk = await getCrazyGamesSDK();
  if (!sdk) return false;

  try {
    const serialized = JSON.stringify(data);
    await sdk.data.setItem('save_data_v1', serialized);
    return true;
  } catch (error) {
    console.error('Cloud save error:', error);
    return false;
  }
}

export async function loadFromCloud(): Promise<SaveData | null> {
  const sdk = await getCrazyGamesSDK();
  if (!sdk) return null;

  try {
    const serialized = await sdk.data.getItem('save_data_v1');
    if (!serialized) return null;
    return JSON.parse(serialized);
  } catch (error) {
    console.error('Cloud load error:', error);
    return null;
  }
}

export async function clearCloudSave(): Promise<boolean> {
  const sdk = await getCrazyGamesSDK();
  if (!sdk) return false;

  try {
    await sdk.data.removeItem('save_data_v1');
    return true;
  } catch (error) {
    console.error('Cloud clear error:', error);
    return false;
  }
}
```

// hooks/useCrazyGamesSettings.ts
import { useEffect, useState } from 'react';
import { Howler } from 'howler';
import { getCrazyGamesSDK } from '@/lib/crazygames/sdk';

export function useCrazyGamesSettings() {
  const [muteAudio, setMuteAudio] = useState(false);
  const [disableChat, setDisableChat] = useState(false);

  useEffect(() => {
    const setupSettings = async () => {
      const sdk = await getCrazyGamesSDK();
      if (!sdk) return;

      // Check initial settings
      const settings = sdk.game.settings;
      if (settings.muteAudio) {
        setMuteAudio(true);
        Howler.mute(true);
      }
      if (settings.disableChat) {
        setDisableChat(true);
      }

      // Listen for changes
      sdk.game.onSettingsChange((newSettings) => {
        if (newSettings.muteAudio !== undefined) {
          setMuteAudio(newSettings.muteAudio);
          Howler.mute(newSettings.muteAudio);
        }
        if (newSettings.disableChat !== undefined) {
          setDisableChat(newSettings.disableChat);
        }
      });
    };

    setupSettings();
  }, []);

  return { muteAudio, disableChat };
}

export async function triggerHappytime() {
  const sdk = await getCrazyGamesSDK();
  if (!sdk) return;

  sdk.game.happytime();
}
