```typescript
describe('CrazyGames Lifecycle', () => {
  let mockSDK: jest.Mocked<CrazyGamesSDK>;

  beforeEach(() => {
    mockSDK = {
      game: {
        gameplayStart: jest.fn(),
        gameplayStop: jest.fn(),
        happyTime: jest.fn(),
        // ... other methods
      },
      // ... other modules
    };
    window.CrazyGames = { SDK: mockSDK };
  });

  it('should call gameplayStart when game begins', async () => {
    const { result } = renderHook(() => useCrazyGamesLifecycle());

    act(() => {
      result.current.startGameplay();
    });

    expect(mockSDK.game.gameplayStart).toHaveBeenCalledTimes(1);
  });

  it('should not double-call gameplayStart', async () => {
    const { result } = renderHook(() => useCrazyGamesLifecycle());

    act(() => {
      result.current.startGameplay();
      result.current.startGameplay();
    });

    expect(mockSDK.game.gameplayStart).toHaveBeenCalledTimes(1);
  });

  it('should throttle happytime calls', async () => {
    jest.useFakeTimers();
    const { result } = renderHook(() => useCrazyGamesLifecycle());

    act(() => {
      result.current.happytime();
      result.current.happytime();
      result.current.happytime();
    });

    // Only first call should go through
    expect(mockSDK.game.happyTime).toHaveBeenCalledTimes(1);

    // After 30 seconds, another call should work
    jest.advanceTimersByTime(31000);
    act(() => result.current.happytime());
    expect(mockSDK.game.happyTime).toHaveBeenCalledTimes(2);

    jest.useRealTimers();
  });
});
```

```typescript
describe('CrazyGames Multiplayer Invites', () => {
  let mockSDK: jest.Mocked<CrazyGamesSDK>;

  beforeEach(() => {
    // Setup mock SDK with invite methods
    mockSDK = createMockCrazyGamesSDK();
    window.CrazyGames = { SDK: mockSDK };
  });

  it('should show invite button when room is created', () => {
    const { result } = renderHook(() =>
      useCrazyGamesInvite({ autoShowInviteButton: true })
    );

    act(() => {
      result.current.showInviteButton('ABC123');
    });

    expect(mockSDK.game.showInviteButton).toHaveBeenCalledWith({ roomId: 'ABC123' });
  });

  it('should hide invite button when room is full', () => {
    const { result, rerender } = renderHook(
      ({ currentPlayers }) =>
        useCrazyGamesInvite({
          maxPlayers: 4,
          currentPlayers,
          gameState: 'waiting',
        }),
      { initialProps: { currentPlayers: 2 } }
    );

    act(() => result.current.showInviteButton('ABC123'));

    // Fill room
    rerender({ currentPlayers: 4 });

    expect(mockSDK.game.hideInviteButton).toHaveBeenCalled();
  });

  it('should detect invite join and call callback', async () => {
    const onInviteJoin = jest.fn();
    mockSDK.game.getInviteParam = jest.fn().mockReturnValue('XYZ789');

    renderHook(() =>
      useCrazyGamesInvite({ onInviteJoin })
    );

    await waitFor(() => {
      expect(onInviteJoin).toHaveBeenCalledWith('XYZ789');
    });
  });
});
```
