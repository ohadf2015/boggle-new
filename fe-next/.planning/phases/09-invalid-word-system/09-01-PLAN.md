  <action>
Add selection state and checkbox UI to InvalidWordsManager:

1. Add state for selected IDs:
   ```typescript
   const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   ```

2. Add selection helper functions:
   ```typescript
   const toggleSelection = (id: string) => {
     setSelectedIds(prev => {
       const next = new Set(prev);
       if (next.has(id)) {
         next.delete(id);
       } else {
         next.add(id);
       }
       return next;
     });
   };

   const selectAll = () => {
     setSelectedIds(new Set(words.map(w => w.id)));
   };

   const clearSelection = () => {
     setSelectedIds(new Set());
   };

   const isSelected = (id: string) => selectedIds.has(id);
   ```

3. Add checkbox to each word card (inside CardContent, before word header):
   ```tsx
   <div className="absolute top-2 left-2">
     <input
       type="checkbox"
       checked={isSelected(word.id)}
       onChange={() => toggleSelection(word.id)}
       className="w-5 h-5 rounded border-slate-600 bg-slate-700 text-neo-yellow focus:ring-neo-yellow cursor-pointer"
       aria-label={`Select ${word.word}`}
     />
   </div>
   ```

4. Update Card component to have relative positioning for checkbox:
   ```tsx
   <Card className={cn(
     "bg-slate-800 border-slate-700 transition-all relative",
     isSelected(word.id) && "ring-2 ring-neo-yellow",
     isProcessing && "opacity-50 pointer-events-none"
   )}>
   ```

5. Clear selection when filter changes (add to useEffect dependencies):
   - When searchQuery, langFilter, minCount, or offset changes, clear selectedIds

6. Clear selection after successful approve/dismiss (already removes word from list, selection auto-clears)
