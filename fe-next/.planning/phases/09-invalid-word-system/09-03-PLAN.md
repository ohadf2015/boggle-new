```typescript
'use client';

import React, { useState } from 'react';
import { CheckCircle, Loader2, AlertTriangle } from 'lucide-react';
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
  AlertDialogTrigger,
} from '@/components/ui/alert-dialog';

interface BulkApproveResult {
  approved: number;
  skipped: number;
  failed: number;
}

interface BulkApproveButtonProps {
  selectedCount: number;
  selectedIds: string[];
  authToken: string;
  onComplete: () => void;
  disabled?: boolean;
}

export function BulkApproveButton({
  selectedCount,
  selectedIds,
  authToken,
  onComplete,
  disabled = false,
}: BulkApproveButtonProps): React.ReactElement {
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<BulkApproveResult | null>(null);
  const [dialogOpen, setDialogOpen] = useState(false);

  const handleApprove = async (): Promise<void> => {
    if (selectedIds.length === 0 || loading) return;

    setLoading(true);
    setResult(null);
    setDialogOpen(false);

    try {
      const response = await fetch('/api/admin/invalid-words/bulk-approve', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${authToken}`,
        },
        body: JSON.stringify({ wordIds: selectedIds }),
      });

      if (!response.ok) {
        const error = await response.json();
        throw new Error(error.error || 'Failed to approve words');
      }

      const data = await response.json();
      setResult({
        approved: data.approved || 0,
        skipped: data.skipped || 0,
        failed: data.failed || 0,
      });

      // Notify parent to refresh list and clear selection
      onComplete();

      // Clear result after 5 seconds
      setTimeout(() => setResult(null), 5000);
    } catch (error) {
      console.error('Bulk approve error:', error);
      setResult({ approved: 0, skipped: 0, failed: selectedIds.length });
    } finally {
      setLoading(false);
    }
  };

  const isDisabled = disabled || selectedCount === 0 || loading;

  return (
    <div className="flex items-center gap-3">
      <AlertDialog open={dialogOpen} onOpenChange={setDialogOpen}>
        <AlertDialogTrigger asChild>
          <button
            disabled={isDisabled}
            className={`
              flex items-center gap-2 px-4 py-2 rounded-lg font-medium
              border-2 border-black shadow-hard
              transition-all duration-150
              ${
                isDisabled
                  ? 'bg-gray-300 text-gray-500 cursor-not-allowed'
                  : 'bg-neo-yellow text-black hover:bg-yellow-400 active:shadow-hard-pressed active:translate-x-[2px] active:translate-y-[2px]'
              }
            `}
          >
            {loading ? (
              <Loader2 className="w-4 h-4 animate-spin" />
            ) : (
              <CheckCircle className="w-4 h-4" />
            )}
            {loading
              ? 'Approving...'
              : `Approve ${selectedCount} Words`}
          </button>
        </AlertDialogTrigger>

        <AlertDialogContent className="bg-slate-800 border-slate-700">
          <AlertDialogHeader>
            <AlertDialogTitle className="text-white flex items-center gap-2">
              <AlertTriangle className="w-5 h-5 text-neo-yellow" />
              Confirm Bulk Approval
            </AlertDialogTitle>
            <AlertDialogDescription className="text-slate-300">
              You are about to approve <span className="font-bold text-neo-yellow">{selectedCount}</span> invalid word{selectedCount !== 1 ? 's' : ''}.
              <br /><br />
              Approved words will be added to the game dictionary and will validate immediately in new games.
              <br /><br />
              This action cannot be easily undone.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel className="bg-slate-700 text-white border-slate-600 hover:bg-slate-600">
              Cancel
            </AlertDialogCancel>
            <AlertDialogAction
              onClick={handleApprove}
              className="bg-neo-yellow text-black hover:bg-yellow-400 border-2 border-black"
            >
              Approve {selectedCount} Words
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>

      {result && (
        <span className="text-sm">
          <span className="text-green-500">{result.approved} approved</span>
          {result.skipped > 0 && (
            <span className="text-yellow-500 ml-2">{result.skipped} skipped</span>
          )}
          {result.failed > 0 && (
            <span className="text-red-500 ml-2">{result.failed} failed</span>
          )}
        </span>
      )}
    </div>
  );
}
```
