```typescript
/**
 * BoardFrame Component
 *
 * Wraps the adventure game grid with world-specific decorative elements.
 * - Meadows: Vine corner decorations with organic feel
 * - Springs: Water splash effects at corners
 * - Caverns: Crystal cluster glow points
 *
 * Respects RTL mode for proper directional styling.
 */

'use client';

import React, { memo } from 'react';
import { cn } from '@/lib/utils';
import { useAdventureTheme } from '@/contexts/AdventureThemeContext';

// ==============================================
// TYPES
// ==============================================

interface BoardFrameProps {
  /** Children to render inside the frame (the grid) */
  children: React.ReactNode;
  /** Additional CSS classes */
  className?: string;
}

// ==============================================
// WORLD FRAME MAPPING
// ==============================================

const WORLD_FRAME_CLASSES: Record<number, string> = {
  1: 'board-frame-meadows',
  2: 'board-frame-springs',
  3: 'board-frame-caverns',
};

// ==============================================
// CORNER DECORATIONS
// ==============================================

interface CornerDecorProps {
  position: 'top-left' | 'top-right' | 'bottom-left' | 'bottom-right';
  worldId: number;
}

const CornerDecor = memo<CornerDecorProps>(({ position, worldId }) => {
  const positionClasses: Record<string, string> = {
    'top-left': 'top-0 left-0',
    'top-right': 'top-0 right-0',
    'bottom-left': 'bottom-0 left-0',
    'bottom-right': 'bottom-0 right-0',
  };

  const rotations: Record<string, string> = {
    'top-left': 'rotate-0',
    'top-right': 'rotate-90',
    'bottom-right': 'rotate-180',
    'bottom-left': '-rotate-90',
  };

  // World-specific decorations
  const decorations: Record<number, React.ReactNode> = {
    // Meadows: Vine curl
    1: (
      <svg
        className={cn(
          'w-6 h-6 text-lime-500/40',
          rotations[position]
        )}
        viewBox="0 0 24 24"
        fill="none"
        stroke="currentColor"
        strokeWidth="2"
      >
        <path d="M3 3 C 12 3, 21 12, 21 21" />
        <path d="M6 3 C 6 8, 8 12, 14 14" opacity="0.5" />
      </svg>
    ),
    // Springs: Water splash
    2: (
      <svg
        className={cn(
          'w-6 h-6 text-cyan-400/40',
          rotations[position]
        )}
        viewBox="0 0 24 24"
        fill="currentColor"
      >
        <circle cx="6" cy="6" r="3" opacity="0.6" />
        <circle cx="10" cy="4" r="2" opacity="0.4" />
        <circle cx="4" cy="10" r="2" opacity="0.4" />
      </svg>
    ),
    // Caverns: Crystal cluster
    3: (
      <svg
        className={cn(
          'w-6 h-6 text-purple-400/40',
          rotations[position]
        )}
        viewBox="0 0 24 24"
        fill="currentColor"
      >
        <polygon points="6,4 10,12 2,12" opacity="0.6" />
        <polygon points="10,2 13,10 7,10" opacity="0.4" />
        <polygon points="4,8 8,16 0,16" opacity="0.3" />
      </svg>
    ),
  };

  return (
    <div
      className={cn(
        'absolute pointer-events-none',
        positionClasses[position]
      )}
      aria-hidden="true"
    >
      {decorations[worldId]}
    </div>
  );
});

CornerDecor.displayName = 'CornerDecor';

// ==============================================
// MAIN COMPONENT
// ==============================================

const BoardFrame = memo<BoardFrameProps>(({ children, className }) => {
  const { theme } = useAdventureTheme();
  const worldId = theme.id;

  // Get frame class for current world (default to no special frame)
  const frameClass = WORLD_FRAME_CLASSES[worldId] || '';

  return (
    <div
      className={cn(
        'board-frame relative',
        frameClass,
        className
      )}
      data-world={worldId}
    >
      {/* Corner decorations */}
      <CornerDecor position="top-left" worldId={worldId} />
      <CornerDecor position="top-right" worldId={worldId} />
      <CornerDecor position="bottom-left" worldId={worldId} />
      <CornerDecor position="bottom-right" worldId={worldId} />

      {/* Grid content */}
      <div className="relative z-10">
        {children}
      </div>
    </div>
  );
});

BoardFrame.displayName = 'BoardFrame';

export default BoardFrame;
export { BoardFrame };
```

```typescript
/**
 * ThemedTile Component
 *
 * Wraps tile content with world-specific styling:
 * - Texture overlay (wood grain, water ripple, stone)
 * - Themed border colors
 * - Letter glow effects
 *
 * CRITICAL: Letter readability is ALWAYS preserved.
 * Texture opacity is kept below 0.1 per CONTEXT.md.
 */

'use client';

import React, { memo } from 'react';
import { cn } from '@/lib/utils';
import { useAdventureTheme } from '@/contexts/AdventureThemeContext';
import type { TileType } from '@/types/adventure';

// ==============================================
// TYPES
// ==============================================

interface ThemedTileProps {
  /** Tile type for special tile handling */
  tileType: TileType;
  /** Whether this is a selected tile */
  isSelected?: boolean;
  /** Children (the letter and badges) */
  children: React.ReactNode;
  /** Additional CSS classes */
  className?: string;
}

// ==============================================
// WORLD STYLING MAPS
// ==============================================

const TEXTURE_CLASSES: Record<number, string> = {
  1: 'tile-texture-meadows',  // Wood grain
  2: 'tile-texture-springs',  // Water ripple
  3: 'tile-texture-caverns',  // Stone
};

const BORDER_CLASSES: Record<number, string> = {
  1: 'tile-border-meadows',  // Vine style
  2: 'tile-border-springs',  // Crystal/water
  3: 'tile-border-caverns',  // Crystal glow
};

const LETTER_GLOW_CLASSES: Record<number, string> = {
  1: 'letter-glow-meadows',  // Warm sun
  2: 'letter-glow-springs',  // Cool water
  3: 'letter-glow-caverns',  // Purple crystal
};

// ==============================================
// MAIN COMPONENT
// ==============================================

const ThemedTile = memo<ThemedTileProps>(({
  tileType,
  isSelected = false,
  children,
  className,
}) => {
  const { theme } = useAdventureTheme();
  const worldId = theme.id;

  // Only apply texture/border theming to standard tiles
  // Special tiles (gold, ice, bomb, rainbow, chain, time) keep their distinct look
  const isStandardTile = tileType === 'standard';

  // Get world-specific classes
  const textureClass = isStandardTile ? TEXTURE_CLASSES[worldId] : '';
  const borderClass = isStandardTile ? BORDER_CLASSES[worldId] : '';
  const letterGlowClass = LETTER_GLOW_CLASSES[worldId] || '';

  return (
    <div
      className={cn(
        'relative',
        // Apply texture and border to standard tiles only
        textureClass,
        borderClass,
        // Letter glow applies to all tile types
        letterGlowClass,
        className
      )}
      data-world={worldId}
      data-tile-type={tileType}
    >
      {children}
    </div>
  );
});

ThemedTile.displayName = 'ThemedTile';

export default ThemedTile;
export { ThemedTile };
```
