/**
 * WorldParticles Component
 *
 * Renders world-specific floating particles with distinct shapes and animations.
 * - Meadows: Butterflies with flutter animation
 * - Springs: Water droplets with fall-splash animation
 * - Caverns: Crystal sparkles with sparkle-drift animation

'use client';

import React, { memo, useMemo } from 'react';
import { motion } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useDevicePerformance } from '@/hooks/useDevicePerformance';
import type { ParticleConfig } from '@/lib/adventure/themes/types';

// ==============================================
// TYPES
// ==============================================

interface WorldParticlesProps {
  /** Particle configuration from world theme */
  config: ParticleConfig;
  /** Additional CSS classes */
  className?: string;
}

interface ParticleData {
  id: number;
  x: number;
  y: number;
  size: number;
  color: string;
  delay: number;
  duration: number;
}

// ==============================================
// SEEDED RANDOM (deterministic positions)
// ==============================================

function seededRandom(seed: number): () => number {
  let t = seed + 0x6D2B79F5;
  return () => {
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const ButterflyParticle = memo<{ color: string; size: number }>(({ color, size }) => (

const renderParticle = (type: ParticleConfig['type'], color: string, size: number) => {
  switch (type) {
    case 'butterflies':
      return <ButterflyParticle color={color} size={size} />;
    case 'droplets':
      return <DropletParticle color={color} size={size} />;
    case 'crystals':
      return <CrystalParticle color={color} size={size} />;
    case 'leaves':
      return <LeafParticle color={color} size={size} />;
    case 'sparkles':
      return <CrystalParticle color={color} size={size} />; // Reuse crystal shape
    case 'bubbles':
      return (
        <div
          className="rounded-full"
          style={{
            width: size,
            height: size,
            backgroundColor: color,
            boxShadow: `inset 0 0 ${size * 0.4}px rgba(255, 255, 255, 0.5)`,
          }}
        />
      );
    default:
      return (
        <div
          className="rounded-full"
          style={{ width: size, height: size, backgroundColor: color }}
        />
      );
  }
};

// ==============================================
// MAIN COMPONENT
// ==============================================

const WorldParticles = memo<WorldParticlesProps>(({ config, className }) => {
  const { maxParticles, prefersReducedMotion, enableComplexAnimations } = useDevicePerformance();

  // Generate particle data with deterministic positions
  const particles = useMemo<ParticleData[]>(() => {
    if (config.type === 'none' || prefersReducedMotion) return [];

    // Adaptive count: use min of config count and device max
    // Per CONTEXT.md: sparse (5-10 visible)
    const count = Math.min(config.count, maxParticles, 10);

    return Array.from({ length: count }, (_, i) => {
      const random = seededRandom(i * 12345 + 67890);
      const size = config.sizeRange[0] + random() * (config.sizeRange[1] - config.sizeRange[0]);
      const colorIndex = Math.floor(random() * config.colors.length);
      const baseDuration = (3 + random() * 4) / config.speed;

      return {
        id: i,
        x: random() * 100,
        y: random() * 100,
        size,
        color: config.colors[colorIndex],
        delay: random() * baseDuration,
        duration: baseDuration,
      };
    });
  }, [config, maxParticles, prefersReducedMotion]);

  // Don't render if no particles or reduced motion
  if (particles.length === 0 || !enableComplexAnimations) {
    return null;
  }

  // Get animation config for particle type
  const getAnimation = (type: ParticleConfig['type']) => {
    // Map to animation key
    const animKey = type as keyof typeof particleAnimations;
    return particleAnimations[animKey] || particleAnimations.sparkles;
  };

  const animation = getAnimation(config.type);

  return (
    <div
      className={cn(
        'absolute inset-0 overflow-hidden pointer-events-none',
        className
      )}
      aria-hidden="true"
    >
      {particles.map((particle) => (
        <motion.div
          key={particle.id}
          className="absolute"
          style={{
            left: `${particle.x}%`,
            top: `${particle.y}%`,
          }}
          initial={{ opacity: 0 }}
          animate={animation.animate(particle.delay, particle.duration)}
          transition={animation.transition(particle.delay, particle.duration)}
        >
          {renderParticle(config.type, particle.color, particle.size)}
        </motion.div>
      ))}
    </div>
  );
});

WorldParticles.displayName = 'WorldParticles';

export default WorldParticles;
export { WorldParticles };

/**
 * WorldBackground Component
 *
 * Renders a multi-layered parallax background based on the current world theme.
 * Includes texture overlays and world-specific particle effects.
 *
 * Enhanced with:
 * - Combined parallax motion (gyro/gesture/ambient) via useParallax
 * - World-specific particle shapes via WorldParticles
 */

'use client';

import React, { memo } from 'react';
import { motion } from 'framer-motion';
import { cn } from '@/lib/utils';
import { useAdventureTheme } from '@/contexts/AdventureThemeContext';
import { useParallax } from '@/hooks/useParallax';
import { useDevicePerformance } from '@/hooks/useDevicePerformance';
import WorldParticles from './WorldParticles';
import type { ParallaxLayer, TextureConfig } from '@/lib/adventure/themes/types';

// ... keep existing interfaces (WorldBackgroundProps, ParallaxLayerComponentProps)

// Update ParallaxLayerComponent to accept and use parallax offset:
interface ParallaxLayerComponentProps {
  layer: ParallaxLayer;
  index: number;
  parallaxOffset: { x: number; y: number };
}

const ParallaxLayerComponent = memo<ParallaxLayerComponentProps>(
  ({ layer, index, parallaxOffset }) => {
    const isGradient = layer.source.startsWith('bg-');
    const { prefersReducedMotion } = useDevicePerformance();

    // Calculate transform based on layer depth
    // Deeper layers (higher depth) move more
    const transform = prefersReducedMotion
      ? undefined
      : {
          x: parallaxOffset.x * layer.depth * 10,
          y: parallaxOffset.y * layer.depth * 10,
        };

    return (
      <motion.div
        key={layer.id}
        className={cn(layer.className, isGradient && layer.source)}
        style={{
          opacity: layer.opacity,
          zIndex: index,
          ...(isGradient
            ? {}
            : {
                backgroundImage: `url(${layer.source})`,
                backgroundSize: 'cover',
                backgroundPosition: 'center',
              }),
        }}
        initial={{ opacity: 0 }}
        animate={{
          opacity: layer.opacity,
          ...(transform && { x: transform.x, y: transform.y }),
        }}
        transition={{
          opacity: { duration: 0.5, delay: index * 0.1 },
          x: { type: 'spring', stiffness: 100, damping: 30 },
          y: { type: 'spring', stiffness: 100, damping: 30 },
        }}
      />
    );
  }
);

ParallaxLayerComponent.displayName = 'ParallaxLayerComponent';

// ... keep existing TextureOverlay component unchanged

// Update main WorldBackground component:
const WorldBackground = memo<WorldBackgroundProps>(({ className, children }) => {
  const { theme, isTransitioning } = useAdventureTheme();
  const { background, containerClass } = theme;
  const { prefersReducedMotion } = useDevicePerformance();

  // Combined parallax input (gyro + gesture + ambient)
  const parallax = useParallax({
    intensity: 1,
    enableGyroscope: true,
    enableGesture: true,
    enableAmbient: true,
    ambientSpeed: 0.8,
  });

  return (
    <div
      className={cn(
        'relative w-full h-full overflow-hidden',
        containerClass,
        isTransitioning && 'opacity-80 transition-opacity duration-300',
        className
      )}
    >
      {/* Base color layer */}
      <div className={cn('absolute inset-0', background.baseColor)} />

      {/* Parallax layers with motion */}
      {background.layers.map((layer, index) => (
        <ParallaxLayerComponent
          key={layer.id}
          layer={layer}
          index={index + 1}
          parallaxOffset={prefersReducedMotion ? { x: 0, y: 0 } : parallax}
        />
      ))}

      {/* Texture overlay */}
      <TextureOverlay texture={background.texture} />

      {/* World-specific particles */}
      <WorldParticles
        config={background.particles}
        className="z-20"
      />

      {/* Content layer */}
      <div className="relative z-50">{children}</div>
    </div>
  );
});

WorldBackground.displayName = 'WorldBackground';

export default WorldBackground;
