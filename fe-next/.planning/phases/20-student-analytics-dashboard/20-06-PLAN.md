    ```typescript
    export interface UseRealtimeClassroomProgressOptions {

    }

    export interface UseRealtimeClassroomProgressReturn {
      isConnected: boolean;
      lastUpdate: Date | null;
      activeStudentsCount: number;  // Students active in last 5 minutes
      recentActivity: Array<{
        studentId: string;
        studentName: string;
        activity: 'word_attempted' | 'lesson_completed' | 'xp_gained';
        timestamp: Date;
      }>;
      connectionStatus: 'connecting' | 'connected' | 'disconnected' | 'error';
    }

    export function useRealtimeClassroomProgress(
      options: UseRealtimeClassroomProgressOptions
    ): UseRealtimeClassroomProgressReturn
    ```

    Implementation:
    1. Subscribe to classroom progress on mount (if enabled)
    2. Track active students (activity within 5 minutes)
    3. Maintain recent activity feed (last 10 events)
    4. Expose connection status for UI feedback
    5. Call onStudentActivity callback for each event
    6. Cleanup subscription on unmount

    Tests:
