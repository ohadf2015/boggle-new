  <behavior>
    ## Query Functions (lib/supabase/analytics.ts)

    ### getClassroomMetrics(classroomId: string)
    Returns: { studentsNeedingHelp, classAverageXp, activeStudentsToday, weeklyEngagement, totalStudents }
    - studentsNeedingHelp: Count of students with overall accuracy <60% in last 7 days
    - classAverageXp: Mean total_xp across all classroom students
    - activeStudentsToday: Students with last_practice_date = today
    - weeklyEngagement: % students who practiced 3+ days in last 7 days
    - totalStudents: Total members in classroom

    ### getCommonMistakes(classroomId: string, limit: number = 5)
    Returns: Array<{ word: string; errorRate: number; studentCount: number }>
    - Aggregates words_attempted JSON across all classroom students
    - Calculates error rate: 1 - (correct / attempts)
    - Returns top N words with highest error rate (>50% errors)

    ### getStudentProgressMetrics(studentId: string, classroomId: string)
    Returns: { vocabularyMastery, accuracyTrend, skillProgression }
    - vocabularyMastery: % words in words_mastered / total lesson words
    - accuracyTrend: Last 7 days accuracy points
    - skillProgression: XP earned per day last 14 days

    ## Hook (hooks/useClassroomAnalytics.ts)

    interface UseClassroomAnalyticsOptions {
      classroomId: string;
    }

    interface ClassroomMetrics {
      studentsNeedingHelp: number;
      classAverageXp: number;
      activeStudentsToday: number;
      commonMistakes: Array<{ word: string; errorRate: number; studentCount: number }>;
      weeklyEngagement: number;
      totalStudents: number;
    }

    Returns: {
      metrics: ClassroomMetrics | null;
      isLoading: boolean;
      error: Error | null;
      refresh: () => Promise<void>;
    }
