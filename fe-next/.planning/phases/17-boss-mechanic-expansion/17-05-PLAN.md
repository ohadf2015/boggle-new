describe('Boss Battle Integration', () => {
  beforeEach(() => {
    jest.clearAllMocks();
  });

  describe('World 1-5 Boss Mechanics', () => {
    test.each([
      { world: 1, mechanic: 'popQuiz', word: 'LETTERS', expectedBonus: true },
      { world: 2, mechanic: 'hiveMind', word: 'BOOKS', expectedBonus: true },
      { world: 3, mechanic: 'etymologyDig', word: 'TELEGRAPH', expectedBonus: true },
      { world: 4, mechanic: 'idiomBattle', word: 'PHRASE', expectedBonus: true },
      { world: 5, mechanic: 'assemblyLine', word: 'BUILD', expectedBonus: true },
    ])('World $world ($mechanic) should evaluate $word correctly', ({ world, word }) => {
      // GIVEN boss config for world
      const boss = getBossConfig(world);
      expect(boss).toBeDefined();

      // WHEN word is checked via useBossMechanics
      const { result } = renderHook(() => useBossMechanics({ worldId: world }));
      const mechanicResult = result.current.checkWord(word);

      // THEN should return valid result
      expect(mechanicResult).toBeDefined();
      expect(mechanicResult.meetsRequirement).toBe(true);
      expect(mechanicResult.scoreMultiplier).toBeGreaterThan(1);
    });

    it('World 1 Ms. Grammar should apply penalty for non-matching words', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 1 }));
      const mechanicResult = result.current.checkWord('X');
      expect(mechanicResult.scoreMultiplier).toBeLessThanOrEqual(1);
    });

    it('World 3 Professor Thesaurus should reward root fragment words', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 3 }));
      const mechanicResult = result.current.checkWord('TELEGRAPH');
      expect(mechanicResult.meetsRequirement).toBe(true);
      expect(mechanicResult.feedbackKey).toBe('adventure.bosses.common.rootFound');
    });
  });
});

  describe('World 6-10 Boss Mechanics', () => {
    test.each([
      { world: 6, mechanic: 'scrambledReality', word: 'WORD', expectedBonus: true },

      { world: 8, mechanic: 'stellarForge', word: 'QUIZ', expectedBonus: true },
      { world: 9, mechanic: 'babelSummit', word: 'GLOBAL', expectedBonus: true },
    ])('World $world ($mechanic) should evaluate $word correctly', ({ world, word }) => {
      // GIVEN boss config for world
      const boss = getBossConfig(world);
      expect(boss).toBeDefined();

      // WHEN word is checked

      const mechanicResult = result.current.checkWord(word);

      // THEN should return valid result
      expect(mechanicResult).toBeDefined();
      expect(mechanicResult.meetsRequirement).toBe(true);
    });

    it('World 7 Reflection King should detect palindromes', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 7 }));
      expect(result.current.checkWord('RACECAR').meetsRequirement).toBe(true);
      expect(result.current.checkWord('LEVEL').meetsRequirement).toBe(true);
      expect(result.current.checkWord('HELLO').meetsRequirement).toBe(false);
    });

    it('World 8 Cosmic Wordsmith should detect supernova letters', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 8 }));
      expect(result.current.checkWord('QUIZ').meetsRequirement).toBe(true);
      expect(result.current.checkWord('XENON').meetsRequirement).toBe(true);
      expect(result.current.checkWord('HELLO').meetsRequirement).toBe(false);
    });
  });

  describe('World 10 Lexicon Dragon (finalWord)', () => {
    it('should start with popQuiz phase', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 10 }));
      expect(result.current.bossState.phase).toBe('popQuiz');
    });

    it('should cycle through phases on advancePhase', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 10 }));
      const phases: string[] = [result.current.bossState.phase!];

      for (let i = 0; i < 3; i++) {
        act(() => {
          result.current.advancePhase();
        });
        phases.push(result.current.bossState.phase!);
      }

      // Should have advanced through distinct phases
      const uniquePhases = new Set(phases);
      expect(uniquePhases.size).toBe(4);
    });

    it('should delegate to current phase mechanic', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 10 }));

      // In popQuiz phase, word with double letters should succeed
      const popQuizResult = result.current.checkWord('LETTERS');
      expect(popQuizResult.meetsRequirement).toBe(true);
    });

    it('should handle all 9 phase transitions', () => {
      const { result } = renderHook(() => useBossMechanics({ worldId: 10 }));
      const allPhases = [
        'popQuiz', 'hiveMind', 'etymologyDig', 'idiomBattle',
        'assemblyLine', 'scrambledReality', 'mirrorMatch', 'stellarForge', 'babelSummit'
      ];

      // Advance through all phases
      for (let i = 0; i < 9; i++) {
        expect(allPhases).toContain(result.current.bossState.phase);
        act(() => {
          result.current.advancePhase();
        });
      }

      // Should wrap back to first phase
      expect(result.current.bossState.phase).toBe('popQuiz');
    });
  });
