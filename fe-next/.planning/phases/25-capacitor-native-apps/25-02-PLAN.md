```typescript
// utils/platform.ts
/**
 * Platform Detection Utilities
 *
 * Detects whether app is running in Capacitor native shell or web browser.
 * Uses lazy evaluation to avoid importing Capacitor on web when not needed.
 */

import { Capacitor } from '@capacitor/core';

export type PlatformType = 'ios' | 'android' | 'web';

/**
 * Check if running in native Capacitor shell (iOS or Android)
 */
export function isNative(): boolean {
  return Capacitor.isNativePlatform();
}

/**
 * Check if running on iOS
 */
export function isIOS(): boolean {
  return Capacitor.getPlatform() === 'ios';
}

/**
 * Check if running on Android
 */
export function isAndroid(): boolean {
  return Capacitor.getPlatform() === 'android';
}

/**
 * Check if running in web browser
 */
export function isWeb(): boolean {
  return Capacitor.getPlatform() === 'web';
}

/**
 * Get current platform identifier
 */
export function getPlatform(): PlatformType {
  return Capacitor.getPlatform() as PlatformType;
}
```

```typescript
/**
 * Native Haptics Utilities
 *
 * Provides native haptic feedback when running in Capacitor,
 * with graceful fallback to Web Vibration API.
 */

import { Haptics, NotificationType, ImpactStyle } from '@capacitor/haptics';
import { isNative } from './platform';
import { triggerHaptic } from './hapticFeedback';

/**
 * Trigger success haptic pattern
 * Native: iOS Taptic Engine / Android vibration
 * Web: Vibration API fallback
 */
export async function vibrateSuccess(): Promise<void> {
  if (isNative()) {
    await Haptics.notification({ type: NotificationType.Success });
  } else {
    triggerHaptic('success');
  }
}

/**
 * Trigger error haptic pattern
 */
export async function vibrateError(): Promise<void> {
  if (isNative()) {
    await Haptics.notification({ type: NotificationType.Error });

}

 * Trigger light tap haptic

  if (isNative()) {
    await Haptics.impact({ style: ImpactStyle.Light });

    triggerHaptic('light');
  }

 * Trigger selection feedback haptic
```

Implement the hook:

```typescript
// hooks/useAppLifecycle.ts
'use client';

import { useEffect, useCallback } from 'react';
import { App } from '@capacitor/app';
import { isNative } from '@/utils/platform';

interface UseAppLifecycleOptions {
  /** Called when app comes to foreground */
  onForeground?: () => void;
  /** Called when app goes to background */
  onBackground?: () => void;
}

/**
 * useAppLifecycle - Handle native app lifecycle events
 *
 * Only activates when running in Capacitor native shell.
 * No-op on web browsers.
 *
 * @example
 * useAppLifecycle({
 *   onForeground: () => {
 *     // Resume game, reconnect socket
 *   },
 *   onBackground: () => {
 *     // Pause game, save state
 *   },
 * });
 */
export function useAppLifecycle({ onForeground, onBackground }: UseAppLifecycleOptions): void {
  const handleStateChange = useCallback(({ isActive }: { isActive: boolean }) => {
    if (isActive) {
      onForeground?.();
    } else {
      onBackground?.();
    }
  }, [onForeground, onBackground]);

  useEffect(() => {
    // Only register listener on native platforms
    if (!isNative()) {
      return;
    }

    const listener = App.addListener('appStateChange', handleStateChange);

    return () => {
      listener.remove();
    };
  }, [handleStateChange]);
}

export default useAppLifecycle;
```

```typescript
// hooks/useNativeHaptics.ts
'use client';

import { useCallback } from 'react';
import { useHapticsEnabled } from '@/contexts/AccessibilityContext';
import {
  vibrateSuccess,
  vibrateError,
  vibrateTap,
  vibrateSelection,
  vibrateMedium,
  vibrateHeavy,
} from '@/utils/nativeHaptics';

/**
 * useNativeHaptics - Hook for native haptic feedback with accessibility support
 *
 * Uses native Capacitor haptics when available, falls back to Web Vibration API.
 * Respects user's haptic feedback setting from AccessibilityContext.
 *
 * @example
 * const { success, error, tap } = useNativeHaptics();
 *
 * // On valid word
 * await success();
 *
 * // On invalid word
 * await error();
 */
export function useNativeHaptics() {
  const hapticsEnabled = useHapticsEnabled();

  const success = useCallback(async () => {
    if (hapticsEnabled) await vibrateSuccess();
  }, [hapticsEnabled]);

  const error = useCallback(async () => {
    if (hapticsEnabled) await vibrateError();
  }, [hapticsEnabled]);

  const tap = useCallback(async () => {
    if (hapticsEnabled) await vibrateTap();
  }, [hapticsEnabled]);

  const selection = useCallback(async () => {
    if (hapticsEnabled) await vibrateSelection();
  }, [hapticsEnabled]);

  const medium = useCallback(async () => {
    if (hapticsEnabled) await vibrateMedium();
  }, [hapticsEnabled]);

  const heavy = useCallback(async () => {
    if (hapticsEnabled) await vibrateHeavy();
  }, [hapticsEnabled]);

  return {
    success,
    error,
    tap,
    selection,
    medium,
    heavy,
    /** Whether haptics are enabled in settings */
    enabled: hapticsEnabled,
  };
}

export default useNativeHaptics;
```
