## Context

Phase 25 implementation created all native hooks and components but failed to wire them into the application. This plan closes 4 critical gaps identified in 25-VERIFICATION.md:

1. **useSafeArea hook** - Exists in `hooks/useSafeArea.ts` but never imported/called
2. **OfflineFallback component** - Exists in `components/native/OfflineFallback.tsx` but never rendered
3. **useNativeHaptics hook** - Exists in `hooks/useNativeHaptics.ts` but never used in UI
4. **useAppLifecycle hook** - Exists in `hooks/useAppLifecycle.ts` but never used

The 5th gap (Socket.IO verification) requires manual testing in native simulators and cannot be automated in this plan.

## Architecture Decision

Create a single `NativeAppProvider` wrapper component that:
- Calls `useSafeArea()` to set CSS custom properties
- Calls `useAppLifecycle()` to handle foreground/background transitions
- Integrates with SocketContext to reconnect on foreground

Create a `NetworkStatusHandler` component that:
- Monitors `navigator.onLine` status
- Renders `OfflineFallback` when offline in native environment
- Provides retry functionality

Haptics will be integrated via the existing `SoundEffectsContext` which already has haptic patterns.

// hooks/useOnlineStatus.ts
'use client';

import { useState, useEffect } from 'react';

export function useOnlineStatus(): boolean {
  const [isOnline, setIsOnline] = useState<boolean>(
    typeof navigator !== 'undefined' ? navigator.onLine : true
  );

  useEffect(() => {
    const handleOnline = () => setIsOnline(true);
    const handleOffline = () => setIsOnline(false);

    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);

    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, []);

  return isOnline;
}
```

// components/native/NetworkStatusHandler.tsx
'use client';

import React, { useState, useCallback } from 'react';
import { useOnlineStatus } from '@/hooks/useOnlineStatus';
import { isNative } from '@/utils/platform';
import { OfflineFallback } from './OfflineFallback';

interface NetworkStatusHandlerProps {
  children: React.ReactNode;
}

export function NetworkStatusHandler({ children }: NetworkStatusHandlerProps): React.ReactElement {
  const isOnline = useOnlineStatus();
  const [isRetrying, setIsRetrying] = useState(false);

  const handleRetry = useCallback(() => {
    setIsRetrying(true);

    // Attempt to reload the page
    window.location.reload();
  }, []);

  // Only show offline fallback in native environment
  // Web browsers have their own offline indicators
  if (!isOnline && isNative()) {
    return <OfflineFallback onRetry={handleRetry} isRetrying={isRetrying} />;
  }

  return <>{children}</>;
}
```

```typescript
// components/native/NativeAppProvider.tsx
'use client';

import React, { useEffect } from 'react';
import { useSafeArea } from '@/hooks/useSafeArea';
import { useAppLifecycle } from '@/hooks/useAppLifecycle';
import { isNative } from '@/utils/platform';
import { getSharedSocketIfExists } from '@/utils/SocketContext';
import logger from '@/utils/logger';

interface NativeAppProviderProps {
  children: React.ReactNode;
}

export function NativeAppProvider({ children }: NativeAppProviderProps): React.ReactElement {
  // Initialize safe area CSS variables (no-op on web)
  // Hook sets --cap-safe-area-top/bottom/left/right on document root
  useSafeArea();

  // Handle app lifecycle transitions
  useAppLifecycle({
    onForeground: () => {
      logger.log('[NativeAppProvider] App came to foreground');

      // Reconnect socket if disconnected
      const socket = getSharedSocketIfExists();
      if (socket && !socket.connected) {
        logger.log('[NativeAppProvider] Reconnecting socket after foreground');
        socket.connect();
      }
    },
    onBackground: () => {
      logger.log('[NativeAppProvider] App went to background');
      // Note: We don't disconnect socket on background
      // iOS/Android will handle connection appropriately
    },
  });

  return <>{children}</>;
}
```

Key behaviors:
- Calls `useSafeArea()` unconditionally (hook handles native detection internally)
- Calls `useAppLifecycle()` with socket reconnection on foreground
- Logs lifecycle transitions for debugging

interface GamePageWrapperProps {
