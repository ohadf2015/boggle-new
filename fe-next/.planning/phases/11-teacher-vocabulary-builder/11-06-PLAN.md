Create a React hook for vocabulary word selection via socket:

```typescript
import { useState, useEffect, useCallback } from 'react';
import { useSocket } from '@/contexts/SocketContext';
import { checkWordIntegration } from './useWordIntegration';
import type { Language } from '@/shared/types';

interface SelectedWord {
  word: string;
  canIntegrate: boolean;
  reason?: string;
}

interface UseVocabularySelectionOptions {
  gameCode: string;
  language: Language;
  isHost: boolean;
  gameState: string;
}

export function useVocabularySelection(options: UseVocabularySelectionOptions) {
  const { socket } = useSocket();
  const [selectedWords, setSelectedWords] = useState<SelectedWord[]>([]);
  const [isSaving, setIsSaving] = useState(false);

  // Listen for selection updates from server
  useEffect(() => {
    if (!socket) return;

    const handleUpdate = (data: { selectedWords: string[] }) => {
      // Enrich with canIntegrate status
      const enriched = data.selectedWords.map(word => ({
        word,
        ...checkWordIntegration(word, options.language)
      }));
      setSelectedWords(enriched);
    };

    socket.on('vocabularySelectionUpdated', handleUpdate);
    return () => { socket.off('vocabularySelectionUpdated', handleUpdate); };
  }, [socket, options.language]);

  const toggleWord = useCallback((word: string) => {
    if (!socket || !options.isHost || options.gameState !== 'finished') return;

    const isSelected = selectedWords.some(w => w.word === word);
    socket.emit('selectVocabularyWord', { word, include: !isSelected });
  }, [socket, options.isHost, options.gameState, selectedWords]);

  const saveAsLesson = useCallback(async (name: string, classroomId?: string) => {
    if (!socket || selectedWords.length === 0) return;

    setIsSaving(true);
    socket.emit('saveVocabularyLesson', { name, classroomId, words: selectedWords });

    return new Promise<void>((resolve, reject) => {
      socket.once('vocabularyLessonSaved', () => { setIsSaving(false); resolve(); });
      socket.once('vocabularyLessonError', (error) => { setIsSaving(false); reject(new Error(error)); });
    });
  }, [socket, selectedWords]);

  return {
    selectedWords,
    toggleWord,
    saveAsLesson,
    isSaving,
    canSelect: options.isHost && options.gameState === 'finished'
  };
}
```

```typescript
'use client';

import { useState, useMemo } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { Check, AlertTriangle, BookOpen, Save, X } from 'lucide-react';
import * as Dialog from '@radix-ui/react-dialog';
import { useVocabularySelection } from '@/hooks/useVocabularySelection';
import { useClassrooms } from '@/hooks/useClassroom';
import { cn } from '@/lib/utils';
import type { Language } from '@/shared/types';

interface WordItem {
  word: string;
  score: number;
  foundBy: string[];
}

interface HostWordSelectorProps {
  gameCode: string;
  language: Language;
  isHost: boolean;
  gameState: string;
  allWords: WordItem[]; // All words found during the game
  t: (key: string, params?: Record<string, string | number>) => string;
}

export function HostWordSelector({
  gameCode,
  language,
  isHost,
  gameState,
  allWords,
  t
}: HostWordSelectorProps) {
  const { selectedWords, toggleWord, saveAsLesson, isSaving, canSelect } = useVocabularySelection({
    gameCode,
    language,
    isHost,
    gameState
  });
  const { data: classrooms } = useClassrooms(/* teacherId from auth */);
  const [saveDialogOpen, setSaveDialogOpen] = useState(false);
  const [lessonName, setLessonName] = useState('');
  const [selectedClassroom, setSelectedClassroom] = useState<string | undefined>();

  // Only show if host and game is finished
  if (!canSelect) return null;

  // Sort words by score (highest first)
  const sortedWords = useMemo(() =>
    [...allWords].sort((a, b) => b.score - a.score),
    [allWords]
  );

  const handleSave = async () => {
    if (!lessonName.trim()) return;
    await saveAsLesson(lessonName, selectedClassroom);
    setSaveDialogOpen(false);
    setLessonName('');
  };

  return (
    <div className="bg-neo-navy/80 border-neo rounded-neo p-4 shadow-hard">
      <div className="flex items-center justify-between mb-4">
        <h3 className="font-neo-display text-lg text-neo-white flex items-center gap-2">
          <BookOpen className="w-5 h-5 text-neo-cyan" />
          {t('teacher.wordSelector.title')}
        </h3>
        {selectedWords.length > 0 && (
          <button
            onClick={() => setSaveDialogOpen(true)}
            className="btn-neo-primary text-sm px-3 py-1 flex items-center gap-2"
          >
            <Save className="w-4 h-4" />
            {t('teacher.wordSelector.saveAsLesson')} ({selectedWords.length})
          </button>
        )}
      </div>

      <div className="grid grid-cols-2 md:grid-cols-3 gap-2 max-h-64 overflow-y-auto">
        {sortedWords.map(({ word, score }) => {
          const isSelected = selectedWords.some(w => w.word === word);
          const integration = selectedWords.find(w => w.word === word);

          return (
            <motion.button
              key={word}
              onClick={() => toggleWord(word)}
              whileTap={{ scale: 0.95 }}
              className={cn(
                "flex items-center justify-between p-2 rounded-neo border-2 transition-colors",
                isSelected
                  ? "bg-neo-cyan/20 border-neo-cyan text-neo-white"
                  : "bg-neo-navy/50 border-gray-600 text-gray-300 hover:border-gray-400"
              )}
            >
              <span className="font-neo-body text-sm uppercase">{word}</span>
              <div className="flex items-center gap-1">
                <span className="text-xs text-gray-400">{score}</span>
                {isSelected && (
                  integration?.canIntegrate
                    ? <Check className="w-4 h-4 text-green-400" />
                    : <AlertTriangle className="w-4 h-4 text-yellow-400" />
                )}
              </div>
            </motion.button>
          );
        })}
      </div>

      {/* Save Lesson Dialog */}
      <Dialog.Root open={saveDialogOpen} onOpenChange={setSaveDialogOpen}>
        <Dialog.Portal>
          <Dialog.Overlay className="fixed inset-0 bg-black/60 z-50" />
          <Dialog.Content className="fixed top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 bg-neo-navy border-neo-thick rounded-neo p-6 shadow-hard-lg z-50 w-96">
            <Dialog.Title className="font-neo-display text-xl text-neo-white mb-4">
              {t('teacher.wordSelector.saveLessonTitle')}
            </Dialog.Title>

            <input
              type="text"
              value={lessonName}
              onChange={(e) => setLessonName(e.target.value)}
              placeholder={t('teacher.wordSelector.lessonNamePlaceholder')}
              className="w-full p-3 bg-neo-navy/50 border-neo rounded-neo text-neo-white mb-4"
            />

            {classrooms && classrooms.length > 0 && (
              <select
                value={selectedClassroom || ''}
                onChange={(e) => setSelectedClassroom(e.target.value || undefined)}
                className="w-full p-3 bg-neo-navy/50 border-neo rounded-neo text-neo-white mb-4"
              >
                <option value="">{t('teacher.wordSelector.noClassroom')}</option>
                {classrooms.map(c => (
                  <option key={c.id} value={c.id}>{c.name}</option>
                ))}
              </select>
            )}

            <div className="flex gap-3 justify-end">
              <Dialog.Close className="btn-neo-secondary px-4 py-2">
                {t('common.cancel')}
              </Dialog.Close>
              <button
                onClick={handleSave}
                disabled={isSaving || !lessonName.trim()}
                className="btn-neo-primary px-4 py-2"
              >
                {isSaving ? t('common.saving') : t('common.save')}
              </button>
            </div>
          </Dialog.Content>
        </Dialog.Portal>
      </Dialog.Root>
    </div>
  );
}
```
