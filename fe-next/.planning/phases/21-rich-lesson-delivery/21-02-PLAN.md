    'use client';

    import { useLanguage } from '@/contexts/LanguageContext';
    import { cn } from '@/lib/utils';
    import { Volume2, VolumeX, Loader2 } from 'lucide-react';

    interface PronunciationButtonProps {
      /** Callback when button clicked */
      onPress: () => void;
      /** Whether TTS is currently speaking */
      isSpeaking?: boolean;
      /** Whether TTS is supported in browser */
      isSupported?: boolean;
      /** Whether voice is available for the word's language */
      hasVoice?: boolean;
      /** Optional IPA to display as fallback */
      pronunciation?: string;
      /** Size variant */
      size?: 'sm' | 'md' | 'lg';
      /** Custom className */
      className?: string;
    }

    /**
     * Neo-Brutalist pronunciation button with TTS integration
     *
     * Shows speaker icon that animates when speaking.
     * Falls back to displaying IPA when voice not available.
     */
    export function PronunciationButton({
      onPress,
      isSpeaking = false,
      isSupported = true,
      hasVoice = true,
      pronunciation,
      size = 'md',
      className,
    }: PronunciationButtonProps) {
      const { t } = useLanguage();

      const sizeClasses = {
        sm: 'w-8 h-8 text-sm',
        md: 'w-10 h-10 text-base',
        lg: 'w-12 h-12 text-lg',
      };

      const iconSizes = {
        sm: 16,
        md: 20,
        lg: 24,
      };

      // Show IPA fallback when no voice available
      if (!hasVoice && pronunciation) {
        return (
          <div
            className={cn(
              'inline-flex items-center gap-2 px-3 py-1',
              'bg-neo-navy/50 border-2 border-neo-white/20 rounded-neo',
              'font-mono text-neo-white/70',
              className
            )}
            title={t('lesson.pronunciationFallback')}
          >
            <VolumeX size={iconSizes[size]} className="opacity-50" />
            <span>{pronunciation}</span>
          </div>
        );
      }

      // Not supported in browser
      if (!isSupported) {
        return null;
      }

      return (
        <button
          type="button"
          onClick={onPress}
          disabled={isSpeaking}
          className={cn(
            'inline-flex items-center justify-center',
            'bg-neo-cyan border-3 border-black rounded-neo',
            'shadow-hard hover:shadow-hard-sm active:shadow-hard-pressed',
            'transition-all duration-150',
            'disabled:opacity-50 disabled:cursor-not-allowed',
            sizeClasses[size],
            className
          )}
          aria-label={isSpeaking ? t('lesson.speaking') : t('lesson.pronounce')}
        >
          {isSpeaking ? (
            <Loader2 size={iconSizes[size]} className="animate-spin text-black" />
          ) : (
            <Volume2 size={iconSizes[size]} className="text-black" />
          )}
        </button>
      );
    }
    ```

    'use client';

    import { useLanguage } from '@/contexts/LanguageContext';
    import { cn } from '@/lib/utils';
    import { PronunciationButton } from './PronunciationButton';
    import { useSpeechSynthesis } from '@/hooks/useSpeechSynthesis';
    import type { EnrichedVocabularyWord, VocabularyExample } from '@/types/vocabulary';
    import { BookOpen, Lightbulb, Sparkles } from 'lucide-react';

    interface VocabularyCardEnrichedProps {
      word: EnrichedVocabularyWord;
      /** Show/hide definition (for flashcard flip) */
      showDefinition?: boolean;
      /** Compact mode for smaller cards */
      compact?: boolean;
      /** Custom className */
      className?: string;
    }

    /**
     * Enriched vocabulary card with definition, pronunciation, and examples
     *
     * Neo-Brutalist design with hard shadows and bold colors.
     * Supports TTS pronunciation and contextual examples from Daily Buzz.
     */
    export function VocabularyCardEnriched({
      word,
      showDefinition = true,
      compact = false,
      className,
    }: VocabularyCardEnrichedProps) {
      const { t, isRTL } = useLanguage();
      const { speak, isSpeaking, isSupported, hasVoice } = useSpeechSynthesis();

      const handlePronounce = () => {
        speak(word.word, word.language);
      };

      return (
        <div
          className={cn(
            'bg-neo-navy border-3 border-black rounded-neo',
            'shadow-hard p-6',
            compact && 'p-4',
            isRTL && 'text-right',
            className
          )}
        >
          {/* Word Header */}
          <div className="flex items-start justify-between gap-4 mb-4">
            <div className="flex-1">
              <h3 className={cn(
                'font-neo-display text-neo-yellow',
                compact ? 'text-2xl' : 'text-3xl'
              )}>
                {word.word}
              </h3>
              {word.partOfSpeech && (
                <span className="text-sm text-neo-white/60 italic">
                  {word.partOfSpeech}
                </span>
              )}
            </div>
            <PronunciationButton
              onPress={handlePronounce}
              isSpeaking={isSpeaking}
              isSupported={isSupported}
              hasVoice={hasVoice(word.language)}
              pronunciation={word.pronunciation}
              size={compact ? 'sm' : 'md'}
            />
          </div>

          {/* Definition */}
          {showDefinition && (
            <div className="mb-4">
              <div className="flex items-center gap-2 mb-2 text-neo-cyan">
                <BookOpen size={16} />
                <span className="text-sm font-semibold uppercase">
                  {t('lesson.definition')}
                </span>
              </div>
              <p className="text-neo-white text-lg">
                {word.definition}
              </p>
            </div>
          )}

          {/* Usage Examples */}
          {showDefinition && word.examples.length > 0 && (
            <ExamplesSection
              title={t('lesson.examples')}
              examples={word.examples}
              icon={<Lightbulb size={16} />}
              iconColor="text-neo-yellow"
              compact={compact}
            />
          )}

          {/* Contextual Examples from Daily Buzz */}
          {showDefinition && word.contextualExamples && word.contextualExamples.length > 0 && (
            <ExamplesSection
              title={t('lesson.contextualExamples')}
              examples={word.contextualExamples}
              icon={<Sparkles size={16} />}
              iconColor="text-neo-pink"
              compact={compact}
              showSource
            />
          )}
        </div>
      );
    }

    interface ExamplesSectionProps {
      title: string;
      examples: VocabularyExample[];
      icon: React.ReactNode;
      iconColor: string;
      compact?: boolean;
      showSource?: boolean;
    }

    function ExamplesSection({
      title,
      examples,
      icon,
      iconColor,
      compact,
      showSource,
    }: ExamplesSectionProps) {
      return (
        <div className={cn('mb-4', compact && 'mb-2')}>
          <div className={cn('flex items-center gap-2 mb-2', iconColor)}>
            {icon}
            <span className="text-sm font-semibold uppercase">{title}</span>
          </div>
          <ul className="space-y-2">
            {examples.slice(0, compact ? 1 : 3).map((example, idx) => (
              <li key={idx} className="text-neo-white/90">
                <p className={cn(compact ? 'text-sm' : 'text-base')}>
                  "{example.sentence}"
                </p>
                {showSource && example.source && (
                  <p className="text-xs text-neo-white/50 mt-1">
                    — {example.source}
                  </p>
                )}
              </li>
            ))}
          </ul>
        </div>
      );
    }
    ```
