    ```typescript
    /**
     * Daily Buzz Context Service
     *
     * Enriches vocabulary words with contextual examples from Daily Buzz
     * trending content. Uses fuzzy word matching to find relevant sentences.
     */

    import { getTodaysBuzzChallenge } from '@/backend/services/buzz/databaseService';
    import type { EnrichedVocabularyWord, VocabularyExample } from '@/types/vocabulary';

    const MAX_EXAMPLES = 3;

    /**
     * Normalize a word for fuzzy matching
     * Handles common English word variations (plural, tense, etc.)
     */
    export function normalizeWord(word: string): string {
      let normalized = word.toLowerCase().trim();

      // Only apply stemming to words longer than 4 characters
      if (normalized.length <= 4) {
        return normalized;
      }

      // Simple suffix removal for common variations
      // Order matters: check longer suffixes first
      const suffixes = [
        'ies',    // companies -> compani
        'ing',    // investing -> invest
        'tion',   // revolution -> revolu (keep short)
        'ed',     // invested -> invest
        's',      // companies (after ies check)
      ];

      for (const suffix of suffixes) {
        if (normalized.endsWith(suffix) && normalized.length - suffix.length >= 3) {
          // Don't stem if result would be too short
          normalized = normalized.slice(0, -suffix.length);
          break;
        }
      }

      return normalized;
    }

    /**
     * Split text into sentences
     */
    function splitIntoSentences(text: string): string[] {
      // Split on . ! ? followed by space or end of string
      return text
        .split(/[.!?]+\s*/)
        .map(s => s.trim())
        .filter(s => s.length > 0);
    }

    /**
     * Check if a sentence contains a word (with fuzzy matching)
     */
    function sentenceContainsWord(sentence: string, word: string): boolean {
      const normalizedWord = normalizeWord(word);
      const sentenceWords = sentence.toLowerCase().split(/\s+/);

      return sentenceWords.some(sentenceWord => {
        const normalizedSentenceWord = normalizeWord(sentenceWord.replace(/[^a-z]/gi, ''));
        return normalizedSentenceWord === normalizedWord ||
               normalizedSentenceWord.startsWith(normalizedWord) ||
               normalizedWord.startsWith(normalizedSentenceWord);
      });
    }

    /**
     * Find contextual examples from Daily Buzz for a vocabulary word
     *
     * @param word - The vocabulary word to find examples for
     * @returns Array of contextual examples (max 3)
     */
    export async function findContextualExamples(word: string): Promise<VocabularyExample[]> {
      const buzzChallenge = await getTodaysBuzzChallenge();

      if (!buzzChallenge || !buzzChallenge.trending_context) {
        return [];
      }

      const sentences = splitIntoSentences(buzzChallenge.trending_context);
      const matchingSentences: VocabularyExample[] = [];

      for (const sentence of sentences) {
        if (sentenceContainsWord(sentence, word)) {
          matchingSentences.push({
            sentence: sentence + '.',  // Add period back
            source: `Daily Buzz: ${buzzChallenge.trending_topic}`,
          });

          if (matchingSentences.length >= MAX_EXAMPLES) {
            break;
          }
        }
      }

      return matchingSentences;
    }

    /**
     * Enrich a vocabulary word with contextual examples from Daily Buzz
     *
     * @param word - The vocabulary word to enrich
     * @returns The word with contextualExamples added
     */
    export async function enrichVocabularyWithContext(
      word: EnrichedVocabularyWord
    ): Promise<EnrichedVocabularyWord> {
      const contextualExamples = await findContextualExamples(word.word);

      return {
        ...word,
        contextualExamples,
      };
    }

    /**
     * Enrich multiple vocabulary words with contextual examples
     * Processes in parallel for efficiency
     *
     * @param words - Array of vocabulary words to enrich
     * @returns Array of enriched words
     */
    export async function enrichVocabularyBatch(
      words: EnrichedVocabularyWord[]
    ): Promise<EnrichedVocabularyWord[]> {
      return Promise.all(words.map(enrichVocabularyWithContext));
    }
    ```

    ```typescript
    import { z } from 'zod';
    import { createHandler } from '../utils/handlerFactory';
    import { enrichVocabularyBatch } from '@/lib/services/dailyBuzzContextService';
    import type { EnrichedVocabularyWord } from '@/types/vocabulary';

    const VocabularyExampleSchema = z.object({
      sentence: z.string(),
      translation: z.string().optional(),
      source: z.string().optional(),
    });

    const EnrichedVocabularyWordSchema = z.object({
      word: z.string(),
      language: z.enum(['en', 'he', 'sv', 'ja', 'es']),
      definition: z.string(),
      partOfSpeech: z.string().optional(),
      pronunciation: z.string().optional(),
      audioUrl: z.string().optional(),
      examples: z.array(VocabularyExampleSchema),
      contextualExamples: z.array(VocabularyExampleSchema).optional(),
      relatedWords: z.array(z.string()).optional(),
      difficulty: z.number().optional(),
    });

    const vocabularyEnrichmentSchema = z.object({
      words: z.array(EnrichedVocabularyWordSchema).min(1).max(50),
    });

    /**
     * WebSocket handler for enriching vocabulary with Daily Buzz context
     *
     * @event vocabulary:enrich
     * @param words - Array of vocabulary words to enrich
     * @returns Array of enriched words with contextualExamples added
     */
    export const vocabularyEnrichmentHandler = createHandler(
      'vocabulary:enrich',
      vocabularyEnrichmentSchema,
      async (socket, data, context) => {
        const { words } = data;

        try {
          const enrichedWords = await enrichVocabularyBatch(words as EnrichedVocabularyWord[]);
          return { words: enrichedWords };
        } catch (error) {
          console.error('Vocabulary enrichment failed:', error);

          // Return original words with empty contextualExamples as fallback
          const fallbackWords = words.map(word => ({
            ...word,
            contextualExamples: [],
          }));

          return { words: fallbackWords };
        }
      }
    );
    ```
