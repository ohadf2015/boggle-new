    ```typescript
    /**
     * Text-to-Speech service using Web Speech API
     *
     * Provides pronunciation audio for vocabulary words.
     * Prefers local voices for lower latency.
     * Falls back gracefully when voices not available.
     */

    type LanguageCode = 'en' | 'he' | 'sv' | 'ja' | 'es';

    interface VoicesByLanguage {
      [key: string]: SpeechSynthesisVoice[];
    }

    /**
     * Get available voices grouped by language code
     */
    export function getAvailableVoices(): VoicesByLanguage {
      if (typeof window === 'undefined' || !window.speechSynthesis) {
        return {};
      }

      const voices = window.speechSynthesis.getVoices();
      const grouped: VoicesByLanguage = {};

      for (const voice of voices) {
        const langCode = voice.lang.split('-')[0];
        if (!grouped[langCode]) {
          grouped[langCode] = [];
        }
        grouped[langCode].push(voice);
      }

      // Sort each language group: local voices first
      for (const lang of Object.keys(grouped)) {
        grouped[lang].sort((a, b) => {
          if (a.localService && !b.localService) return -1;
          if (!a.localService && b.localService) return 1;
          return 0;
        });
      }

      return grouped;
    }

    /**
     * Find the best voice for a language
     * Prefers local voices over network voices for lower latency
     */
    function findVoice(langCode: LanguageCode): SpeechSynthesisVoice | null {
      const voices = getAvailableVoices();
      const langVoices = voices[langCode];

      if (!langVoices || langVoices.length === 0) {
        return null;
      }

      // Return first voice (already sorted by local preference)
      return langVoices[0];
    }

    /**
     * Speak a word using Web Speech API
     *
     * @param word - The word to speak
     * @param lang - Language code (defaults to 'en')
     * @returns true if speech started, false if no voice available
     */
    export function speakWord(word: string, lang: LanguageCode = 'en'): boolean {
      if (typeof window === 'undefined' || !window.speechSynthesis) {
        return false;
      }

      // Cancel any ongoing speech
      window.speechSynthesis.cancel();

      const voice = findVoice(lang);
      if (!voice) {
        return false;
      }

      const utterance = new SpeechSynthesisUtterance(word);
      utterance.voice = voice;
      utterance.lang = voice.lang;
      utterance.rate = 0.9; // Slightly slower for learning
      utterance.pitch = 1;
      utterance.volume = 1;

      window.speechSynthesis.speak(utterance);
      return true;
    }

    /**
     * Cancel any ongoing speech
     */
    export function cancelSpeech(): void {
      if (typeof window === 'undefined' || !window.speechSynthesis) {
        return;
      }
      window.speechSynthesis.cancel();
    }

    /**
     * Check if speech is currently playing
     */
    export function isSpeaking(): boolean {
      if (typeof window === 'undefined' || !window.speechSynthesis) {
        return false;
      }
      return window.speechSynthesis.speaking;
    }
    ```

    import { useState, useCallback, useEffect } from 'react';
    import { speakWord, cancelSpeech, isSpeaking as checkIsSpeaking, getAvailableVoices } from '../lib/speech/textToSpeech';

    type LanguageCode = 'en' | 'he' | 'sv' | 'ja' | 'es';

    interface UseSpeechSynthesisReturn {
      /** Speak a word in the specified language */
      speak: (word: string, lang?: LanguageCode) => boolean;
      /** Cancel any ongoing speech */
      cancel: () => void;
      /** Whether speech is currently playing */
      isSpeaking: boolean;
      /** Whether Web Speech API is supported */
      isSupported: boolean;
      /** Check if a voice is available for a language */
      hasVoice: (lang: LanguageCode) => boolean;
    }

    /**
     * React hook for Text-to-Speech functionality
     *
     * @example
     * ```tsx
     * const { speak, isSpeaking, isSupported } = useSpeechSynthesis();
     *
     * return (
     *   <button
     *     onClick={() => speak('hello', 'en')}
     *     disabled={!isSupported || isSpeaking}
     *   >
     *     {isSpeaking ? 'Speaking...' : 'Pronounce'}
     *   </button>
     * );
     * ```
     */
    export function useSpeechSynthesis(): UseSpeechSynthesisReturn {
      const [isSpeaking, setIsSpeaking] = useState(false);
      const [isSupported, setIsSupported] = useState(false);
      const [voices, setVoices] = useState<Record<string, unknown[]>>({});

      // Check browser support on mount
      useEffect(() => {
        const supported = typeof window !== 'undefined' && 'speechSynthesis' in window;
        setIsSupported(supported);

        if (supported) {
          // Voices may load asynchronously
          const loadVoices = () => {
            setVoices(getAvailableVoices());
          };

          loadVoices();
          window.speechSynthesis.addEventListener('voiceschanged', loadVoices);

          return () => {
            window.speechSynthesis.removeEventListener('voiceschanged', loadVoices);
          };
        }
      }, []);

      // Poll speaking state (Web Speech API doesn't have reliable events)
      useEffect(() => {
        if (!isSupported) return;

        const interval = setInterval(() => {
          setIsSpeaking(checkIsSpeaking());
        }, 100);

        return () => clearInterval(interval);
      }, [isSupported]);

      const speak = useCallback((word: string, lang: LanguageCode = 'en'): boolean => {
        if (!isSupported) return false;
        return speakWord(word, lang);
      }, [isSupported]);

      const cancel = useCallback(() => {
        if (!isSupported) return;
        cancelSpeech();
        setIsSpeaking(false);
      }, [isSupported]);

      const hasVoice = useCallback((lang: LanguageCode): boolean => {
        return Boolean(voices[lang] && (voices[lang] as unknown[]).length > 0);
      }, [voices]);

      return {
        speak,
        cancel,
        isSpeaking,
        isSupported,
        hasVoice,
      };
    }
