    import { useCallback, useMemo } from 'react';
    import { useMotionValue, useTransform, PanInfo, MotionValue } from 'framer-motion';

    export type SwipeDirection = 'left' | 'right';

    export interface SwipeConfig {
      threshold: number;        // Minimum swipe distance (default 150px)
      rotationRange: number;    // Max rotation angle (default 50 degrees)
    }

    interface UseSwipeGestureProps {
      onSwipe: (direction: SwipeDirection) => void;
      threshold?: number;
      rotationRange?: number;
      disabled?: boolean;
    }

    interface UseSwipeGestureReturn {
      // Motion values for animating card
      x: MotionValue<number>;
      rotate: MotionValue<number>;
      opacity: MotionValue<number>;

      // Config
      config: SwipeConfig;

      // Event handlers
      handleDragEnd: (event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => void;
      handleKeyDown: (event: KeyboardEvent) => void;

      // Derived state for UI feedback
      swipeDirection: MotionValue<SwipeDirection | null>;
      swipeProgress: MotionValue<number>;
    }

    /**
     * Hook for swipe gesture detection with Framer Motion
     *
     * Provides motion values for drag position, rotation, and opacity transforms.
     * Detects swipe completion when drag exceeds threshold.
     * Includes keyboard shortcuts for accessibility.
     *
     * @example
     * ```tsx
     * const { x, rotate, opacity, handleDragEnd } = useSwipeGesture({
     *   onSwipe: (dir) => handleAnswer(dir === 'right'),
     * });
     *
     * <motion.div
     *   drag="x"
     *   dragConstraints={{ left: 0, right: 0 }}
     *   onDragEnd={handleDragEnd}
     *   style={{ x, rotate, opacity }}
     * >
     *   {children}
     * </motion.div>
     * ```
     */
    export function useSwipeGesture({
      onSwipe,
      threshold = 150,
      rotationRange = 50,
      disabled = false,
    }: UseSwipeGestureProps): UseSwipeGestureReturn {
      // Motion value for drag position
      const x = useMotionValue(0);

      // Transform drag position to rotation (-range to +range based on drag)
      const rotate = useTransform(
        x,
        [-threshold * 2, 0, threshold * 2],
        [-rotationRange, 0, rotationRange]
      );

      // Transform drag position to opacity (fade at edges)
      const opacity = useTransform(
        x,
        [-threshold * 1.5, -threshold, 0, threshold, threshold * 1.5],
        [0, 1, 1, 1, 0]
      );

      // Derived values for UI feedback
      const swipeDirection = useTransform(x, (value): SwipeDirection | null => {
        if (value > threshold * 0.3) return 'right';
        if (value < -threshold * 0.3) return 'left';
        return null;
      });

      const swipeProgress = useTransform(x, (value) => {
        return Math.min(Math.abs(value) / threshold, 1);
      });

      // Handle drag end - detect swipe completion
      const handleDragEnd = useCallback(
        (event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => {
          if (disabled) {
            x.set(0);
            return;
          }

          const { offset } = info;

          if (offset.x > threshold) {
            // Swipe RIGHT → "Got It"
            onSwipe('right');
          } else if (offset.x < -threshold) {
            // Swipe LEFT → "Don't Know"
            onSwipe('left');
          } else {
            // Insufficient swipe - snap back to center
            x.set(0);
          }
        },
        [onSwipe, threshold, disabled, x]
      );

      // Keyboard shortcuts for accessibility
      const handleKeyDown = useCallback(
        (event: KeyboardEvent) => {
          if (disabled) return;

          switch (event.key) {
            case 'ArrowLeft':
              event.preventDefault();
              onSwipe('left');
              break;
            case 'ArrowRight':
              event.preventDefault();
              onSwipe('right');
              break;
            // No default - ignore other keys
          }
        },
        [onSwipe, disabled]
      );

      const config = useMemo(
        () => ({
          threshold,
          rotationRange,
        }),
        [threshold, rotationRange]
      );

      return {
        x,
        rotate,
        opacity,
        config,
        handleDragEnd,
        handleKeyDown,
        swipeDirection,
        swipeProgress,
      };
    }
    ```

    Key implementation details:
    - `useMotionValue(0)` for x position - doesn't trigger re-renders
    - `useTransform` maps x to rotate and opacity for smooth visual feedback
    - Default 150px threshold from research
    - Keyboard ArrowLeft/ArrowRight for accessibility
    - Disabled state prevents swipes during transitions
    - swipeDirection and swipeProgress for UI indicators (green/red glow)
