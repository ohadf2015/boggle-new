  <name>Task 2: Integrate Vocabulary Enrichment</name>
  <files>components/practice/FlashcardReview.tsx</files>
  <action>
    Add vocabulary enrichment integration:

    1. Import enrichment service and types:
    ```typescript
    import { enrichVocabularyBatch } from '@/lib/services/dailyBuzzContextService';
    import type { EnrichedVocabularyWord } from '@/types/vocabulary';
    ```

    2. Convert lesson words to enriched format:
    ```typescript
    const [enrichedWords, setEnrichedWords] = useState<EnrichedVocabularyWord[]>([]);
    const [isEnriching, setIsEnriching] = useState(false);

    useEffect(() => {
      async function enrichWords() {
        if (!words || words.length === 0) return;

        setIsEnriching(true);
        try {
          // Convert basic words to enriched format
          const basicEnriched: EnrichedVocabularyWord[] = words.map(word => ({
            word: word.word,
            language: 'en', // or from lesson language
            definition: word.definition || '',
            examples: word.examples?.map(ex => ({ sentence: ex })) || [],
          }));

          // Enrich with Daily Buzz context
          const fullyEnriched = await enrichVocabularyBatch(basicEnriched);
          setEnrichedWords(fullyEnriched);
        } catch (error) {
          console.error('Failed to enrich vocabulary:', error);
          // Fallback to basic words
          setEnrichedWords(words.map(w => ({
            word: w.word,
            language: 'en',
            definition: w.definition || '',
            examples: [],
          })));
        } finally {
          setIsEnriching(false);
        }
      }

      enrichWords();
    }, [words]);
    ```

    3. Show loading state while enriching:
    ```typescript
    {isEnriching && (
      <div className="text-center text-neo-white/60 py-4">
        <Loader2 className="animate-spin mx-auto mb-2" />
        {t('lesson.enrichingContent')}
      </div>
    )}
    ```
  </action>
  <verify>npm run lint && npm run build</verify>
  <done>Vocabulary enrichment integrated with Daily Buzz context</done>
</task>
