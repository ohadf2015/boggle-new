# AdMob Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement Google AdMob (rewarded, interstitial, banner) in LexiClash for Android and iOS via `@capacitor-community/admob`, with ads suppressed on web.

**Architecture:** `lib/admob-config.ts` (pure config) → `contexts/AdMobContext.tsx` (init, session counting, hasNoAds stub) → `hooks/useAdMob.ts` (ad-show functions) → `components/ads/AdBanner.tsx` (declarative banner). Provider mounts in `essential-providers.tsx` so it's available on all pages.

**Tech Stack:** `@capacitor-community/admob@8.0.0`, `@capacitor/core` (Capacitor.isNativePlatform()), React context + hooks, Vitest + @testing-library/react

// fe-next/lib/admob-config.ts

export type AdPlatform = 'android' | 'ios';

export interface AdmobConfig {
  rewardedAdId: string;
  interstitialAdId: string;
  bannerAdId: string;
}

const DEFAULTS = {
  android: {
    rewardedAdId: 'ca-app-pub-3940256099942544/5224354917',
    interstitialAdId: 'ca-app-pub-3940256099942544/1033173712',
    bannerAdId: 'ca-app-pub-3940256099942544/6300978111',
  },
  ios: {
    rewardedAdId: 'ca-app-pub-3940256099942544/1712485313',
    interstitialAdId: 'ca-app-pub-3940256099942544/4411468910',
    bannerAdId: 'ca-app-pub-3940256099942544/2934735716',
  },
} as const;

export function getAdmobConfig(platform: AdPlatform): AdmobConfig {
  const suffix = platform === 'android' ? 'ANDROID' : 'IOS';
  return {
    rewardedAdId:
      process.env[`NEXT_PUBLIC_ADMOB_REWARDED_${suffix}`] ??
      DEFAULTS[platform].rewardedAdId,
    interstitialAdId:
      process.env[`NEXT_PUBLIC_ADMOB_INTERSTITIAL_${suffix}`] ??
      DEFAULTS[platform].interstitialAdId,
    bannerAdId:
      process.env[`NEXT_PUBLIC_ADMOB_BANNER_${suffix}`] ??
      DEFAULTS[platform].bannerAdId,
  };
}

// fe-next/contexts/AdMobContext.tsx
'use client';

import { createContext, useContext, useEffect, useRef, ReactNode } from 'react';
import { Capacitor } from '@capacitor/core';
import { AdMob } from '@capacitor-community/admob';
import { getAdmobConfig, type AdPlatform } from '@/lib/admob-config';

interface AdMobContextValue {
  recordGameEnd: () => void;
  shouldShowInterstitial: () => boolean;
  hasNoAds: () => boolean;
  getConfig: () => ReturnType<typeof getAdmobConfig> | null;
}

const AdMobContext = createContext<AdMobContextValue | null>(null);

export function AdMobProvider({ children }: { children: ReactNode }) {
  const isNative = Capacitor.isNativePlatform();
  const platform = Capacitor.getPlatform() as AdPlatform;

  // Warmup: skip first 3 total game ends across all sessions
  const totalGameEnds = useRef(0);
  // Per-session game count for every-3rd-game trigger
  const sessionGameCount = useRef(0);
  const isInitialized = useRef(false);

  useEffect(() => {
    if (!isNative || isInitialized.current) return;
    isInitialized.current = true;

    const isDev = process.env.NODE_ENV !== 'production';
    AdMob.initialize({ initializeForTesting: isDev }).catch(() => {});
  }, [isNative]);

  function hasNoAds(): boolean {
    return false; // TODO: check IAP entitlement 'no_ads_unlock'
  }

  function recordGameEnd() {
    totalGameEnds.current += 1;
    sessionGameCount.current += 1;
  }

  function shouldShowInterstitial(): boolean {
    if (hasNoAds()) return false;
    if (totalGameEnds.current <= 3) return false; // warmup
    return sessionGameCount.current % 3 === 0;
  }

  function getConfig() {
    if (!isNative) return null;
    return getAdmobConfig(platform);
  }

  return (
    <AdMobContext.Provider value={{ recordGameEnd, shouldShowInterstitial, hasNoAds, getConfig }}>
      {children}
    </AdMobContext.Provider>
  );
}

export function useAdMobContext(): AdMobContextValue {
  const ctx = useContext(AdMobContext);
  if (!ctx) throw new Error('useAdMobContext must be used within AdMobProvider');
  return ctx;
}
```

// fe-next/hooks/useAdMob.ts
import { useCallback } from 'react';
import { AdMob, BannerAdSize, BannerAdPosition } from '@capacitor-community/admob';
import { useAdMobContext } from '@/contexts/AdMobContext';

export function useAdMob() {
  const { recordGameEnd, shouldShowInterstitial, hasNoAds, getConfig } = useAdMobContext();
  const isDev = process.env.NODE_ENV !== 'production';

  const showRewarded = useCallback(async (onReward: () => void) => {
    if (hasNoAds()) return;
    const config = getConfig();
    if (!config) return;

    try {
      await AdMob.prepareRewardVideoAd({ adId: config.rewardedAdId });
      await AdMob.showRewardVideoAd();
      onReward();
    } catch {
      // Ad failed to load or show — silent failure is intentional
    }
  }, [hasNoAds, getConfig]);

  const showInterstitial = useCallback(async () => {
    recordGameEnd();
    if (hasNoAds()) return;
    if (!shouldShowInterstitial()) return;
    const config = getConfig();
    if (!config) return;

    try {
      await AdMob.prepareInterstitial({ adId: config.interstitialAdId });
      await AdMob.showInterstitial();
    } catch {
      // Silent failure
    }
  }, [recordGameEnd, hasNoAds, shouldShowInterstitial, getConfig]);

  const showBanner = useCallback(async (position = BannerAdPosition.BOTTOM_CENTER) => {
    if (hasNoAds()) return;
    const config = getConfig();
    if (!config) return;

    try {
      await AdMob.showBanner({
        adId: config.bannerAdId,
        adSize: BannerAdSize.ADAPTIVE_BANNER,
        position,
        isTesting: isDev,
      });
    } catch {
      // Silent failure
    }
  }, [hasNoAds, getConfig, isDev]);

  const hideBanner = useCallback(async () => {
    try {
      await AdMob.hideBanner();
    } catch {
      // Already hidden
    }
  }, []);

  return { showRewarded, showInterstitial, showBanner, hideBanner };
}
```

// fe-next/components/ads/AdBanner.tsx
'use client';

import { useEffect } from 'react';
import { Capacitor } from '@capacitor/core';
import { useAdMob } from '@/hooks/useAdMob';

// Height of ADAPTIVE_BANNER on most devices (~50dp)
const BANNER_HEIGHT = 50;

export function AdBanner() {
  const isNative = Capacitor.isNativePlatform();
  const { showBanner, hideBanner } = useAdMob();

  useEffect(() => {
    if (!isNative) return;
    showBanner();
    return () => { hideBanner(); };
  }, [isNative, showBanner, hideBanner]);

  if (!isNative) return null;

  return (
    <div
      data-testid="ad-banner-spacer"
      style={{ height: BANNER_HEIGHT }}
      aria-hidden="true"
    />
  );
}
