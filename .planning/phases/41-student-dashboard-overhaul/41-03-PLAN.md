    Add useEffect to fetch duel data alongside existing achievements fetch:
    ```typescript
    useEffect(() => {
      async function fetchDuelData() {
        if (!user) { setIsLoadingDuels(false); return; }
        const [statsResult, historyResult] = await Promise.all([
          getDuelStats(user.id),
          getDuelHistory(user.id, 5), // Last 5 duels
        ]);
        if (statsResult.data) setDuelStats(statsResult.data);
        if (historyResult.data) setRecentDuels(historyResult.data);
        setIsLoadingDuels(false);
      }
      fetchDuelData();
    }, [user]);
    ```
