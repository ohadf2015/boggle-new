```typescript
/**
 * Duel Namespace Socket.IO Event Types
 * Defines client-to-server and server-to-client events for /duel namespace
 *
 * Room naming convention: `duel:${duelId}` (prefixed to avoid collision with default namespace)
 * Lobby room: `duel:lobby:${classroomId}` (for browsing available duels)
 */

export interface DuelClientEvents {
  // Duel lifecycle
  'duel:create': (data: { lessonId: string; opponentId: string; duelType: 'async' | 'realtime' }) => void;
  'duel:accept': (data: { duelId: string }) => void;
  'duel:decline': (data: { duelId: string }) => void;
  'duel:cancel': (data: { duelId: string }) => void;

  // Gameplay
  'duel:submit-word': (data: { duelId: string; word: string }) => void;
  'duel:submit-score': (data: { duelId: string; score: number; wordsFound: string[] }) => void;

  // Lobby
  'duel:join-lobby': (data: { classroomId: string }) => void;
  'duel:leave-lobby': () => void;

  // Presence
  'duel:join-room': (data: { duelId: string }) => void;
  'duel:leave-room': (data: { duelId: string }) => void;
}

export interface DuelServerEvents {
  // Duel lifecycle
  'duel:created': (data: { duelId: string; challengerId: string; opponentId: string }) => void;
  'duel:accepted': (data: { duelId: string }) => void;
  'duel:declined': (data: { duelId: string }) => void;
  'duel:cancelled': (data: { duelId: string }) => void;
  'duel:completed': (data: { duelId: string; winnerId: string; scores: Record<string, number> }) => void;

  // Real-time gameplay
  'duel:opponent-progress': (data: { wordCount: number; score: number }) => void;
  'duel:opponent-word': (data: { word: string }) => void;

  // Lobby updates
  'duel:lobby-update': (data: { activeDuels: Array<{ duelId: string; status: string }> }) => void;
  'duel:challenge-received': (data: { duelId: string; challengerName: string; lessonName: string }) => void;

  // Errors
  'duel:error': (data: { code: string; message: string }) => void;
}
```

```typescript
/**
 * Duel Namespace Handler Registry
 * Registers all duel-specific event handlers for the /duel Socket.IO namespace.
 *
 * Architecture:
 * - Namespace: /duel (isolated from default namespace /)
 * - Room naming: duel:${duelId} for game rooms, duel:lobby:${classroomId} for lobbies
 * - All events prefixed with 'duel:' to prevent naming conflicts
 * - Middleware: authentication required (added in socketSetup.ts)
 *
 * TODO: Implement handler bodies in Phase 38 (async) and Phase 39 (real-time)
 */

import type { Namespace, Socket } from 'socket.io';
import type { DuelClientEvents, DuelServerEvents } from './types';

/**
 * Register all duel-specific event handlers
 * Pattern matches registerAllHandlers in handlers/index.ts
 */
export function registerDuelHandlers(
  namespace: Namespace,
  socket: Socket
): void {
  // --- Duel Lifecycle ---

  socket.on('duel:create', (data) => {
    // TODO (Phase 38): Create duel record, notify opponent via lobby room

  });

  socket.on('duel:accept', (data) => {
    // TODO (Phase 38): Update duel status, join both players to duel room
    console.log(`[DUEL] Accept request from ${socket.id}`, data);
  });

  socket.on('duel:decline', (data) => {
    // TODO (Phase 38): Update duel status, notify challenger
    console.log(`[DUEL] Decline from ${socket.id}`, data);
  });

  socket.on('duel:cancel', (data) => {
    // TODO (Phase 38): Cancel pending duel
    console.log(`[DUEL] Cancel from ${socket.id}`, data);
  });

  // --- Gameplay ---

  socket.on('duel:submit-word', (data) => {
    // TODO (Phase 39): Validate word, broadcast to opponent in duel room
    console.log(`[DUEL] Word submitted in duel ${data.duelId}`, data.word);
  });

  socket.on('duel:submit-score', (data) => {
    // TODO (Phase 38/39): Record turn score, check for duel completion
    console.log(`[DUEL] Score submitted for duel ${data.duelId}`, data.score);
  });

  // --- Lobby ---

  socket.on('duel:join-lobby', (data) => {
    const lobbyRoom = `duel:lobby:${data.classroomId}`;
    socket.join(lobbyRoom);
    console.log(`[DUEL] ${socket.id} joined lobby ${lobbyRoom}`);
  });

  socket.on('duel:leave-lobby', () => {
    // Leave all lobby rooms
    for (const room of socket.rooms) {
      if (room.startsWith('duel:lobby:')) {
        socket.leave(room);
      }
    }
  });

  // --- Room Management ---

  socket.on('duel:join-room', (data) => {
    const duelRoom = `duel:${data.duelId}`;
    socket.join(duelRoom);
    console.log(`[DUEL] ${socket.id} joined room ${duelRoom}`);
  });

  socket.on('duel:leave-room', (data) => {
    const duelRoom = `duel:${data.duelId}`;
    socket.leave(duelRoom);
    console.log(`[DUEL] ${socket.id} left room ${duelRoom}`);
  });

  // --- Disconnect Cleanup ---

  socket.on('disconnect', () => {
    // Socket.IO automatically removes socket from all rooms on disconnect
    console.log(`[DUEL] ${socket.id} disconnected, rooms cleaned up`);
  });
}

export type { DuelClientEvents, DuelServerEvents };
```
